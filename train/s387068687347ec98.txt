typedef long long ll;

map<ll, ll> prime_facto(ll n)
{
    map<ll, ll> res;
    for (ll i = 2; i * i <= n; i++)
    {
        while (n % i == 0)
        {
            n /= i;
            res[i]++;
        }
    }

    if (n != 1)
    {
        res[n] = 1;
    }

    return res;
}

void solve1()
{
    ll n, p;
    cin >> n >> p;

    map<ll, ll> num;
    num = prime_facto(p);

    ll maxnum = 1;
    for (auto itr = num.begin(); itr != num.end(); itr++)
    {
        if (itr->second >= n)
        {
            maxnum *= itr->first*(itr->second / n);
        }
    }

    cout << maxnum << endl;
}

int main()
{
    solve1();
}
