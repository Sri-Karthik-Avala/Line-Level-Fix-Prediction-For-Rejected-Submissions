mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());

int P[100005], n;
int main(){
    fio;
    cin >> n;
    if((n & (n-1)) == 0){
        cout << "No\n";
        return 0;
    }
    cout << "Yes\n";
    vector<pii> E;
    int x = 1;
    for(; x-1 <= n; x <<= 1); x >>= 1;
    debug("x = %d\n",x);
    for(int i = 1; i < x; i++) P[i] = i;
    for(int i = x; i < n; i += 2){
        E.ep(i, i+1);
        E.ep(i+1, 1);
        E.ep(n + i, n + i + 1);
        E.ep(n + i, n + 1);
    }
    if(~n & 1){
        int p = n ^ (n-1);
        swap(P[p], P[2]);
        E.ep(n, n-1);
        E.ep(n + p, n + n);
    }
    for(int i = 1; i < x-1; i++){
        E.ep(P[i], P[i+1]);
        E.ep(P[i] + n, P[i+1] + n);
    }
    E.ep(P[x-1], n + P[1]);
    debug("E = %d\n",szz(E));
    for(pii p : E) cout << p.va << ' ' << p.vb << '\n';
    return 0;
}
