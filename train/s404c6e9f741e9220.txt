#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (int)(n); i++)
int main(){
  int n;
  cin >> n;
  vector<int> vec(n);
  int sum=0;
  rep(i,n){
    cin >> vec.at(i);
    sum+=vec.at(i);
  }
  sort(vec.begin(),vec.end());
  int i=0;
  while(sum%10==0 && i<n){
    if(vec.at(i)%10!=0){
    sum=sum-vec.at(i);
    }
    i++;
  }
  cout << sum;
}
