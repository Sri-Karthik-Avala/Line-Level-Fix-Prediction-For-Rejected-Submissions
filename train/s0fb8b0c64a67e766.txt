#!usr/bin/env python
#-*- coding: utf-8 -*-

while True:
    (m, nmin, nmax) = map(int, raw_input().split())
    if m == 0:
        break

    scores = []
    for i in range(m):
        scores.append(int(raw_input()))
    
    submax = 0
    for i in range(nmin, nmax):
        subtmp = scores[i-1] - scores[i]
        if subtmp >= submax:
            ans = i
            submax = subtmp

    print ans
