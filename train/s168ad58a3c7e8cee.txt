#include <iostream>
#include <cstdio>
#include <algorithm>
#include <cstring>
using namespace std;
const int MOD = 1e9 + 7;
int n, m, a[20];
long long dp[20][200001], fac[200001], inv[200001];
long long ksm(long long a, int b)
{
    long long s = 1;
    while (b)
    {
        if (b & 1)
        {
            s = s * a % MOD;
        }
        a = a * a % MOD;
        b >>= 1;
    }
    return s;
}
void init()
{
    fac[0] = 1;
    for (int i = 1; i <= 200000; i++)
    {
        fac[i] = fac[i - 1] * i % MOD;
    }
    inv[200000] = ksm(fac[200000], MOD - 2);
    for (int i = 199999; i >= 0; i--)
    {
        inv[i] = inv[i + 1] * (i + 1) % MOD;
    }
}
long long C(long long a, long long b)
{
    return fac[a] * inv[a - b] % MOD * inv[b] % MOD;
}
int main(int argc, char const *argv[])
{
    scanf("%d%d", &n, &m);
    for (int i = 1; i <= m; i++)
    {
        scanf("%d", &a[m - i + 1]);
    }
    init();
    dp[0][0] = 1;
    for (int i = 0; i < m; i++)
    {
        for (int S = 0; S < (1 << n); S++)
        {
            if (dp[i][S])
            {
                dp[i + 1][S] = (dp[i + 1][S] + dp[i][S]) % MOD;
                for (int j = 0; j < n; j++)
                {
                    if (!((S >> j) & 1))
                    {
                        dp[i + 1][S | (1 << j)] = (dp[i + 1][S | (1 << j)] + dp[i][S] * fac[(1 << j)] % MOD * C((1 << n) - a[i + 1] - S, (1 << j) - 1) % MOD) % MOD;
                    }
                }
            }
        }
    }
    long long ans = 0;
    for (int S = 0; S < (1 << n); S++)
    {
        int f = __builtin_popcount(S) & 1 ? -1 : 1;
        ans = (ans + f * dp[m][S] * fac[(1 << n) - S - 1] % MOD) % MOD;
    }
    printf("%lld", (ans * (1 << n) % MOD + MOD) % MOD);
    return 0;
}
