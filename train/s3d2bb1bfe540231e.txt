#define FOR(I,A,B) for(int I = (A); I < (B); ++I)
#define CLR(mat) memset(mat, 0, sizeof(mat))
typedef long long ll;
int main()
{
  ios::sync_with_stdio(false);
  cin.tie(0);
  int N; cin >> N;
  int a[N]; FOR(i,0,N) cin >> a[i];
  int ans = 0;
  FOR(i,0,N) {
    FOR(j,i+1,N) {
      int pro = a[i] * a[j];
      int x = 10;
      bool ok = true;
      while(pro) {
        if(pro % 10 != x - 1) {
          ok = false;
          break;
        }
        x = pro % 10;
        pro /= 10;
      }
      if(ok) ans = max(ans, a[i] * a[j]);
    }
  }
  if(ans == 0) ans = -1;
  cout << ans << endl;
  return 0;
}
