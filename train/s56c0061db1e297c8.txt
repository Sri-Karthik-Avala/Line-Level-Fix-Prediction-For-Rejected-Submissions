template <typename T>
istream &operator>>(istream &is, vector<T> &v) {
    for (int i = 0; i < v.size(); ++i) {
        is >> v[i];
    }
    return is;
}
template <typename T>
ostream &operator<<(ostream &os, const vector<T> &v) {
    if (v.size() == 0) return os;
    for (int i = 0; i < v.size() - 1; ++i) {
        os << v[i] << " ";
    }
    os << v[v.size() - 1];
    return os;
}

void solve() {
    int n;
    cin >> n;
    vint a(n);
    cin >> a;
    vint cnts(3, 0);
    ll ans = 1;
    REP(i, n) {
        int now = a[i], cnt = 0, add_j;
        REP(j, 3) {
            if (cnts[j] == now) {
                ++cnt;
                add_j = j;
            }
        }
        ans = (ans * cnt) % MOD;
        cnts[add_j]++;
    }
    cout << ans << endl;
}


int main() {
    solve();
    char tmp;
    while (cin >> tmp) {
        cin.putback(tmp);
        solve();
    }
}
