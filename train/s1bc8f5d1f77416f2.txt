#include<bits/stdc++.h>
using namespace std;
int main(){
  cout<<fixed<<setprecision(12);
  int n;
  int s[100];
  double sum,ans;
  while(1){
  cin>>n;
  if(n==0) break;
  sum=0;
  ans=0;
  for(int i=0;i<n;i++){
    cin>>s[i];
    sum+=s[i];
  }
  sum/=n;
  for(int i=0;i<n;i++){
    ans+=pow(sum-s[i],2);
  }
  ans=sqrt(ans/n);
  cout<<ans<<endl;
  }
  return 0;
}
