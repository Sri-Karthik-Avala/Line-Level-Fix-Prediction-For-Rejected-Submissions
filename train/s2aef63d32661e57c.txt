#include<bits/stdc++.h>
using namespace std;
typedef long long int ll;
const int maxn=1E6+5;
int n;
ll ans;
string str[maxn];
int tot,ch[maxn][26],have[maxn][27];
int bucket[26];
inline void insert(string s)
{
	int p=0;
	for(int i=s.size()-1;i>=0;--i)
		++bucket[s[i]-'a'];
	for(int i=s.size()-1;i>=0;--i)
	{
		int x=s[i]-'a';
		if(ch[p][x])
			p=ch[p][x];
		else
		{
			ch[p][x]=++tot;
			p=tot;
		}
		--bucket[x];
		for(int j=0;j<26;++j)
			if(bucket[j])
				++have[p][j];
		if(i==1)
		{
			x=s[0]-'a';
			for(int j=0;j<26;++j)
				ans+=have[ch[p][j]][x];
		}
		if(i==0)
			ans+=have[p][26];
		++have[p][26];
	}
	if(s.size()==1)
		for(int i=0;i<26;++i)
			if(i!=s[0]-'a')
				ans+=have[ch[0][i]][s[0]-'a'];
	++have[p][26];
}
inline void solve()
{
	tot=0;
	memset(ch,0,sizeof(ch));
	memset(have,0,sizeof(have));
	for(int i=1;i<=n;++i)
		insert(str[i]);
}
int main()
{
	ios::sync_with_stdio(false);
	cin>>n;
	for(int i=1;i<=n;++i)
		cin>>str[i];
	solve();
	reverse(str+1,str+n+1);
	solve();
	cout<<ans<<endl;
	return 0;
}
