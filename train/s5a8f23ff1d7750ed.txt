#include <algorithm>
#include <iostream>
#include <tuple>
#include <vector>
#define rep(i, n) for (int i = 0; i < (n); i++)
using namespace std;

int main(void) {
  ios::sync_with_stdio(false);
  int N, M;
  int64_t P;
  cin >> N >> M >> P;
  vector<tuple<int, int, int64_t>> E(M);
  rep(i, M) {
    int A, B;
    int64_t C;
    cin >> A >> B >> C;
    A--, B--;
    E.emplace_back(B, A, P - C);
  }
  sort(E.begin(), E.end());

  // Bellman-Ford
  vector<int64_t> dist(N, INT64_MAX);
  dist[0] = 0;
  rep(i, N - 1) {
    for (auto& [to, fr, cost] : E) {
      if (dist[fr] == INT64_MAX) continue;
      dist[to] = min(dist[to], dist[fr] + cost);
    }
  }
  // one-more loop
  bool update = false;
  for (auto& [to, fr, cost] : E) {
    if (dist[fr] == INT64_MAX) continue;
    if ((dist[fr] + cost) < dist[to] && (to == N - 1)) update = true;
    dist[to] = min(dist[to], dist[fr] + cost);
  }

  if (update)
    cout << -1 << endl;
  else if (0 <= dist[N - 1])
    cout << 0 << endl;
  else
    cout << -dist[N - 1] << endl;
  return 0;
}
