#include<bits/stdc++.h>
using namespace std;
const int N=20,M=1e9+7;
int n,m,Q,x,y,flag[N][N],ne[1<<N][N],g[1<<N],dp[1<<N];
inline void reduce(int &x){if (x>=M)x-=M;}
int main(){
	scanf("%d%d%d",&n,&m,&Q);
	n--;
	while (Q--){
		scanf("%d%d",&x,&y);
		y--;
		scanf("%d",&flag[x][y]);
		flag[x][y]++;
	}
	dp[0]=1;
	for (int i=0;i<1<<n;i++)
		for (int j=n-1;j>=0;j--)
			if (i&(1<<(j+1)))ne[i][j]=1<<(j+1);
			else ne[i][j]=ne[i][j+1];
	for (int i=1;i<=m;i++)
		for (int j=0;j<n;j++){
			for (int k=0;k<1<<n;k++)
				if (flag[i][j]!=2){
					if (!(k&(1<<j)))reduce(g[k]+=dp[k]);
				}
			for (int k=0;k<1<<n;k++)
				if (flag[i][j]!=1){
					if (!(k&(1<<j)))reduce(g[k^(1<<j)^(ne[k][j])]+=dp[k]);
					else reduce(g[k]+=dp[k]);
				}
			for (int k=0;k<1<<n;k++)dp[k]=g[k],g[k]=0;
		}
	int ans=0;
	for (int i=0;i<1<<n;i++)reduce(ans+=dp[i]);
	printf("%d\n",ans);
}
