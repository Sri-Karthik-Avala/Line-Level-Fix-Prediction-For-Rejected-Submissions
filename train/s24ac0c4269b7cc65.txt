using namespace std;

typedef pair<long long int, long long int> P;

long long int INF = 1e18;
long long int MOD = 1e9 + 7;

unordered_map<string, bool> m;
string str = "";

bool func(int pos, int num){
    if(pos == num){
        if(m.find(str) == m.end()){
            cout << str << endl;
            return true;
        }
        return false;
    }
    for(int i = 0; i < 26; i++){
        str[pos] = 'a' + i;
        if(func(pos + 1, num)){
            return true;
        }
    }
    return false;
}

int main(){
    int N;
    cin >> N;
    for(int i = 0; i < N; i++){
        string S;
        cin >> S;
        for(int i = 1; i <= 4; i++){
            for(int j = 0; j < S.size() - i + 1; j++){
                m[S.substr(j, i)] = true;
            }
        }
    }
    for(int i = 1; i <= 4; i++){
        str += " ";
        if(func(0, i)){
            return 0;
        }
    }
    return 0;
}
