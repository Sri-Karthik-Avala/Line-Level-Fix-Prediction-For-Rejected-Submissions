#include <queue>
#define REP(i,n) for(int i=0; i<(int)(n); i++)

#include <set>
#include <cstdio>
inline int getInt(){ int s; scanf("%d", &s); return s; }

using namespace std;

int g[10][10];
int cnt[10];

int main(){
  const int n = getInt();
  const int m = getInt();

  REP(i,m){
    const int a = getInt() - 1;
    const int b = getInt() - 1;
    g[a][b] = g[b][a] = 1;
    cnt[a]++; cnt[b]++;
  }

  const int lim = 1000;
  double ans = 0.0;

  vector<vector<double> > dp(1 << n, vector<double>(n));
  dp[1][0] = 1.0;

  REP(cc, lim){
    vector<vector<double> > next(1 << n, vector<double>(n));

    REP(i,1<<n) REP(j,n) if(dp[i][j] > 0.0){
      if(i == (1 << n) - 1){
        ans += dp[i][j] * cc;
      }else{
        REP(k,n) if(g[j][k]){
          next[i | (1 << k)][k] += dp[i][j] / cnt[j];
        }
      }
    }

    dp.swap(next);
  }

  printf("%.8f\n", ans);

  return 0;
}
