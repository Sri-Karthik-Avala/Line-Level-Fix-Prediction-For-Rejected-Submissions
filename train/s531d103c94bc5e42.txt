#include <iostream>
#include <algorithm>
#include <iomanip>
using namespace std;
int main(void){
    int n,p[200000];
    cin>>n;
    n*=2;
    for(int i=0;i<n;i++){
        cin>>p[i];
    }
    sort(p,p+n);
    double u=0,m=n-1;
    for(int i=0;i<n;i++){
        u+=p[i]*i;
    }
    u/=m;
    cout<<setprecision(24)<<u<<endl;
}
