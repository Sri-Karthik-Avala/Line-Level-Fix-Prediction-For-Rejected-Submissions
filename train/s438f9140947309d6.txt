template<typename C>
struct Trie {
    struct Node {
        int exist,len;
        vector<int> accept;
        map<C, int> ch;
        Node() : exist(0), len(0) {}  
    };

    vector<Node> node;

    Trie() {node.emplace_back();}

    template<typename S>
    void add (const S &A, int id) {
        int cur = 0;
        for (int i = 0; i < (int)A.size(); i++) {
            if (!node[cur].ch.count(A[i])) {
                int nex = node.size();
                node.emplace_back();
                node[nex].len = node[cur].len + 1;
                node[cur].ch.emplace(A[i],nex);
            }
            node[cur].exist++;
            cur = node[cur].ch[A[i]];
        }
        node[cur].exist++;
        node[cur].accept.emplace_back(id);
    }

    template<typename S>
    int find(const S &A) {
        int cur = 0;
        for (int i = 0; i < (int)A.size(); i++) {
            if (!node[cur].ch.count(A[i])) return -1;
            cur = node[cur].ch[A[i]];
        }
        return cur;
    }

    int count() const {return node[0].eixst;}

    int size() const {return(int)node.size();}
};

int underbit(ll x) {
    if (x==0) return 0;
    rep(i,64) {
        if ((x>>i)&1) return i+1;
    }
}

int main() {
    ios::sync_with_stdio(false); cin.tie(nullptr);
    ll N,L; cin >> N >> L;
    Trie<char> T;

    ll g = 0;
    rep(_,N) {
        string S; cin >> S;
        int K = S.size();
        int cur = 0, len = 0;
        rep(i,K) {
            if (!T.node[cur].ch.count(S[i])) {
                len = T.node[cur].len;
                break;
            }
            cur = T.node[cur].ch[S[i]];
        }
        ll val = L - len;
        g ^= underbit(val);
        ll l = L - K + 1;
        for (ll i = l; i < val; i++) g ^= i;
        T.add(S,_);
    }

    cout << (g ? "Alice" : "Bob") << ln;
}
