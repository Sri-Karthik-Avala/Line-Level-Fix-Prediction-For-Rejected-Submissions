int digit_sum(int n){
	int ret = 0;
	while(n > 0){
		ret += n%2;
		n >>= 1;
	}
	return ret;
}

ll n,m;
ll a[22];
ld p[22];
ll gcd(ll x,ll y){
	if(y == 0)return x;
	return gcd(y,x%y);
}
ll lcm(ll x,ll y){
	if(x == -1)return x;
	x /= gcd(x,y);
	if(log((ld)x)+log((ld)y) > log((ld)m))return -1;
	return x*y;
}

int main(){
	cin >> n >> m;
	rep(i,n)cin >> a[i];
	rep(i,n)cin >> p[i];
	rep(i,n)p[i] /= 100.0;
	
	ld ret = 0.0;
	rep1(i,(1<<n)-1){
		ll LCM = 1;
		ld pp = 1.0;
		rep(j,n){
			if((i>>j)&1){
				LCM = lcm(LCM,a[j]);
				pp *= p[j];
			}
		}
		if(LCM != -1)ret += ((digit_sum(i)%2==1)?1.0:(-1.0))*(m/LCM)*pp;
	}
	
	printf("%.10llf\n", ret);
