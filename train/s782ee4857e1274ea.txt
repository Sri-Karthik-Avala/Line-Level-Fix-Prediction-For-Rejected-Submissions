a = 0
b = 0

n = int(raw_input())

while n > 0:
    for _ in range(n):
        t_a, t_b = map(int, raw_input().split())
        if t_a > t_b:
            a += t_a + t_b
        elif t_a < t_b:
            b += t_a + t_b
        else:
            a += t_a
            b += t_b
    print "%d %d" % (a, b)
    n = int(raw_input())
