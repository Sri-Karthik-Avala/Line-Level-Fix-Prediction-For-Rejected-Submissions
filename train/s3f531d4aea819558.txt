char tmp_str[2][100005];

int solve(){
  string s,t;
  cin>>s>>t;

  if(s.size() == t.size()){
    if(s==t) return 1;
    else return 0;
  }

  int r = 1;
  int cur = t.size();
  while(s.size() >= cur){
    cur += (cur+1)*t.size();
    r++;
  }

  auto f = [&](int x){
    if(x==0) return true;
    auto s1 = tmp_str[0];
    auto s2 = tmp_str[1];
    int p_size = 0;
    int q_size = 0;
    rep(i,x){
      q_size = 0;
      rep(j,p_size) s2[q_size++] = s1[j];
      rep(j,t.size()){
        s2[q_size++] = t[j];
        rep(k,p_size) s2[q_size++] = s1[k];
      }
      swap(s1, s2);
      swap(p_size, q_size);
    }
    // dbg(x,string(s1, s1+p_size));
    int is=0,it=0;
    while(is<s.size() && it<p_size){
      if(s[is] == s1[it]){
        is++; it++;
      } else {
        is++;
      }
    }
    return it == p_size;
  };

  int l=0; r++;
  while(r-l>1){
    int mid = (l+r)/2;
    if(f(mid)) l = mid;
    else r = mid;
  }
  return l;
}
