const int MAXN = 210000;

ll x[MAXN];
ll y[MAXN];
int n;

int cnk(ll a, ll b) {
	if (b < 0 || b > a)
		return 0;
	return (a & b) == b;
}

int get(ll t) {
	int ans = 0;
	for (int i = 0; i < n; ++i) {
		if (y[i] > 0) {
			if (x[i] <= t && t <= x[i] + y[i]) {
				ans ^= cnk(y[i] - 1, t - x[i]);
			}
		}
		else if (y[i] == 0) {
			if (x[i] <= t)
				ans ^= 1;
		}
		else {
			if (t >= x[i]) {
				ans ^= cnk(-y[i] + (t - x[i]) + 1, t - x[i]);
			}
		}
	}
	return ans;
}

int main() {
#ifdef FASTIO
	ios_base::sync_with_stdio(false), cin.tie(0), cout.tie(0);
#endif
	cin >> n;
	for (int i = 0; i < n; ++i) {
		cin >> x[i] >> y[i];
	}
	ll lb = -1e18;
	ll rb = 1e18;
	while (rb - lb > 1) {
		ll mid = (lb + rb) >> 1;
		if (get(mid) == 1)
			rb = mid;
		else
			lb = mid;
	}
	cout << rb << "\n";
	return 0;
}
