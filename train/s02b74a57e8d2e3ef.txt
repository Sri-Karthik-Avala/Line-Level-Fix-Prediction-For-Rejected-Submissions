#include <iostream>
#include <cstdio>
#include <cstring>
#include <cstdlib>
#include <algorithm>
#include <string>
#include <queue>
#include <utility>
using namespace std;
int main() {
	string S;
	while(cin>>S) {
		int J=0,I=0;
		for(int i=0;i<S.size()-2;i++) {
			if(S[i+1]!='O'||S[i+2]!='I') continue;
			if(S[i]=='J') {
				J++;
			}else {
				I++;
			}
		}
		printf("%d\n%d\n",J,I);
	}
}
