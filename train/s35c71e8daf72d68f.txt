#include<iostream>
#include<string>
#include<algorithm>
#define N 50

using namespace std;

int n, m;
int data[N][N];

int char_to_int(const char& c){
  int x = c - 48;
  return x;
}

void solve(){
  for(int j=1; j<m; ++j){
    data[0][j] += data[0][j-1];
  }
  for(int i=1; i<n; ++i){
    data[i][0] += data[i-1][0];
    for(int j=0; j<m; ++j){
      data[i][j] += min(data[i-1][j], data[i][j-1]);
    }
  }
}

int main(){
  string s;
  cin >> n >> m;
  for(int i=0; i<n; ++i){
    cin >> s;
    for(int j=0; j<m; ++j){
      data[i][j] = char_to_int(s[j]);
    }
  }
  solve();
  cout << data[n-1][m-1] << endl;
  return 0;
}
