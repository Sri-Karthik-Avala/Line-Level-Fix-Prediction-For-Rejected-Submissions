#include<iostream>
#include<sstream>
#include<vector>
#include<map>
#include<algorithm>
using namespace std;
typedef pair<int,string> P;
vector<string> t;
vector<P> u;
int n,c;
string str,tmp;

int main(){
  while(cin>>n&&n){
    t.clear();
    u.clear();
    cin.ignore();
    for(int i=0;i<n;i++){
      getline(cin,str);
      stringstream ss(str);
      while(ss>>tmp){
	t.push_back(tmp);
      }
    }
    sort(t.begin(),t.end());
    c=1;
    for(int i=1;i<(int)t.size();i++){
      if(t[i-1]==t[i])c++;
      else{
	c=1;
	u.push_back(P(-c,t[i-1]));
      }
    }
    u.push_back(P(-c,t[t.size()-1]));
    sort(u.begin(),u.end());
    char ch;
    cin>>ch;
    c=0;
    for(int i=0;i<(int)u.size();i++){
      if(ch==u[i].second[0]){
	if(c)cout<<' ';
	cout<<u[i].second;
	c++;
      }
      if(c==5)break;
    }
    if(c==0)cout<<"NA"<<endl;
    else cout<<endl;
  }
  return 0;
}
