N,P = map(int,input().split())

# nを因数分解する
# arr = [素因数,個数] のリスト
# n==1 のとき [1,1] となるので注意
def factorization(n):
    arr = []
    temp = n
    for i in range(2, int(-(-n**0.5//1))+1):
        if temp % i == 0:
            cnt = 0
            while temp % i ==0:
                cnt += 1
                temp //= i
            arr.append([i, cnt])

    if temp != 1:
        arr.append([temp, 1])

    if arr == []:
        arr.append([n, 1])

    return arr

l = factorization(P)
#print(l)
ans = []
for i in range(len(l)):
    if l[i][1] >= N:
        ans.append(l[i][0]*(l[i][1]//N))
if ans:
    k = 1
    for i in ans:
        k *= i
    print(k)
else:
    print(1)
