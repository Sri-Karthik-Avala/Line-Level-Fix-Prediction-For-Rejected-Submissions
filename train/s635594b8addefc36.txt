#include <iostream>
#include <cmath>
#include <complex>
#include <string>
#include <sstream>
#include <limits>
#include <algorithm>
#include <functional>

using namespace std;

int main()
{
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	cout.tie(nullptr);

	int K;
	cin >> K;

	int odd = 0;
	int even = 0;
	for (int i = 2; i <= K; i++)
	{
		if (i % 2 == 0) even++;
		else odd++;
	}

	cout << even * odd;

	return 0;
}
