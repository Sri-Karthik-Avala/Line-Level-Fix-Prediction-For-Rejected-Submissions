using namespace std;
using ll = long long;

/**
 * Fenwick Tree 
 *
 * 0-indexed
 */
template <int N>
struct FenwickTree {
    using D = ll; /// ????´???????
    D seg[N+1];
    
    /// ????´???????????????????
    void init() {
        for (int i = 0; i <= N; i++) {
            seg[i] = 0;
        }
    }

    /// i?????????????´????x???????????????
    void add(int i, D x) {
        i++;
        while (i <= N) {
            seg[i] += x;
            i += i & -i;
        }
    }

    /// [0, i)???sum
    D sum(int i) {
        D s = 0;
        while (i > 0) {
            s += seg[i];
            i -= i & -i;
        }
        return s;
    }

    /// [a, b)???sum
    D sum(int a, int b) {
        return sum(b) - sum(a);
    }
};

using P = pair<ll, ll>;
const int MN = 100100;
P p[MN];
FenwickTree<MN> fw;

int main() {
    fw.init();
    int n;
    cin >> n;
    ll sm = 0;
    p[0] = P(0, 0);
    for (int i = 1; i <= n; i++) {
        int d;
        cin >> d;
        sm += d;
        if (sm <= 0) {
            cout << -1 << endl;
            return 0;
        }
        p[i] = P(sm, i);
    }
    sort(p, p+n+1, [&](const P &l, const P &r){
        return l.first < r.first;
    });
    // for (int i = 0; i <= n; i++) {
    //     printf("%d %lld %lld\n", i, p[i].first, p[i].second);
    // }
    ll u = 0;
    for (int i = 1; i <= n; i++) {
        int l;
        cin >> l;
        ll re = p[i].first - p[i-1].first;
        if (re < l) {
            cout << -1 << endl;
            return 0;
        }
        u += fw.sum(p[i].second);
        fw.add(p[i].second, 1);
    }
    cout << u << endl;
    return 0;
}
