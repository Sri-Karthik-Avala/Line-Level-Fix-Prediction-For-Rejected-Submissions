#define FOR(i, a, b) for(int i = (a); i < (int)(b); ++i)
#define rep(i, n) FOR(i, 0, n)
#define ALL(v) v.begin(), v.end()
#define REV(v) v.rbegin(), v.rend()
#define MEMSET(v, s) memset(v, s, sizeof(v))
#define UNIQUE(v) (v).erase(unique(ALL(v)), (v).end())
#define MP make_pair
#define MT make_tuple

using namespace std;

typedef long long ll;
typedef pair<ll, ll> P;

int fr[110][110];
vector<P> G[110];
vector<int> cand[20];


int v[20];

int calc(int n){
	//rep(i, n) cout << v[i] << ' ';
	//cout << endl;
	int res = 0;
	rep(i, n){
		int tmp = 1e9;
		rep(j, n){
			if (i == j) continue;
			tmp = min(tmp, fr[v[i]][v[j]]);
		}
		res += tmp;
	}
	return res;
}

int dfs(int i, int k, int sz, int n){
	if (i == n) return calc(n);

	int res = 0;
	for (int j = k; j < sz; ++j){
		v[i] = j;
		res = max(res, dfs(i + 1, j + 1, sz, n));
		if (i + sz - k < n) break;
	}
	return res;
}

int main(){
	cin.tie(0);
	ios::sync_with_stdio(false);

	int n, m;
	cin >> n >> m;


	rep(i, m){
		int a, b, f;
		cin >> a >> b >> f;
		--a, --b;
		G[a].push_back(MP(b, f));
		G[b].push_back(MP(a, f));
		fr[a][b] = fr[b][a] = f;
	}

	rep(i, n){
		FOR(j, 1, 15){
			if (G[i].size() >= j) cand[j+1].push_back(i);
		}
	}

	int ans = 0;
	FOR(i, 2, 15){
		ans = max(ans, dfs(0, 0, cand[i].size(), i));
	}
	cout << ans << endl;

	return 0;
}
