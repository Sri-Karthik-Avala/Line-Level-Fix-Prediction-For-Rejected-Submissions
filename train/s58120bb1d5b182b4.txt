int mod=100000;
int dp[2][50][3001];
void add(int& x,int y){
	x+=y;
	if(x>=mod) x-=mod;
}
int main(){
	while(true){
		int n,m,s;
		cin>>n>>m>>s;
		if(n==0) break;
		n=n*n;
		rep(j,n+1) rep(k,s+1) dp[0][j][k]=0;
		dp[0][0][0]=1;
		rep(i,m){
			rep(j,n+1) rep(k,s+1) dp[1-i%2][j][k]=0;
			rep(j,n+1){
				rep(k,s+1){
					add(dp[1-i%2][j][k],dp[i%2][j][k]);
					if(j+1<=n&&k+i+1<=s) add(dp[1-i%2][j+1][k+i+1],dp[i%2][j][k]);
				}
			}
		}
		cout<<dp[m%2][n][s]<<endl;
	}
}
