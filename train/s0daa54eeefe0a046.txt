#include <iostream>
#include <string>
#include <algorithm>
#include <functional>
#include <vector>
#include <utility>
#include <cstring>
#include <iomanip>
#include <numeric>
#include <cmath>
#include <cassert>
using namespace std;
using ll = long long;
const int INF = 1<<30;
const int MOD = 1e9 + 7;
signed main()
{
    cin.tie(0);
    ios::sync_with_stdio(false);
    int h, w; cin >> h >> w;
    int table[h + 1][w + 1] = {};
    int dp[h + 1][w + 1] = {};
    for(int i = 1; i <= h; i++)
    {
        for(int j = 1; j <= w; j++)
        {
            cin >> table[i][j];
            dp[i][j] = (table[i][j] ? 0 : 1);
        }
    }
    int ans = 0;
    for(int i = 1; i <= h; i++)
    {
        for(int j = 1; j <= w; j++)
        {
            if(table[i][j]) continue;
            dp[i][j] = min({dp[i][j], dp[i - 1][j - 1], dp[i - 1][j], dp[i][j - 1]}) + 1;
            ans = max(ans, dp[i][j]);
        }
    }
    cout << ans * ans << endl;
    return 0;
}
