while 1:
    n = int(input())
    if n == 0:break
    c = {}
    for _ in range(n):
        s = input().split()
        for e in s:
            if e in c:c[e] += 1
            else:c[e] = 1
    c = [(e, c[e]) for e in c.keys()]
    c.sort(key = lambda x:x[1], reverse = True)
    c.sort(key = lambda x:x[0])
    s = input()
    a = []
    for i in range(len(c)):
        if c[i][0][0] == s:a.append(c[i][0])
    if a == []:print("NA")
    else:print(*a[:5])
