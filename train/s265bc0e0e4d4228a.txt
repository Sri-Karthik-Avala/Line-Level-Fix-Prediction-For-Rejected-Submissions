int main(void) {
    cin.tie(0); ios_base::sync_with_stdio(false);
    // solve(0);
    // cout << fixed << setprecision(15) << solve<ll>() << endl;
    int N;
    while(cin >> N,N){
        vector<int> a(N);
        vector<char> x(N);
        for(int i = 0; i < N;i++){
            string s; cin >> s;
            if(s == "x"){
                x[i] = true;
                continue;
            }
            a[i] = atoi(s.c_str());
        }
        bool ng = false;
        for(int i = 1; i < N;i++){ // x x
            if(x[i] && a[i-1] == a[i]) ng = true;
            if(!x[i] && !x[i-1]){
                if(i&1){
                    if(a[i] <= a[i-1]) ng = true;
                }else{
                    if(a[i] >= a[i-1]) ng = true;
                }
            }
        }
        if(ng){
            cout << "none" << endl;
            continue;
        }
        int l = -2*INF,r = 2*INF;
        for(int i = 0; i < N;i++){
            if(!x[i]) continue;
            if(i!=0){
                if(i&1){
                    l = max(l,a[i-1]+1);
                }else{
                    r = min(r,a[i-1]-1);
                }
            }
            if(i!=N-1){
                if(i&1){
                    l = max(l,a[i+1]+1);
                }else{
                    r = min(r,a[i+1]-1);
                }
            }
        }
        if(l==r){
            cout << l << endl;
        }else if(l < r){
            cout << "ambiguous" << endl;
        }else{
            cout << "none" << endl;
        }
    }
    return 0;
}
