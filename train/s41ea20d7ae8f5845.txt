#include<bits/stdc++.h>
using namespace std;
vector<int> G[100010];

#define MAX_N 1<<17
int n,dat[2*(MAX_N)-1];

void init(int n_){
  n=1;
  while(n<n_)n*=2;
  for(int i=0;i<2*n-1;i++)dat[i]=0;
}

int find(int k){
  k+=n-1;
  int ans=dat[k];
  while(k>0){
    k=(k-1)/2;
    ans+=dat[k];
  }
  return ans;
}


void update(int a,int b,int x,int k=0,int l=0,int r=n){
  if(r<=a||b<=l)return;
  if(a<=l&&r<=b){
    dat[k]+=x;
  }else{
    update(a,b,x,k*2+1,l,(l+r)/2);
    update(a,b,x,k*2+2,(l+r)/2,r);
  }
}

int num[2][200010];
int a;
void dfs(int k=0){
  num[0][k]=a;
  a++;
  for(int i=0;i<G[k].size();i++){
    dfs(G[k][i]);
  }
  num[1][k]=a;
}

int main(){
  int N;
  cin>>N;
  for(int i=0;i<N;i++){
    int k;
    cin>>k;
    for(int j=0;j<k;j++){
      int a;
      cin>>a;
      G[i].push_back(a);
    }
  }
  dfs();
  init(a);
  // cout<<num[1][0]<<" happy"<<endl;
  int q;
  cin>>q;
  for(int i=0;i<q;i++){
    int x;
    cin>>x;
    if(x==0){
      int w,v;
      cin>>v>>w;
      update(num[0][v],num[1][v]+1,w);
    }else{
      int u;
      cin>>u;
      cout<<find(num[0][u])<<endl;
    }
  }
  return 0;
}
