#include<iostream>
#include<bits/stdc++.h>

using namespace std;

int main()
{
    string a, b;
    cin>>a>>b;
    int m, n;
    m = a.length();
    n = b.length();
    int dp[m+1][n+1];
    for(int i = 0; i <= m; i++)
    {
        for(int j = 0; j <= n; j++)
        {
            if(i == 0 || j == 0)
                dp[i][j] = 0;
            else if(a[i] == b[j])
                dp[i][j] = 1 + dp[i-1][j-1];
            else
                dp[i][j] = max(dp[i-1][j], dp[i][j-1]); 
        }
    }
    string c;
    int i = m;
    int j = n;
    while(i > 0 && j > 0)
    {
        if(a[i-1] == b[j-1])
        {
            c.push_back(a[i-1]);
            i--;
            j--;
        }
        else
        {
            if(dp[i-1][j] > dp[i][j-1])
                i--;
            else
                j--;
            
        }
    }
    reverse(c.begin(),c.end());
    cout << c;
    return 0;
}
