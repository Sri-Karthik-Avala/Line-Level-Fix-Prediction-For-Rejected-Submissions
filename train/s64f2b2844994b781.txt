	vi x(m),y(m),z(m);
	rep(i,0,m){
		cin>>x[i]>>y[i]>>z[i];
	}
	cin>>k;
	int ind=k;
	for(int i=m-1;i>=0;i--){
		if(x[i]==0 && y[i]<=ind && ind<=z[i]){
			ind=z[i]+y[i]-ind;
		}
	}
	int now=0;
	now=a+d*(ind-1);
	rep(i,0,m){
		if(y[i]<=ind && ind<=z[i]){
			if(x[i]==0){
				ind=z[i]+y[i]-k;
			}else if(x[i]==1){
				now++;
			}else if(x[i]==2){
				now/=2;
			}
		}
	}
	o(now);
}
