vector<int> z_func(string &s) {
  vector<int> z(len(s), 0);
  int l = 0, r = 1;

  rep(i, 1, sz(z) - 1) {
    if (i >= r) {
      l = r = i;
      while (i + z[i] < len(s) && s[z[i]] == s[i + z[i]]) z[i]++, r++;
    } else if (z[i - l] < r - i) {
      z[i] = z[i - l];
    } else {
      z[i] = r - i;
      l = i;
      while (i + z[i] < len(s) && s[z[i]] == s[i + z[i]]) z[i]++, r++;
    }
  }

  return z;
}

bool is_good(string s) {
  auto z = z_func(s);
  rep(i, 1, len(s) / 2) {
    bool can_divide = len(s) % i == 0 && z[i] == i * (len(s) / i - 1);
    if (can_divide) return false;
  }
  return true;
}

void solve() {
  string s;
  cin >> s;

  auto z = z_func(s);

  rep(i, 1, len(s) / 2) {
    bool can_divide = len(s) % i == 0 && z[i] == i * (len(s) / i - 1);
    if (!can_divide) continue;

    if (i == 1) {
      cout << len(s) << nl << 1;
      return;
    } 

    int res = 0;

    rep(j, 1, len(s) - 1) {
      bool is_good_left = is_good(s.substr(0, j));
      bool is_good_right = is_good(s.substr(j));
      if (is_good_left && is_good_right) {
        res++;
      }
    }

    cout << 2 << nl << res;
    return;
  }

  cout << 1 << nl << 1;
}
