coins = (10,50,100,500)
isfirst = True
while 1:
    cost = int(raw_input())
    if cost == 0: break
    if not isfirst: print
    have = map(int,raw_input().split())

    budget = sum(map(lambda x,y:x*y, have,coins))
    change = budget-cost

    ans = [0]*4
    for i in reversed(range(4)):
        c = change/coins[i]
        ans[i] = c
        change -= c*ans[i]

    for i in range(4):
        if have[i]-ans[i] <= 0: continue
        print coins[i],have[i]-ans[i]

    isfirst = False
