template<class T>
ostream& operator<<(ostream& os, vector<T>& v)
{
	const string delimiter = "\n";
	REP(i, v.size())
	{
		os << v[i];
		if (i != v.size() - 1) os << delimiter;
	}
	return os;
}
/*--------------------template--------------------*/

int n;
vector<vi> tree;
ll hs[111111];
ll r = 456789065ll;

ll get_hash(int v)
{
	ll tmp = 0;
	for (auto u : tree[v])
	{
		tmp += get_hash(u) * r;
	}
	tmp += tree[v].size() * r;
	return hs[v] = tmp;
}

{
	cin.sync_with_stdio(false); cout << fixed << setprecision(10);

	tree.resize(n);
	REP(i, n - 1)
	{
		int a, b; cin >> a >> b;
		a--; b--;
		tree[a].push_back(b);
	}
	get_hash(0);
	map<ll, int> mp;
	REP(i, n)
	{
		mp[hs[i]]++;
	}
	ll ans = 0;
	for (auto i : mp)
	{
		ans += i.second * (i.second - 1) / 2;
	}
	cout << ans << endl;
	return 0;
}
