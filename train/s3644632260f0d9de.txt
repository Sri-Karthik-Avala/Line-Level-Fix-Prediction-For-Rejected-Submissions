#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;


int n,m;
vector <int> a,b,c,p,st;

int main() {


 cin >> n >> m;

 p.resize(m+1);
 for(int i=1;i<=m;i++) cin >> p[i];

 a.resize(n+1);
 b.resize(n+1);
 c.resize(n+1);

 for(int i=1;i<=n-1;i++) cin >> a[i] >> b[i] >> c[i];

 st.resize(n+1);
 for(int i=0;i<=n;i++) st[i] = 0;

 int mint,t;

 for(int i=1;i<m;i++) {
   mint = min(p[i],p[i+1]);
   t = max(p[i],p[i+1]); 
   st[mint]++;
   st[t]--;

 }

 for(int i=2;i<n;i++)st[i] += st[i-1];

 long long sum = 0;

 for(int i=1;i<n;i++){
   sum += min(a[i]*st[i],c[i]+b[i]*st[i]);
 }

 cout << sum << endl;

 return 0;

}
