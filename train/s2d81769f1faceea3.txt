#include <cstdio>
#include <cstring>
#include <iostream>
#include <vector>
#define debug(...) fprintf(stderr,__VA_ARGS__)
using namespace std;
template<class T> void read(T &x) {
	x=0; int f=1,ch=getchar();
	while(ch<'0'||ch>'9'){if(ch=='-')f=-1;ch=getchar();}
	while(ch>='0'&&ch<='9'){x=x*10-'0'+ch;ch=getchar();}
	x*=f;
}
template<class T> inline bool Cmin(T &x,T y) {return x>y?x=y,1:0;}
const int inf=1e9;
const int maxn=1e5+50;
int N;
int an,mn;
int head[maxn],deg[maxn];
int siz[maxn];
int dp[maxn][2];
char c[maxn];
struct edge {
	int to,nex;
	edge(int to=0,int nex=0):to(to),nex(nex){}
};
vector<edge> G;
inline void addedge(int u,int v) {
	G.push_back(edge(v,head[u])),head[u]=G.size()-1;
	G.push_back(edge(u,head[v])),head[v]=G.size()-1;
	++deg[u],++deg[v];
}
void dfs0(int u,int fa) {
//	debug("%d %d\n",u,fa);
	siz[u]=c[u]=='W';
	for(int i=head[u];~i;i=G[i].nex) {
		int v=G[i].to; if(v==fa) continue;
		dfs0(v,u);
		siz[u]+=siz[v];
	}
	if(siz[u]==0) --deg[fa];
}
void dfs1(int u,int fa) {
//	debug("%d %d\n",u,fa);
	int w=deg[u]==1?0:((c[u]=='B')^deg[u])&1;
	if(deg[u]>1&&c[u]=='W') ++an,--w;
	for(int i=head[u];~i;i=G[i].nex) {
		int v=G[i].to; if(v==fa||!siz[v]) continue;
		an+=2;
		dfs1(v,u);
		int t=dp[v][0]-1;
		if(t<dp[u][0]) dp[u][1]=dp[u][0],dp[u][0]=t;
		else if(t<dp[u][1]) dp[u][1]=t;
	}
	dp[u][0]+=w;
	Cmin(mn,dp[u][0]+dp[u][1]);
}
int main() {
	read(N);
	memset(head,-1,sizeof(head));
	for(int i=1;i<N;++i) {
		int u,v; read(u),read(v);
		addedge(u,v);
	}
	scanf("%s",c+1);
	int rt=-1;
	for(int i=1;i<=N;++i) if(c[i]=='W') {rt=i;break;}
	if(rt==-1) {puts("0"); return 0;}
	dfs0(rt,0);
	if(deg[rt]==0) {puts("1"); return 0;}
	an=1,mn=inf,dfs1(rt,0);
//	debug("%d %d\n",an,mn);
	an+=mn;
	printf("%d\n",an);
	return 0;
}
