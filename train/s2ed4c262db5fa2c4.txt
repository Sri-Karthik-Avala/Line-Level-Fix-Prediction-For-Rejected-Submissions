#include<iostream>
#include<cstdio>
using namespace std;
int main(void)
{
	double N=0,x=0,y=0,b=0,p=0,sum1=0;
	scanf("%lf",&N);
	for(int i=0;i<N;i++)
	{
		scanf("%lf %lf %lf %lf",&x,&y,&b,&p);
		sum1=x*b+y*p;
		if(b>=5&&p>=2){printf("%.0lf\n",(x*b+y*p)*0.8);}
		else
		{
			if(b<5&&p<2){if(sum1>(x*5+y*2)*0.8){sum1=(x*5+y*2)*0.8;}}
			else if(b<5){if(sum1>(x*5+y*p)*0.8){sum1=(x*5+y*p)*0.8;}}
			if(p<2){if(sum1>(x*b+y*2)*0.8){sum1=(x*b+y*2)*0.8;}}
			printf("%.0lf\n",sum1);
		}
	}
	return 0;
}
