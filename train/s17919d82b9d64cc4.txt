#include<iostream>
using namespace std;
struct EDGE{
	int u,v,w;
}e[65];
int cnt=0;
void add(int u,int v,int w){
	e[++cnt]=(EDGE){u,v,w};
}
int l;
int main(){
	cin>>l;
	int tmp=l,v=0;
	while(tmp){
		v++;
		tmp/=2;
	}
	for(int i=1;i<v;i++){
		add(i,i+1,0);
		add(i,i+1,1<<(v-i-1));
	}
	tmp=1<<(v-1);
	for(int i=v-1;i>=1;i--){
		if(l&(1<<(i-1))){
			add(i,v-i+1,tmp);
			tmp+=1<<(i-1);
		}
	}
	cout<<v<<" "<<cnt<<endl;
	for(int i=1;i<=cnt;i++){
		cout<<e[i].u<<" "<<e[i].v<<" "<<e[i].w<<endl;
	}
}
