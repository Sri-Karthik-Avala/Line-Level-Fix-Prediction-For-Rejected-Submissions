#include<iostream>
#include<cmath>
#include<iomanip>
using namespace std;
int main(){
int n,sum,i,cou;
sum=0;
cou=0;
int s[1000];
while(cin>>n,n){
sum=0;
cou=0;
	for(i=0;i<n;i++){
		cin>>s[i];
		sum+=s[i];
	}
	for(i=0;i<n;i++){
		cou+=(s[i]-sum/n)*(s[i]-sum/n);
	}
	cout<<fixed<<setprecision(8)<<sqrt(cou/n)<<endl;
}
return 0;
}
