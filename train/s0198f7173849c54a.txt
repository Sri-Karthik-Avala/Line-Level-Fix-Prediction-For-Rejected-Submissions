{

	burning_T = n;

	for (i = 0; i < n; i++)
	{
		scanf("%lld", &come_T[i]);
	}

	LL use_k = n;
	LL poket_T;
	LL T[100000];
	LL cnt = 0;

	for (i = 0; i < n - 1; i++)
	{
		T[i] = come_T[i + 1] - come_T[i] - 1;
	}
