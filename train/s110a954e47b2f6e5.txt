#include <iostream>
#include <algorithm>
using namespace std;

int by[300][3];
int cost[300];
int N, M;

int dp(int k) {
  int maxi = -1;
  int value[M+1];
  fill(value, value+M+1, 0);

  for(int i = 0; i < N; i++) {
    for(int j = 0; j <= M; j++) {
      value[j] = max(value[j], by[i][k] + value[j-cost[i]]);
      maxi = max(value[j], maxi);
    }
  }
  return maxi;
}

main() {
  while(1) {
    cin >> N >> M;
    if(cin.eof()) break;
    for(int i = 0; i < N; i++) {
      string dumy;
      getchar();
      getline(cin, dumy);
      cin >> cost[i] >> by[i][0] >> by[i][1] >> by[i][2];
    }

    int maxi = 0;
    for(int k = 0; k < 3; k++) {
      maxi = max(maxi, dp(k));
    }
    cout << maxi << endl;
  }
}
