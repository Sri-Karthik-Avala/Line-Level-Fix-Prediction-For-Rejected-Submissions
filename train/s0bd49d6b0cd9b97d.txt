int bubuwa(int t) {

	int i, j;
	memset(dp, 0, sizeof(dp));
	dp[0][0] = true;
	for (i = 0; i < n; i++) {
		//if (i != t) {
			for (j = 0; j <= k; j++) {

				dp[i + 1][j] |= dp[i][j];
				if (j >= a[i] && i != t)dp[i + 1][j] |= dp[i][j - a[i]];

			}
		//}
	}

	for (i = k - a[t]; i < k; i++) {
		if (dp[n][i] == true) {
			return 1;
		}
	}
	return 0;

}
