#include<stdio.h>
#include<queue>
#include<algorithm>
using namespace std;
int ijk[1010000];
int v[1010000];
int main(){
	int a,b,c;
	while(scanf("%d%d%d",&a,&b,&c),a){
		if(b<c)swap(c,b);
		for(int i=0;i<b;i++)ijk[i]=999999999;
		for(int i=0;i<b;i++)v[i]=0;
		
		priority_queue<pair<int,int> >Q;
		ijk[0]=0;
		Q.push(make_pair(0,0));
		while(Q.size()){
			int cost=-Q.top().first;
			int at=Q.top().second;
			Q.pop();
			if(v[at])continue;
			v[at]=1;
			if(!v[(at+c)%b]&&ijk[(at+c)%b]>cost+c){
				ijk[(at+c)%b]=cost+c;
				Q.push(make_pair(-cost-c,(at+c)%b));
			}
		}
		int ret=0;
		for(int i=0;i<b;i++){
		//	printf("%d ",ijk[i]);
			ret+=min((a+b-1-i)/b,(ijk[i])/b);
		}
		printf("%d\n",ret);
	}
}
