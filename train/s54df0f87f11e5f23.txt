int main(){
	ll n,m;
	while(cin>>n>>m,n+m){
		vi in(m);
		rep(i,m)cin>>in[i];
		vi ha(m);
		rep(i,m)rep(j,m)if(i!=j&&in[i]%in[j]==0)ha[i]=1;
		vi tmp;
		rep(i,m)if(ha[i]==0)tmp.pb(in[i]);
		in=tmp;
		ll out=n*(n+1)/2,cnt=n;
		ll m=in.size();
		ll N=1<<m;
		loop(i,1,N){
			ll t=1;
			ll co=0;
			rep(j,m)if(i&1<<j){
				t*=in[j],co++;
				if(t>n)break;
			}
			if(t>n)continue;
			ll q=n/t;
			if(co%2)out-=t*q*(q+1)/2,cnt-=q;
			else out+=t*q*(q+1)/2,cnt+=q;
		}
		if(cnt==0)cnt=1;
		cout<<shosu(9)<<(double)out/cnt<<endl;
