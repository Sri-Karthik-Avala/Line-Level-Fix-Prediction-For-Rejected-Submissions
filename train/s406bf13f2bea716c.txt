#include<bits/stdc++.h>
using namespace std;
int main(){
  int n;
  cin>>n;
  vector<int> d(n);
  for(int i=0;i<n;i++)
    cin>>d.at(i);
  if(n>12){
    vector<int> b(13,0);
    for(int i=0;i<n;i++)
      b.at(d.at(i))++;
    bool f=(b.at(0)>0&&b.at(12)>1?0:1);
    for(int i=1;i<12;i++)
      f=(b.at(i)>2?0:f);
    cout<<(f?1:0)<<endl;
    return 0;
  }
  int ma=0;
  for(int x=0;x<(1<<n);x++){
    int b=x;
    vector<int> t(n+1,0);
    for(int i=0;i<n;i++){
      t.at(i)=(b%2==1?d.at(i):24-d.at(i));
      b/=2;
    }
    int mi=12;
    for(int i=0;i<=n;i++)
      for(int j=i+1;j<=n;j++)
        mi=min(mi,min(abs(t.at(i)-t.at(j)),24-abs(t.at(i)-t.at(j))));
    ma=max(ma,mi);
  }
  cout<<ma<<endl;
}
