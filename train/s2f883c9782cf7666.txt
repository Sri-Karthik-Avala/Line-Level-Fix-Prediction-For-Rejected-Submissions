const int N = 300000;
const int K = 100000;
const int mod = 1e9 + 7;
int dp[N], f[N], invf[N], n, a[N];
int binpow(int x, int y){
    int res = 1;
    while(y){
        if (y % 2) res = res * x % mod;
        y /= 2;
        x = x * x % mod;
    }
    return res;
}
int c(int n, int k){
    if (k > n || n < 0) return 0;
    return f[n] * invf[k] % mod * invf[n - k] % mod;
}
main(){
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    //freopen("input.txt", "r", stdin);
    //freopen("output.txt", "w", stdout);
    int n;
    cin >> n;
    f[0] = 1;
    for (int i = 1; i <= n; i++) f[i] = f[i - 1] * i % mod;
    invf[n] = binpow(f[n], mod - 2);
    for (int i = n - 1; i >= 0; i--) invf[i] = invf[i + 1] * (i + 1) % mod;
    for (int i = 0; i < n; i++){
        a[i] = c(n - 1 - (i + 1), n - 1 - i - (i + 1)) * f[i] % mod * f[n - 1 - i] % mod;
    }
    int ans = 0;
    for (int i = 0; i < n; i++){
        ans = (ans + (a[i] - a[i + 1]) * (n - i - 1)) % mod;
    }
    cout << ans;
}
