def main(kwargs):
    n = kwargs["n"]
    m = kwargs["m"]
    H = kwargs["H"]
    A = kwargs["A"]
    B = kwargs["B"]
    tmp = [1 for i in range(n)]

    for i in range(m):
        if H[A[i] - 1] >= H[B[i] - 1]:
            tmp[B[i] - 1] = 0

        elif H[A[i] - 1] <= H[B[i] - 1]:
            tmp[A[i] - 1] = 0

    res = sum(tmp)
    return res
