#include <iostream>
#include <cstring>
using namespace std;

const int MOD = 100000;
int W, H;

enum DIR {
	EAST,
	NORTH,
	NONE
};

int solve(int x, int y, DIR d, bool turn)
{
	//cerr << x << " " << y << " ";
	
	if (x==W-1 && y==H-1) {
	//	cerr << "OK" << endl;
		return 1;
	} else if (x>=W || y>=H) {
	//	cerr << endl;
		return 0;
	}
	//cerr << endl;
	
	int res=0;
	
	if (turn) {
		if (d!=NORTH)
			res += solve(x+1, y, EAST, false);
		if (d!=EAST)
			res += solve(x, y+1, NORTH, false);
	} else {
		res += solve(x+1, y, EAST, NORTH==d);
		res += solve(x, y+1, NORTH, EAST==d);
	}
	return res%MOD;
}


int main()
{
	while (cin>>W>>H, W) {
		cout << solve(0, 0, NONE, false) << endl;
	}
}
