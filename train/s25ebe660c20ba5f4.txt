h, w = map(int, raw_input().split())
c = [map(int, raw_input().split()) for i in xrange(h)]
cnt = [[0]*w for i in xrange(h)]

for j in xrange(w):
    seq = 0
    for i in xrange(h):
        if c[i][j] == 0:
            seq += 1
        else:
            seq = 0
        cnt[i][j] = seq
ans = 0
st = [(0, -1)]
for i in xrange(h):
    for j in xrange(w):
        v = cnt[i][j]
        while v < st[-1][0]:
            l, k = st.pop()
            ans = max(ans, min(l, j-k))
        if st[-1][0] < v:
            st.append((v, j))
    while 0 < st[-1][0]:
        l, k = st.pop()
        ans = max(ans, min(l, w-k))
print ans**2
