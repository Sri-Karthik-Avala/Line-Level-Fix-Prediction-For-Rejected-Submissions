#include<bits/stdc++.h>
#define ll long long
using namespace std;
ll A[1010101],B[1010101],cac[1010101],n;
ll caca[1010101];
map<ll,ll> xx;
inline bool check(){
	xx.clear();
	for (int i=1;i<=n;i++) xx[caca[i]]++;
	for (int i=1;i<=n;i++){
		if (A[i]==B[i]){
			if (xx[A[i]]>=2) return 1;
		}else{
			if (xx[A[i]]&&xx[B[i]]) return 1;
		}
	}
	return 0;
}
inline ll get(){
	ll ans=0;
	for (int i=1;i<=n;i++) ans+=caca[i];
	return ans;
}
int main(){
	scanf("%lld",&n);
	for (int i=1;i<=n;i++) scanf("%lld%lld",&A[i],&B[i]);
	ll ans=10101010101010101;
	for (int i=1;i<=n;i++) caca[i]=A[i];
	ans=min(ans,get());
	for (int i=1;i<=n;i++) caca[i]=B[i];
	ans=min(ans,get());
	for (int i=1;i<=n;i++){
		cac[2*i-1]=A[i],cac[2*i]=B[i];
	}
	sort(cac+1,cac+1+2*n);
	for (int i=1;i<=n;i++) caca[i]=cac[i];
	if (check()){
		printf("%lld\n",min(ans,get()));
		return 0;
	}else{
		caca[n]=cac[n+1];
		if (check()) printf("%lld\n",min(ans,get()));
		else{
			caca[n-1]=cac[n];
			if (check()) ans=min(ans,get());
			caca[n]=cac[n+1],caca[1]=cac[n];
			if (check()) ans=min(ans,get());
			cout<<ans;
		}
	}
	return 0;
}
