#include "bits/stdc++.h"
using namespace std;
int dp[300][300];//dp[l][r] [l,r]でだるまを落としたときの落とした個数

int main()
{
	while(1)
	{
		int n;
		cin>>n;
		if(n==0)
			break;
		
		int da[300];
		for(int i=0;i<n;i++)
			cin>>da[i];
		fill(dp[0],dp[300],0);
		for(int i=1;i<n;i++)
		{
			int dif=da[i]-da[i-1];
			if(dif<2 && dif>-2)
				dp[i-1][i]=2;
			else
				dp[i-1][i]=0;
		}
		//ここからdp
		for(int k=2;k<n;k++)
		{
			for(int i=0;i<n-k;i++)
			{
				//分割して調べる
				for(int j=i;j<i+k-1;j++)
					dp[i][i+k]=max(dp[i][i+k],dp[i][j]+dp[j+1][i+k]);
				//真ん中が全部消える場合も調べる
				int dif=da[i]-da[i+k];
				if(dp[i+1][i+k-1]==k-1 && dif<2 && dif>-2)
					dp[i][i+k]=k+1;
			}
		}
		cout<<dp[0][n-1]<<endl;
	}
	return 0;
}
