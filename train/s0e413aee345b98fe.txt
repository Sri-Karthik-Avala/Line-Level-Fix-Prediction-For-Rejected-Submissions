#include <bits/stdc++.h>
#include <assert.h>

using namespace std;

int main()
{
	int n, p[100];
	cin >> n;
	for (int i = 0; i < n; i++)
		cin >> p[i];
	int ans = 0;
	for (int i = 1; i < 100; i++)
	{
		int cnt = 0;
		for (int j = 0; j < n; j++)
			if (p[j] >= i)
				cnt++;
		if (i <= cnt)
			ans = i;
	}
	cout << ans << endl;

	return 0;
}
