using namespace std;
typedef long long ll;
typedef pair<int, int>P;

int num[10], used[10], ans;

void check(int a) {
	if (a == 9) {
		int res1 = 100 * num[3] + 10 * (num[1] + num[4]) + (num[0] + num[2] + num[5]);
		int res2 = 100 * num[6] + 10 * num[7] + num[8];
		if (res1 == res2)ans++;
	}
	else if (num[a] != -1)check(a + 1);
	else {
		for (int i = 1; i <= 9; i++) {
			if (used[i] == 1)continue;
			used[i] = 1;
			num[a] = i;
			check(a + 1);
			used[i] = 0;
			num[a] = -1;
		}
	}
}

int main() {
	for (int i = 0; i < 9; i++)
		cin >> num[i];

	for (int i = 0; i < 9; i++) {
		if (num[i] != -1)
			used[i] = 1;
	}
	cout << ans << endl;
	return 0;
}
