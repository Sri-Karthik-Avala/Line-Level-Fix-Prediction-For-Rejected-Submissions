bool solve(string st) {
	int sum = 0;
	for (int i = 0; i < 11; ++i) {
		int n = 11 - i;
		if (n >= 7) {
			sum += (st[i] - '0')*(n - 5);
		}
		else {
			sum += (st[i] - '0')*(n + 1);
		}
	}
	sum %= 11;
	sum = 11 - sum;
	if (sum ==10)sum = 0;
	return sum==st[11]-'0';
}

int main() {
	string st; cin >> st;
	int a = st.find('?');
	vector<int>anss;
	for (int i = 0; i < 10; ++i) {
		st[a] = '0' + i;
		if (solve(st))anss.push_back(i);
		st[a] = '?';
	}
	if (anss.size() >= 2)cout << "MULTIPLE" << endl;
	else cout << anss[0] << endl;
	return 0;
}
