#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ld = long double;
#define rep(i, n) for (int i = 0; i < (int)(n); i++)
int main() {
    ll N;
    cin>>N;
    vector<ll>A(N);
    ll ans=0;
    rep(i,N){
        cin>>A[i];
    }
    sort(A.begin(),A.end());
    ll cnt=0;
    rep(i,N-1){
        cnt++;
        if(A[i]!=A[i+1]){
            if(cnt%2==1){
                ans++;
            }
            cnt=0;
        }
    }
    if(cnt%2==1){
        ans++;
    }
    cout<<ans<<endl;
}
