bool in(int64 f, int64 t, int64 x) {
	t = t - 60*(f/60); f = f%60; x = x%60;
	if (f > t) f -= 60;
	return f <= x && x <= t;
}

int main(void){
	cin.tie(0);
	ios::sync_with_stdio(false);
	string s;
	while (cin >> s && s != "#") {
		int64 n = 0, two = 1;
		int64 dc = 0, dm = 1;
		for (int i = 0; i < s.size();) {
			n++;
			if (s.substr(i, 4) == "west") {
				if (90 * dm != dc) {
					dc = dc * (two/dm) + 90;
					dm = two;
				}
				i += 4;
			} else {
				if (0 != dc) {
					dc = dc * (two/dm) - 90;
					dm = two;
				}
				i += 5;
			}
			two *= 2;
		}
		int64 d = __gcd(dc, dm);
		dc /= d; dm /= d;
		if (dm == 1) cout << dc << endl;
		else cout << dc << "/" << dm << endl;
	}
}
