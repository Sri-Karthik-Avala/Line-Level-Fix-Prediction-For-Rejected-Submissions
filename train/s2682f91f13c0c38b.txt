signed main() {
	int N;
    cin>>N;
    int ans=0;

    vector<int>p(N);
    for(int i=0;i<N;i++){
        cin>>p[i];
    }
    for(int i=1;i<N;i++){
        int A=max(p[i-1],max(p[i],p[i+1]));
        int B=min(p[i-1],min(p[i],p[i+1]));
        int C=p[i-1]+p[i]+p[i+1]-A-B;
        if(C==p[i]){
            ans++;
        }
    }
    cout<<ans<<endl;
}
