#include<bits/stdc++.h>
using namespace std;
int rd(){
	int x=0;char ch=getchar();
	while(!isdigit(ch))ch=getchar();
	while(isdigit(ch))x=x*10+ch-'0',ch=getchar();
	return x;
}
int len,cnt[65];
char ch[]={"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklnmopqrstuvwxyz0123456789"};
bool check(string S){
	cout<<"? "<<S<<endl;
	return len-S.size()==rd();
}
string merge(int l,int r){
	if(l==r) return string(cnt[l],ch[l]);
	int mid=(l+r)>>1;
	string A=merge(l,mid),B=merge(mid+1,r),C="";
	int i=0,j=0;
	while(i<A.size() && j<B.size()){
		if(check(C+A[i]+B[j])) C+=A[i],++i;
		else C+=B[j],++j;
	}
	while(i<A.size()) C+=A[i],++i;
	while(j<B.size()) C+=B[j],++j;
	return C;
}
