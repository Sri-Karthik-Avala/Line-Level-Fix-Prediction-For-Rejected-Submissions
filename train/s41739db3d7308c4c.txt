#include <cstdio>
#define mod 100000

int n, m, ans, p, t;
int s[100000];

int main(){
	scanf("%d%d", &n, &m);
	for(int i = 1; i < n; i++) scanf("%d", &s[i]);
	for(int i = 1; i < n; i++) s[i] += s[i-1];
	for(int i = 0; i < m; i++){
		scanf("%d", &t);
		if(t > 0){
			ans += s[p+t]-s[p];
		} else{
			ans += s[p]-s[p+t];
		}
		p += t;
		if(ans > mod) ans -= mod;
	}
	printf("%d\n", ans);
}
