using namespace std;
typedef long long ll;
typedef pair<int, int> P;

const ll MD = 1e9+7;

int n;
string st;


P expr(int &s);

P numb(int &s) {
//    printf("numb %d\n", s);
    while (isdigit(st[s])) s++;
    return P(1, 1);
}

P fact(int &s) {
//    printf("fact %d\n", s);
    P a;
    switch (st[s]) {
    case 'x':
        s++;
        a = numb(s);
        break;
    case '~':
        s++;
        a = fact(s);
        swap(a.first, a.second);
        break;
    case '(':
        s++;
        a = expr(s);
        s++;
        break;
    }
    return a;
}

P term(int &s) {
//    printf("term %d\n", s);
    P p = P(0, 1);
    while (true) {
        P d = fact(s);
//        printf("term_res %lld %lld\n", d.first, d.second);
        p.first = (ll)(p.first+p.second)*(d.first+d.second) % MD;
        p.first -= (ll)p.second*d.second % MD;
        p.first = (p.first % MD + MD) % MD;
        p.second = (ll)p.second*d.second % MD;
        if (s >= n) break;
        if (st[s] != '&') break;
        s++;
    }
    return p;
}

P expr(int &s) {
//    printf("expr %d\n", s);
    P p = P(1, 0);
    while (true) {
        P d = term(s);
        p.second = (ll)(p.first+p.second) * (d.first+d.second) % MD;
        p.second -= (ll)p.first * d.first % MD;
        p.second = (p.second % MD + MD) % MD;
        p.first = (ll)p.first*d.first % MD;
        if (s >= n) break;
        if (st[s] != '|') break;
        s++;
    }
    return p;
}
