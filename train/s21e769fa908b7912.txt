##  input functions for me
def rsa(sep = ''):
    if sep == '' :
        return input().split() 
    else: return input().split(sep)
def rip(sep = ''):
    if sep == '' :
        return map(int, input().split()) 
    else: return map(int, input().split(sep))
def ria(sep = ''): 
    return list(rip(sep))
def ri(): return int(input())
def rd(): return float(input())
def rs(): return input()
##
def main():
    inf = int(1e9)
    N = ri()
    S = rs()
    dp = [[inf] * N for i in range(2)]
    # forbidden : '#.'
    if S[0] == '.':
        dp[0][0] = 0
        dp[1][0] = 1
    else:
        dp[0][0] = 1
        dp[1][0] = 0
    for i in range(1, N):
        if S[i] == '.':
            dp[0][i] = min(dp[0][i], dp[0][i-1])
            dp[1][i] = min(dp[1][i], dp[0][i-1] + 1)
            dp[1][i] = min(dp[0][i], dp[1][i-1] + 1)
        else :
            dp[0][i] = min(dp[0][i], dp[0][i-1] + 1)
            dp[1][i] = min(dp[0][i], dp[0][i-1])
            dp[1][i] = min(dp[0][i], dp[1][i-1])
    print(min(dp[0][N-1], dp[1][N-1]))



if __name__ == "__main__":
    main()
