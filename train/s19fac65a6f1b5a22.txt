#include <bits/stdc++.h>
using namespace std;
using ll=long long;
int main(){
  ll N,ans=0; cin>>N;
  vector<vector<ll>> A(310,vector<ll>(310,0)),B(310,vector<ll>(310,1));
  for(ll i=0; i<N; i++) for(ll j=0; j<N; j++) cin>>A[i][j];
  for(ll k=0; k<N; k++) for(ll i=0; i<N; i++) for(ll j=0; j<N; j++){
    ll z=A[i][k]+A[k][j];
    if(A[i][j]>z){
      cout<<-1<<endl;
      return 0;
    }
    if(A[i][j]==z) B[i][j]=0;
  }
  for(ll i=0; i<N; i++) for(ll j=0; j<N; j++)ans+=A[i][j]*B[i][j];
  cout<<ans<<endl;
}
