const ll INF = 100100100;
typedef pair<ll, ll> P;
const ll mod = 1000000007;
ll gcd(ll a, ll b) {//a>b
	if (a < b)swap(a, b);
	if (a % b == 0) return b;
	else return gcd(b, a % b);
}
int main() {
	ll N, M;
	cin >> N >> M;//N > M
	if (N < M)swap(N, M);
	ll ans = 1;
	if (N - M > 2)cout << "0" << endl;
	else if (N - M == 1) {
		for (int i = 1; i <= N; i++) {
			ans = ans % mod * i;
		}
		for (int i = 1; i <= M; i++) ans = ans % mod * i;
		cout << ans % mod << endl;
	}
	else {
		for (int i = 1; i <= N; i++) {
			ans = ans % mod * i;
		}
		for (int i = 1; i <= M; i++) ans = ans % mod * i;
		cout << ans * 2 % mod << endl;
	}
}
