ll GCD (ll X, ll Y) {
    return Y ? GCD(Y, X % Y) : X;
}

ll num(ll N, ll C, ll D) {
    ll G = GCD(C, D);
    ll L = D / G * D;
    return N - N / C - N / D + N / L;
}
