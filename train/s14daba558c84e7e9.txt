#include<bits/stdc++.h>
using namespace std;

int main(){

    int c, n, m; cin >> c >> n >> m;
    vector<int> s(n);
    vector<int> p(n);
    for(int i = 0; i < n; i++) cin >> s[i] >> p[i];
    
    vector<vector<int> > dp(n + 1, vector<int> (c + 1, 0));
    for(int i = 0; i < n; i++){
        for(int j = 0; j < c; j++){
            //take  
            if(j + s[i] <= c) dp[i + 1][j + s[i]] = max(dp[i + 1][j + s[i]], dp[i][j] + p[i]);
            //no
            dp[i + 1][j] = max(dp[i + 1][j], dp[i][j]);
        }
    }

    for(int i = 1; i <= c; i++){
        dp[n][i] = max(dp[n][i], dp[n][i - 1]);
    }

    for(int i = 1; i <= m; i++){
        cout << dp[n][c / i] * i << endl;
    }

    return 0;
}
