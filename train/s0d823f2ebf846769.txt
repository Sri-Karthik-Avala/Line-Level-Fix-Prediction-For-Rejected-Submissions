int main(int argc, char const *argv[])
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, M;
    cin >> N >> M;

    vector<ull> H(N, 0);
    REPinit(i, 0, N) cin >> H[i];

    vector<int> hcomp(N, 0);
    int A,B;
    REPinit(i,0,M) {
        cin >> A >> B;
        if (H[A - 1] > H[B - 1]) {
            hcomp[B - 1] += 1;
        }
        if (H[A - 1] < H[B - 1]) {
            hcomp[A - 1] += 1;
        }
    }

    int ans = 0;
    REPinit(i, 0, N) {
        if (hcomp[i] != 0) ans++;
    }

    cout << ans << endl;
    return 0;
}
