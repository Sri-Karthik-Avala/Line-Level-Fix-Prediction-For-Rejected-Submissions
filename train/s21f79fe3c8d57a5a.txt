using namespace std;
typedef long long LL;
const int MR=1e6+10;
int N;
LL M;
int a[MR];
int s[MR],temp[MR];

LL mergeSort(int l, int r){
    LL t = 0;
    if(r - l > 1){
        int m = (l + r) / 2;
        t = mergeSort(l, m) + mergeSort(m, r);
        for(int i = l; i < r; i++) temp[i] = s[i];
        int in1 = l, in2 = m, out = l;
        while(in1 < m && in2 < r){
            if(temp[in1] <= temp[in2]){
                s[out++] = temp[in1++];
            }
            else{
                s[out++] = temp[in2++];
                t += m - in1;
            }
        }
        while(in1 < m) s[out++] = temp[in1++];
        while(in2 < r) s[out++] = temp[in2++];
    }
    return t;
}


bool check(int x){
	s[0]=0;
	for(int i=1;i<=N;i++){
		if(a[i]>=x) s[i]=s[i-1]+1;
		else s[i]=s[i-1]-1;
	}
	//printf("x=%d\n",x);
	//for(int i=0;i<=N;i++)printf("%d ",s[i]);printf("\n");
	LL t=mergeSort(0,N+1);
    //for(int i=0;i<=N;i++)printf("%d ",s[i]);printf("\n");
	return M-t >= M/2;
}
int main(){
    scanf("%d",&N);
    for(int i=1;i<=N;i++){
        scanf("%d",a+i);
    }
    int l=1,r=1e9+10,m;
    M=1ll*(N+1)*N/2;
    while(l<r-1){
    	m=(l+r)/2;
    	if(check(m)){
    		l=m;
    	}
    	else{
    		r=m;
    	}
    }
    printf("%d\n",l);
    return 0;
}
