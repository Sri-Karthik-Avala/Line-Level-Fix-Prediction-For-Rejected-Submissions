#include <bits/stdc++.h>
using namespace std;

int n,t,ans;
int a[100005];

int main() {
    ios_base::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    cin>>n>>t;
    while (n--) {
        int l,r; cin>>l>>r;
        ++a[l], --a[r+1];
    }
    for (int i=1;i<=t;i++) a[i]+=a[i-1];
    for (int i=0;i<=t;i++) ans=max(ans,a[i]);
    cout<<ans<<'\n';
}
