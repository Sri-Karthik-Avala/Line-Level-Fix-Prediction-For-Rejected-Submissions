
n = int(input())
dp = [[0 for _ in range(3)] for _ in range(n+1)]
ls  = [[1,2],[0,2],[0,1]]
for i in range(1,n+1):
    day = list(map(int,input().split()))
    dp[i][0] = day[0] + max(dp[i-1][1],dp[i][2])
    dp[i][1] = day[1] + max(dp[i-1][0],dp[i][2])
    dp[i][2] = day[2] + max(dp[i-1][0],dp[i][1])
print(max(dp[n]))
