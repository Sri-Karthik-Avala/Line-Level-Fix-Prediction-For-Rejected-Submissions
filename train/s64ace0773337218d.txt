#include <iostream>

using namespace std;

int main() {
  int i,j,n,m,s,p,d[100];
  while(cin >> n && n>0) {
	  s=p=0;
	  for (i=0;i<n;i++) { cin >> d[i]; s+=d[i];}
	  for (i=1,m=0;i<=100;i++) { m+=i; if (s==m) break;}
	  if (i<101) m=i; else { cout << -1 << endl; continue;}
	  for (i=0;i<=10000;i++) {
		  if (m==n && d[0]==1 && d[m-1]==m) { for (j=0;j<m;j++) if (d[j]!=j+1) j=9999;  if (j==m) break; }
		  for (j=0;j<n;j++) d[j]--;
		  for (j=0,p=0;j<n;j++) if (d[j]>0) d[p++]=d[j];
		  d[p]=n; n=p+1; 
	  } 	    
	  if (i>10000) cout << -1 << endl; else cout << i << endl;
  }
  return 0;
 }
