#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
int main(){
int N[1000],M[1000],hyo[1000];
int n,m;
cin>>n>>m;
int tmp;
for(int i=0;i<n;i++)cin>>N[i];
for(int i=0;i<m;i++)cin>>M[i];

	for(int i=0;i<m;i++){
		for(int j=0;j<n;j++){
			if(M[i]>=N[j]){
			hyo[j]++;
			break;
			}
		}
	}
	int a=0;int ma=0;
	for(int i=0;i<n;i++){
		if(hyo[i]>ma)ma=hyo[i],a=i+1;
	}
	//if(a==0)cout<<1<<endl;
	cout<<a<<endl;


return 0;
}
