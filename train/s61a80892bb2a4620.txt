s=input()
N=len(s)
rl=[]
tmp=""
t=0
ans=[-1,-1]

for i in range(N):
    if tmp=="":
        tmp=s[i]
        t +=1
    else:
        if s[i]==tmp:
            t +=1
            ans = [i-t+1,i]
        else:
            rl.append([tmp,t,i])
            tmp=s[i]
            t=1
else:
    rl.append([tmp,t,N])
    
#print(rl)
if ans!=[-1,-1]:
    print(*ans)
else:
    leng=len(rl)
    if leng>=3:
        for j in range(2,leng):
            if rl[j][0]==rl[j-2][0]:
                ans=[rl[j][2]-2,rl[j][2]]
                break
    print(*ans)
