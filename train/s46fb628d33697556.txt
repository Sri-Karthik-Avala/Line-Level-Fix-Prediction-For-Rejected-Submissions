#include <algorithm>
#include <iostream>
#include <queue>
using namespace std;

struct Graph
{
    int n;
    vector<vector<int>> g;
    
    Graph(int n) : n(n){
        g.resize(n);
    }
    
    void add_edge(int from, int to){
        g[from].push_back(to);
    }
};

bool dfs(int u, Graph &g, int *a){
    for(int v : g.g[u]){
        if(a[v] < a[u]){
            if(!dfs(v, g, a)) return true;
        }
    }
    return false;
}

int main()
{
    int n;
    cin >> n;
    int a[3005];
    for(int i = 0; i < n; i++) cin >> a[i];
    Graph g(n);
    for(int i = 0; i < n; i++){
        int u, v;
        cin >> u >> v;
        u--; v--;
        g.add_edge(u, v);
        g.add_edge(v, u);
    }
    for(int i = 0; i < n; i++){
        if(dfs(i, g, a)) cout << i + 1 << " ";
    }
    cout << endl;
}
