void solve(int a){
	if(a==27){
		int ret=0;
		for(int i=0;i<9;i++){
			ret+=t[i*3+2][2];
		}
		S.insert(ret);return;
	}
	for(int i=0;i<sz;i++){
		if(a<9&&b[2-a/3][a%3]&&b[2-a/3][a%3]!=pat[i][0])continue;
		if(a%9/3==0&&c[a/9][a%3]&&c[a/9][a%3]!=pat[i][1])continue;
		if(a%3&&pat[i][3]+t[a-1][2]!=7)continue;
		if(a%9/3&&pat[i][1]+t[a-3][4]!=7)continue;
		if(a/9&&pat[i][0]+t[a-9][5]!=7)continue;
		for(int j=0;j<6;j++)t[a][j]=pat[i][j];
		solve(a+1);
	}
}

	int T;scanf("%d",&T);

	while(T--){
		for(int i=0;i<3;i++)for(int j=0;j<3;j++)scanf("%d",&b[i][j]);
		for(int i=0;i<3;i++)for(int j=0;j<3;j++)scanf("%d",&c[i][j]);
		S.clear();
		solve(0);
		if(!S.size())printf("0\n");
		else{
			bool fi=false;
			for(set<int>::iterator it=S.begin();it!=S.end();it++){
				if(fi)printf(" ");
				fi=true;
				printf("%d",(*it));
			}
			printf("\n");
		}
	}
}
