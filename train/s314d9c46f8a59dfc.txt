#include <bits/stdc++.h>

using namespace std;

int main(){
  int n, k;
  int sum = 0;
  int L[100] = {0};
  scanf("%d%d", &n, &k);
  for(int i = 0;i < n;++i) scanf("%d", &L[i]);
  sort(L, L + n - 1);
  for(int i = 0;i < n - k;++i) L[i] = 0;
  for(int i = 0;i < n;++i) sum += L[i];
  printf("%d\n", sum);
  return 0;
}
