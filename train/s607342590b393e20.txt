//repetition

//constant
//--------------------------------------------
const double EPS = 1e-10;
const double PI  = acos(-1.0);

int dp[11][100010];
int solve(int N, VI& d, VI& p, int X){
  fill((int*)dp, (int*)dp+11*100010, -1);
  dp[0][0] = 0;
  REP(i,N){
	REP(x,X+1){
	  if(dp[i][x] < 0) continue;
	  dp[i+1][x] = max(dp[i+1][x], dp[i][x]);
	  
	  for(int tx=x,j=0;tx<=X;tx+=p[i],++j){
		dp[i+1][tx] = max(dp[i+1][tx], dp[i][x]+d[i]*j);
	  }
	}
  }
  
  return dp[N][X];
}

int main(){
  cin.tie(0);
  ios_base::sync_with_stdio(false);

  int N, D, X; cin >> N >> D >> X;
  VVI p(D, VI(N));
  REP(d,D) REP(i,N) cin >> p[d][i];
  REP(d,D) p[d].PB(1);

  REP(d,D-1){
	X = solve(N+1, p[d+1], p[d], X);
  }
  cout << X << endl;
  
  return 0;
}
