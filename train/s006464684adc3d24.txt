typedef pair<int,int> P;
typedef pair<int,P> P1;
typedef pair<P,P> P2;

#define pu push
#define pb push_back
#define mp make_pair
#define eps 1e-7
#define INF 1000000000
#define fi first
#define sc second
#define rep(i,x) for(int i=0;i<x;i++)
int m,n,kk,d;
int convert(string s)
{
	if(s == "H") return 0;
	if(s == "D") return m+n+1;
	if(s[0] == 'C')
	{
		return (s[1]-'0');
	}
	if(s[0] == 'L')
	{
		int t = 0;
		for(int i=1;i<s.size();i++)
		{
			t = t*10+(s[i]-'0');
		}
		return m+t;
	}
}
int dp[64][110];
int main()
{
	while(1)
	{
		cin >> m >> n >> kk >> d; if(m == 0) return 0;
		vector<P>edge[110];
		int a[10];
		for(int i=0;i<m;i++)
		{
			cin >> a[i];
		}
		for(int i=0;i<d;i++)
		{
			string a;
			string b;
			int dd;
			cin >> a >> b >> dd;
			int s = convert(a);
			int t = convert(b);
			edge[s].pb(mp(t,dd));
			edge[t].pb(mp(s,dd));
		}
		for(int i=0;i<64;i++)
		{
			for(int j=0;j<110;j++)
			{
				dp[i][j] = INF;
			}
		}
		dp[0][0] = 0;
		while(1)
		{
			bool upd = 0;
			for(int i=0;i<(1<<m);i++)
			{
				for(int j=0;j<=n+m+1;j++)
				{
					for(int k=0;k<edge[j].size();k++)
					{
						int x = edge[j][k].fi;
						int y = edge[j][k].sc;
						int mask = i;
						int g = 0;
						if(1<=x && x<=m)
						{
							if(((mask >> (x-1))&1)) continue;
							mask |= (1<<(x-1));
							g = a[x-1];
						}
						if(dp[mask][x] > dp[i][j]+y*kk-g)
						{
							dp[mask][x] = dp[i][j]+y*kk-g;
							upd = true;
						}
					}
				}
			}
			if(upd) break;
		}
		int res = INF;
		for(int i=0;i<(1<<m);i++) res = min(res,dp[i][m+n+1]);
		cout << res << endl;
	}
}
