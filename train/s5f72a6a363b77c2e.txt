#include <iostream>
#include <cstdio>

using namespace std;

#define MAX_N 100000
#define CUT 1000

double ans[MAX_N+1] = {0};
double b[MAX_N+1];
double memo[3][CUT+1] = {0};

double E(int rest, int con) {
	if (con == 0 && ans[rest] > 0) return ans[rest];
	if (rest == 0) return 0;
//	if (con > 1000) return E(rest-1, 0);

	double p = b[con];
	double ret = p * (E(rest-1, con+1) + 1) + (1-p) * E(rest-1, 0);
	if (con == 0) ans[rest] = ret;
	return ret;
}

int main() {
	b[0] = 1.0;
	for (int i = 1; i <= MAX_N; ++i) {
		b[i] = b[i-1] / 2;
	}
	for (int i = 1; i <= MAX_N; ++i) {
		for (int j = 0; j < min(CUT, MAX_N-i+1); ++j) {
			if (j == CUT) b[j] * ((1-b[j+1])*memo[(i+1)%3][0]+1) + (1-b[j]) * memo[(i-1)%3][0];
			else memo[i%3][j] = b[j] * (memo[(i-1)%3][j+1]+1) + (1-b[j]) * memo[(i-1)%3][0];
		}
		ans[i] = memo[i%3][0];
	}
	ans[0] = 0;

//	E(MAX_N, 0);

	int n;
	while ( cin >> n, n ) {
//		printf("%.10f\n", E(MAX_N, 0));
		printf("%.10f\n", ans[n]);
	}
	for (int i = 0; i < 100; ++i) printf("%.10f\n", ans[i]);
//	printf("%.10f\n", E(MAX_N, 0));
//	printf("%.10f\n", ans[MAX_N]);
}
