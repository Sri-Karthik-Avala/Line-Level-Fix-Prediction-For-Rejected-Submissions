#include<iostream>
#include<string>
using namespace std;
char C[63] = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz";
int convert(int p, string S) {
	int power[20]; power[0] = 0; for (int i = 1; i <= S.size(); i++)power[i] = power[i - 1] * p;
	int cnt = 0;
	for (int i = 0; i < S.size(); i++) {
		for (int j = 0; j < p; j++) {
			if (C[j] != S[i])continue;
			cnt += power[S.size() - 1 - i] * j;
		}
	}
	return cnt;
}
int main() {
	int n; cin >> n; int W = 0;
	for (int i = 0; i < n; i++) {
		string U; int a; cin >> a >> U; a = convert(a, U);
		W ^= a;
	}
	if (W == 0)cout << "lose" << endl;
	else cout << "win" << endl;
	return 0;
}
