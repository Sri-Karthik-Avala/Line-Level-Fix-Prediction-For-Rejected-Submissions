const int inf=1000000010;
const ll INF=10000000000000010LL;
const int mod=1000000007;
const int MAXN=1010, M=4;

struct Matrix{
	ll A[M][M];
	Matrix(){
		memset(A, 0, sizeof(A));
	}
	ll* operator [](int i){
		return A[i];
	}
};
Matrix operator *(Matrix A, Matrix B){
	Matrix C;
	for (int i=0; i<M; i++) for (int k=0; k<M; k++) for (int j=0; j<M; j++)
		C[i][j]=(C[i][j] + A[i][k]*B[k][j])%mod;
	return C;
}
Matrix operator ^(Matrix A, ll k){
	Matrix B;
	for (int i=0; i<M; i++) B[i][i]=1;
	for (; k; k>>=1, A=A*A) if (k&1) B=B*A;
	return B;
}

ll n, m, k, u, v, x, y, t, a, b, c, ans, cnt;
char A[MAXN][MAXN];
Matrix T;

int main(){
	ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);
	//freopen("input.txt", "r", stdin);
	//freopen("output.txt", "w", stdout);
	cin>>n>>m>>k;
	for (int i=1; i<=n; i++) for (int j=1; j<=m; j++) cin>>A[i][j];
	for (int i=1; i<=n; i++) if (A[i][1]=='#' && A[i][m]=='#') a++;
	for (int i=1; i<=m; i++) if (A[1][i]=='#' && A[n][i]=='#') b++;
	for (int i=1; i<=n; i++) for (int j=1; j<=m; j++) cnt+=(A[i][j]=='#');
	if (a && b) kill(1)
	if (!a && !b){
		T[0][0]=cnt;
		T=T^k;
		kill(T[0][0])
	}
	if (b){
		for (int i=1; i<MAXN; i++) for (int j=1; j<i; j++) swap(A[i][j], A[j][i]);
		swap(n, m);
		swap(a, b);
	}
	T[0][0]=cnt;
	for (int i=1; i<=n; i++) for (int j=1; j<m; j++)
		T[0][1]-=(A[i][j]=='#' && A[i][j+1]=='#');
	T[1][1]=a;
	T=T^(k-1);
	ans=(T[0][0] + T[0][1])%mod;
	if (ans<0) ans+=mod;
	cout<<ans<<'\n';
	
	return 0;
}
