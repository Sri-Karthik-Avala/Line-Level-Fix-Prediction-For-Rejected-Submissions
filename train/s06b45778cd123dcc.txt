int N,M;
int distnce[1024];
int weather[1024];
int memo[1024][1024];
int solve(int index,int day){
	if(day == M){
		if(index == N)return 0;
		return INF;
	}
	if(memo[index][day])return memo[index][day];
	int ans = INF;
	ans = min(solve(index,day+1),solve(index+1,day+1)+weather[day]*distnce[index]);
	return memo[index][day] = ans;
}

		cin >> weather[i];
	}
	printf("%d\n",solve(0,0));

}
