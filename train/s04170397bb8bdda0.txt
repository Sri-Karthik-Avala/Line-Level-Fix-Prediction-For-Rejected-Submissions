#include <iostream>
#include <algorithm>
#define int long long
using namespace std;

int n, p, q;
int c[500000];
int rc[500001];

signed main() {
	int i;
	
	cin >> n >> p >> q;
	for (i = 0; i < n; i++) { cin >> c[i]; c[i] += i * p; }
	
	sort(c, c + n);
	for (i = 1; i <= n; i++) { rc[i] = rc[i - 1] + c[i - 1]; }
	
	int ans = -1e+18;
	for (i = 0; i <= n; i++) {
		int jisui = i * q + 2 * p * i * (i - 1) / 2;
		int syoku = rc[n] - rc[i];
		ans = max(ans, jisui + syoku);
	}
	ans -= p * n * (n - 1) / 2;
	cout << ans << endl;
	return 0;
}
