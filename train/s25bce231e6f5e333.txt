#include <iostream>
#include <algorithm>
using namespace std;
int n, a, c[126]; long long s;
int main() {
	cin >> n;
	for (int i = 0; i < n; i++) {
		cin >> a; s += a;
		for (int i = 1; i <= 125 && a <= i; i++) c[i]++;
	}
	for (int i = 1; i <= 125; i++) s -= max(c[i] - i * 4, 0);
	cout << s + 1 << endl;
	return 0;
}
