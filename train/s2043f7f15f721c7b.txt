#include<stdio.h>
int main(void){
    int a[1000],x,i,j,n=3;
    scanf("%d",&n);
    for(i=0;i<n;i++){
            scanf("%d",&a[i]);
    }
    for(i=0;i<n-1;i++){
        for(j=i+1;j<n;j++){
            if(a[i]>a[j]){
                x=a[i];
                a[i]=a[j];
                a[j]=x;
            }
        }
    }
    for(i=0;i<n;i++){
        if(i!=0){
            printf(" ");
        }
            printf("%d",a[i]);
    }
    printf("\n");
return 0;
}
