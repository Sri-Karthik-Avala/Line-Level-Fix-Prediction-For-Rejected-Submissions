int eval(int b,int e){

	int l,o,r,d;
	for(o=b,d=0;o<e;o++){
		char c=expr[o];
		if(c=='(')d++;
		if(c==')')d--;
		if(d==0&&(c=='&'||c=='|'||c=='^'))break;
	}

	l=eval(b,o),r=eval(o+1,e);char c=expr[o];
	return c!='&'?c=='^'?l^r:l|r:l&r;
}
