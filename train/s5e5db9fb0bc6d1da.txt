#include <cstdio>
#include <cstring>

char s[401000];
int n, k;

int main()
{
	scanf("%d%d%s", &n, &k, s + 1);
	bool ok = 1;
	for(int i = 1; i <= n; i++)
		if(s[i] == 'A') {ok = 0; break;} 
	if(ok)
	{
		k--;
		for(int i = 1; i <= n; i++)
			s[i] = 'A';
	}
	int h = 1, t = 0;
	if(n == 1) t = k & 1;
	else
		while(k--)
		{
			if(h > n) break;
			if(t) s[h] = s[h] == 'A' ? 'B' : 'A';
			if(s[h] == 'B')
			{
				t ^= 1;
				if(t == 1) s[n + h] = 'B';
				else s[n + h] = 'A';
				h++;
			}
			else
			{
				if(t == 1) s[h] = 'A';
				else s[h] = 'B';
			}
		}
	if(t)
	{
		for(int i = h; i < h + n; i++)
			s[i] = s[i] == 'A' ? 'B' : 'A';
	}
	if(k >= 0 && (k & 1) && s[h] == 'A') s[h] = 'B';
	printf("%s\n", s + h);
}
