#include <bits/stdc++.h>
#define _GLIBCXX_DEBUG
using namespace std;
using ll = long long;
#define rep(i, n) for (int i = 0; i < (int)(n); i++)
#define rep2(i, s, n) for (int i = (s); i < (int)(n); i++)
int main() {
  int W, A, B;
  cin >> W >> A >> B;
  int X;
  if(A <= B) {
    X = B - A + W;
  }
  else {
    X = A - B - W;
  }
  if(X < 0) {
    X = 0;
  }
  cout << X << endl;
}
