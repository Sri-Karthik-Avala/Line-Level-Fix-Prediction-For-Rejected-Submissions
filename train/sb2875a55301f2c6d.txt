import queue
adjacent=((1,3), (0,2,4), (1,5), (0,4,6), (1,3,5,7), (2,4,8), (3,7), (4,6,8), (5,7))

class State:
    def __init__(self, board, space, prev):
        self.board = board
        self.space = space
        self.prev = prev

def bf_search(start, GOAL):
    q = queue.Queue()
    q.put(State(start, start.index(0), None))
    table = {}
    table[tuple(start)] = True
    while not q.empty():
        a = q.get()
        for x in adjacent[a.space]:
            b = a.board[:]
            b[a.space] = b[x]
            b[x] = 0
            key = tuple(b)
            if key in table: continue
            c = State(b,x,a)
            if b == GOAL:
                return print_answer(c)
            q.put(c)
            table[key] = True

cnt = -1
def print_answer(x):
    global cnt
    if x is not None:
        cnt += 1
        print_answer(x.prev)
    return str(cnt)

GOAL = [1,2,3,4,5,6,7,8,0]
start = []
for i in range(3):
 x,y,z = map(int, input().split())
 start += [x,y,z]

print(bf_search(start, GOAL))
