#define maxn 300005

int a[maxn] , cnt[maxn] , pos[maxn] ;

int main()
{
    int n , m ;
    int q ;

    scanf("%d %d",&n,&m) ;
    scanf("%d",&q) ;

    for(int i=1; i<=q ; i++) scanf("%d",&a[i]) ;

    vector <int> perm ;
    set<int> st ;
    for(int i=1 ; i<=m ; i++) st.insert(i) ;

    for(int i=q ; i>=1 ; i--)
    {
        if( st.find(a[i]) != st.end() )
        {
            perm.pb(a[i]) ;
            st.erase( st.find(a[i]) ) ;
        }
    }

    for(int v: st) perm.pb(v) ;

    while( perm.size() >= 2 )
    {
        int sz = perm.size() ;
        if( perm[sz-2] < perm[sz-1] ) perm.pop_back() ;
        else break ;
    }

    perm.pop_back() ;

    for(int i=1 ; i<=n ; i++) pos[i] = -1 ;
    for(int i=0 ; i<perm.size() ; i++) pos[ perm[i] ] = i ;

    for(int i=q ; i>=1 ; i--)
    {
        if( pos[ a[i] ] == -1 ) continue ;

        if( pos[ a[i] ] == 0 ) cnt[ pos[ a[i] ] ]++ ;
        else{
            cnt[ pos[ a[i] ] ]++ ;
            cnt[ pos[ a[i] ] ] = min( cnt[ pos[a[i]] ] , cnt[ pos[a[i]] - 1 ] ) ;
        }
    }

    int fl = 1 ;

    for(int i=0 ; i<perm.size() ; i++)
    {
        if( cnt[i] < n ) fl = 0 ;
    }

    if(fl) printf("Yes\n") ;
    else printf("No\n") ;

    return 0 ;
}
