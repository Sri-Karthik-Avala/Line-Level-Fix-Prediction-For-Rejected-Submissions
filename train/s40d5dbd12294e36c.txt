#include<bits/stdc++.h>
#include<cmath>
using namespace std;

int main(){
    long long a,v,b,w,t; cin >> a >> v >> b >> w >> t;
    if (abs(v-w) == 0)
    {
        cout << "NO" << endl;
    }
    
    else if ((abs(a-b)+abs(v-w)-1)/abs(v-w)<=t)
    {
        cout << "YES" << endl;
    }
    else
    {
        cout << "NO" << endl;
    }
    
    
}
