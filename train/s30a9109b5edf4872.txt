class Octahedra{
	public:
		string color[8];
		void right(){
			string tmp[8];
			rep(i,8) tmp[i]=color[i];
			color[0]=tmp[4],color[1]=tmp[5],color[2]=tmp[1],color[3]=tmp[0];
			color[4]=tmp[7],color[5]=tmp[6],color[6]=tmp[2],color[7]=tmp[3];
		}
		void top(){
			string tmp[8];
			rep(i,8) tmp[i]=color[i];
			color[0]=tmp[4],color[1]=tmp[0],color[2]=tmp[3],color[3]=tmp[7];
			color[4]=tmp[5],color[5]=tmp[1],color[6]=tmp[2],color[7]=tmp[6];
		}
		void rotation(){
			rep(i,3) swap(color[i],color[i+1]);
			rep(i,3) swap(color[i+4],color[i+1+4]);
		}

		bool operator<(Octahedra a) const{
			rep(i,8) if(color[i]!=a.color[i]) return color[i]<a.color[i];
			return color[7]<a.color[7];
		}
};
