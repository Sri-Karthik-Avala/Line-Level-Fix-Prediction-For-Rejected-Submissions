#include "bits/stdc++.h"
#include<unordered_map>
#include<unordered_set>
#pragma warning(disable:4996)
using namespace std;
using ld =long double;
const ld eps = 1e-9;

#define Seg_Max_N (1<<18) 

using Value = int;
const Value ini = 0;
struct segtree {
	int N;
	vector<Value>dat;

	segtree() {}
	segtree(int n) :dat(2 * Seg_Max_N) {
		N = 1;
		while (N < n) N *= 2;
		for (int i = 0; i < 2 * N - 1; i++) {
			dat[i] = ini;
		}
	}
	Value connect(const Value&l, const Value&r) {
		return l+r;
	}
	// update k th element
	void update(int k, Value a) {
		k += N - 1;
		dat[k] = a;

		while (k > 0) {
			k = (k - 1) / 2;
			const Value al(dat[k * 2 + 1]);
			const Value ar(dat[k * 2 + 2]);
			dat[k] = connect(al, ar);
		}
	}
	// min [a, b)
	Value  query(int a, int b) { return query(a, b, 0, 0, N); }
	Value  query(int a, int b, int k, int l, int r) {
		if (r <= a or b <= l) return ini;
		if (a <= l and r <= b) return dat[k];
		const int m = (l + r) / 2;
		const Value al(query(a, b, k * 2 + 1, l, m));
		const Value ar(query(a, b, k * 2 + 2, m, r));
		return connect(al, ar);
	}
};
int main()
{
	int N,M,T;cin>>N>>M>>T;
	vector<vector<long long int>>dp(20,vector<long long int>(T,-1e9));

	for (int i = 0; i < N; ++i) {
		int a;cin>>a;
		dp[0][a%T]=a/T;
	}

	for (int i = 0; i < 19; ++i) {
		for (int x = 0; x < T; ++x) {
			for (int y = 0; y < T; ++y) {
				if (dp[i][x] >= 0 && dp[i][y] >= 0) {
					dp[i+1][(x+y)%T]=max(dp[i+1][(x+y)%T],dp[i][x]+dp[i][y]+(x+y>=T));
				}
			}
		}
	}
	bitset<20>bs(M);
	vector<long long int>ndp(T, -1e9);
	{
		ndp[0]=0;
		for (int i = 0; i < 20; ++i) {
			if (bs[i]) {
				auto next_dp(ndp);
				for (int x = 0; x < T; ++x) {
					for (int y = 0; y < T; ++y) {
						if (ndp[x] >= 0 && dp[i][y] >= 0) {
							next_dp[(x + y) % T] = max(next_dp[(x + y) % T], ndp[x]+ dp[i][y] + (x + y >= T));
						}
					}
				}
				ndp=next_dp;
			}
		}
	}
	int Q;cin>>Q;
	while (Q--) {
		int A,B;cin>>A>>B;
		bool ok=ndp[A]>=B;
		if(ok)cout<<"yes"<<endl;
		else cout<<"no"<<endl;
	}
	return 0;
}
