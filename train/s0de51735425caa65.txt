class Point
{
public:
    int y, x;
    Point(){
        y = x = 0;
    }
    Point(int y0, int x0){
        y = y0;
        x = x0;
    }
    Point operator+(const Point& p) const{
        return Point(y + p.y, x + p.x);
    }
    Point operator-(const Point& p) const{
        return Point(y - p.y, x - p.x);
    }
    Point operator*(int a) const{
        return Point(y * a, x * a);
    }
    bool operator==(const Point& p) const{
        return y == p.y && x == p.x;
    }
    bool operator<(const Point& p) const{
        return make_pair(y, x) < make_pair(p.y, p.x);
    }
};

const Point d[] = {Point(0, -1), Point(0, 1), Point(1, 0), Point(-1, 0)};

int main()
{
    int h, w;
    cin >> h >> w;

    vector<Point> init;
    Point goal;
    for(int y=0; y<h; ++y){
        for(int x=0; x<w; ++x){
            char c;
            cin >> c;
            if(c == 'o')
                init.push_back(Point(y, x));
            else if(c == '@')
                goal = Point(y, x);
        }
    }
    if(init.size() > 2){
        cout << "yes" << endl;
        return 0;
    }

    set<vector<Point> > history;
    queue<vector<Point> > q;
    history.insert(init);
    q.push(init);
    while(!q.empty()){
        vector<Point> v = q.front();
        q.pop();
        for(unsigned i=0; i<v.size(); ++i){
            if(v[i] == goal){
                cout << "yes" << endl;
                return 0;
            }
        }

        for(unsigned i=0; i<v.size(); ++i){
            for(int j=0; j<4; ++j){
                vector<Point> v2 = v;
                for(;;){
                    Point p = v2[i] + d[j];
                    if(!(0 <= p.y && p.y < h && 0 <= p.x && p.x < w))
                        break;

                    for(unsigned k=0; k<v.size(); ++k){
                        if(v2[k] == p)
                            v2[k] = v2[k] - d[j];
                    }
                    v2[i] = p;
                }

                if(history.find(v2) == history.end()){
                    history.insert(v2);
                    q.push(v2);
                }
            }
        }
    }

    cout << "no" << endl;
    return 0;
}
