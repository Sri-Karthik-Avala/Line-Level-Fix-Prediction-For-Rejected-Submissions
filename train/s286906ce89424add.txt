using namespace std;
typedef long long ll;

map<string,ll> num;
ll par[5000],_rank[5000];
 
void init(ll n){
    for(ll i=0;i<n;i++) par[i]=i,_rank[i]=1;
}
 
ll find(ll a){
    if(par[a]==a) return a;
    else return par[a] = find(par[a]);
}
 
void unite(ll _a,ll _b){
    ll a = find(_a);
    ll b = find(_b);
    if(a==b) return;
    if(_rank[a]>_rank[b]){
        par[b] = a;
    }
    else{
        par[a] = b;
        if(_rank[a]==_rank[b]) _rank[b]++;
    }
}

	ll N;
	cin>>N;

		cin>>A[i]>>X[i];

	ll M;
	cin>>M;
	init(N);
	for(ll i=0;i<M;i++){
		string S,T;
		cin>>S>>T;
		unite(num[S],num[T]);
	}
	ll min_[5000]; fill_n(min_,N,LLONG_MAX);
