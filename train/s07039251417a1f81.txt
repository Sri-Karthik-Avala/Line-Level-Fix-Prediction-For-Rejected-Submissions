#include <iostream>
#include <map>
#include <algorithm>
using namespace std;

int main(){
  int N, M;
  cin >> N >> M;
  
  map<int, int> d;
  for(int i = 0;i < N;i++){
    int A, B;
    cin >> A >> B;
    d[A] = B;
  }
  
  long long int ans = 0;
  for(auto [A, B] : d){
    long long int b = min(M, B);
    ans += A * b;
    M -= b;
    if(!M){
      cout << ans << endl;
      return 0;
    }
  }
  
  return 0;
}
