#include<bits/stdc++.h>
#define MAX {INT_MAX,INT_MAX}
#define P pair<int,int>
#define rep(i,n)for(int i=0;i<(n);i++)
using namespace std;

int N;
P dat[600000];
void init(int n_) {
    N = 1;
    while (N < n_)N *= 2;
    for (int i = 0; i < 2 * N - 1; i++)dat[i] =MAX;
}
void update(int k, int a) {
    k += N - 1;
    dat[k] = {a,k-(N-1)};
    while (k > 0) {
        k = (k - 1) / 2;
        dat[k] = min(dat[k * 2 + 1], dat[k * 2 + 2]);
    }
}
P query(int a, int b, int k, int l, int r) {
    if (r <= a || b <= l)return MAX;
    if (a <= l&&r <= b)return dat[k];
    else {
        P vl = query(a, b, k * 2 + 1, l, (l + r) / 2);
        P vr = query(a, b, k * 2 + 2, (l + r) / 2, r);
        return min(vl, vr);
    }
}
int bit[200008], n;
void add(int k, int x) {
	k++;
    while (k <= n) {
        bit[k] += x;
        k += k&-k;
    }
}
int sum(int k) {
	k++;
    int res = 0;
    while (k > 0) {
        res += bit[k];
        k -= k&-k;
    }
    return res;
}
signed main(){
	string a;cin>>a;
	int b;cin>>b;
	init(a.length());
	n=a.length()+5;
	for(int i=0;i<(int)a.length();i++){update(i,a[i]-'a');add(i,i);add(i+1,-i);}
	string ans;
	for(int i=0;i<(int)a.length();i++){
		int l=0,r=N;//(l,r]
		while(r-l>1){
			int k=(l+r)/2;
			if(sum(k)<=b)l=k;
			else r=k;
			}
		P t=query(0,r,0,0,N);
		ans+=char(t.first+'a');
		add(t.second+1,-1);
		b-=sum(t.second);
		update(t.second,INT_MAX);
		}
	cout<<ans<<endl;
	}
