#include <iostream>
#include <vector>
using namespace std;

struct CODE{
	char before;
	char after;
};

int main()
{
	vector<CODE> conversion;
	int n,m;
	
	while(cin >> n, n != 0){
		while(n-- > 0){
			CODE tmp;
			cin >> tmp.before >> tmp.after;
			conversion.push_back(tmp);
		}
	
		cin >> m;
	
		while(m-- > 0){
			char c;
		
			cin >> c;
		
			for (int i = 0; i < conversion.size(); i++){
				if (conversion[i].before == c){
					c = conversion[i].after;
					break;
				}
			}
			cout << c;
		}
	
		cout << endl;
	}
	return (0);
}
