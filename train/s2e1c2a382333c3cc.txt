while 1:
    n, m, s = map(int, raw_input().split())
    if n == 0: break
    dp = [[0]*(s+1) for i in range(n*n+1)]
    dp[0][0] = 1
    for i in range(1, n * n + 1):
        for j in range(s+1):
            if j - i >= 0: dp[i][j] += (dp[i-1][j-i] + dp[i][j-i]) % 100000
            if j - m - 1 >= 0: dp[i][j] += (100000 - dp[i-1][j-m-1]) % 100000
    print dp[n*n][s]
