void solve()
{
	int n, d; cin >> n >> d;
	vector<vector<int>> x(n);
	for (int i = 0; i < n; ++i)
	{
		for (int j = 0; j < d; ++j)
		{
			int c; cin >> c;
			x[i].pb(c);
		}
	}
	set<int> sqr;
	for (int i = 0; i < 25; ++i)
	{
		sqr.insert(i * i);
	}
	ll res = 0;
	for (int i = 0; i < n; ++i)
	{
		for (int j = i + 1; j < n; ++j)
		{
			ll check = 0;
			for (int k = 0; k < d; ++k)
			{
				check += (x[i][k] - x[j][k]) * (x[i][k] - x[j][k]);
			}
			if (sqr.find(check) != sqr.end()) res++;
		}
	}
	cout << res;
}
