int n;

int dfs(int sum, int d, int c){
  //  printf("%d %d\n", sum, c);
  if(sum == n) return 1;
  if(c >= 4) return 0;
  int res = 0;
  REP(i, d, sqrt(n)+1){
    if(sum + i*i > n) break;;
    res += dfs(sum + i*i, i, c+1);
  }
  return res;
}

    printf("%d\n", dfs(0, 1, 0));
  }
