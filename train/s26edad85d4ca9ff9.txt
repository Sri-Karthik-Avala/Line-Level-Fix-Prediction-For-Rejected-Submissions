#include<stdio.h>
#include<algorithm>
#include<math.h>
#include<queue>
using namespace std;
double x[510];
double y[510];
int bfs[510];
int main(){
	int a,b,c;
	scanf("%d%d%d",&a,&b,&c);
	for(int i=0;i<c;i++){
		scanf("%lf%lf",x+i,y+i);
	}
	double left=0;
	double right=1100000000;
	for(int i=0;i<50;i++){
		double M=(left+right)/2;
		for(int j=0;j<c;j++)bfs[j]=0;
		queue<int>Q;
		for(int j=0;j<c;j++){
			if(x[j]<M){
				bfs[j]=1;
				Q.push(j);
			}
		}
		while(Q.size()){
			int at=Q.front();Q.pop();
			for(int j=0;j<c;j++){
				if(bfs[j])continue;
				if((x[j]-x[at])*(x[j]-x[at])+(y[j]-y[at])*(y[j]-y[at])<4*M*M){
					bfs[j]=1;
					Q.push(j);
				}
			}
		}
		bool dame=false;
		for(int j=0;j<c;j++){
			if(bfs[j]&&x[j]>(double)b-M){
				dame=true;
			}
		}
		if(dame)right=M;
		else left=M;
	}
	printf("%.12f\n",left);
}
