#include<cstdio>
 
int n,m,q_n;
int a[2000001],q;
 
int main(void){
    scanf("%d%d%d",&n,&m,&q_n);
    for(int i=0;i<m;i++)scanf("%d",&a[i]);
    for(int i=0;i<q_n;i++){
    	scanf("%d",&q);
       	int flag=0,p=q,b=0;
       	for(int j=0;j<m;j++){
       		if(flag==1)continue;
       		if(a[j]%2==0)b=(b+a[j])%(n-j);
       		else b=(b-a[j]+(n-j)*100)%(n-j);
       		if(p==b)flag=1;
       		else if(p>b)p--;
       		if(j!=m-1)b=(b+1)%(n-j);
       	}
        printf("%d\n",1-flag);
    }
    return 0;
}
