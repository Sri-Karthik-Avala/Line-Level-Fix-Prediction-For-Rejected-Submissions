ll n, m, l;
ll a[2000];
ll b[100000];
ll s[100010];
ll dp[2000][2000];

ll sum(ll i, ll j){
  return (s[max(i, j) + 1] - s[min(i, j)]) / l;
}

ll dfs(ll i, ll j){
  if (i == n) return sum(a[i - 1], j);
  if (~dp[i][j]) return dp[i][j];
  ll vl = sum(a[i - 1], a[i]) + dfs(i + 1, j);
  ll vr = sum(a[j], a[i]) + dfs(i + 1, i - 1);
  return dp[i][j] = min(vl, vr);
}

int main(){
  cin >> n >> m >> l;
  rep(i, n) cin >> a[i], a[i]--;
  rep(i, m) cin >> b[i];
  rep(i, m) s[i + 1] = s[i] + b[i];
  sort(a, a + n);
  memset(dp, -1, sizeof(dp));
  cout << dfs(1, 0) << endl;
}
