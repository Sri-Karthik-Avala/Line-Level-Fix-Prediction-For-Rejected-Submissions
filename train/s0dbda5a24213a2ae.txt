#include <bits/stdc++.h>
using namespace std;

int main() {
    int a, b, c, x, y;
    cin >> a >> b >> c >> x >> y;
    
    int ans = 0;
    int minPizza = min(x, y);
    if (a + b > 2 * c && x >= 1 && y >= 1) {
        ans += minPizza * 2 * c;
        x -= minPizza;
        y -= minPizza;
    }
    if (a > 2 * c && x >= 1) {
        ans += 2 * c * x;
        y -= x;
        x = 0;
    }
    if (b > 2 * c && y >= 1) {
        ans += 2 * c * y;
        x -= y;
        y = 0;
    }
    
    ans += a * x;
    ans += b * y;
    
    cout << ans << endl;
    
}
