def f1(a, b):
    x, y = (a, b) if a >= b else (b, a)
    g = gcd(x, y)
    return lcm(x, y, g)
    
def f2(l, *s):
    for v in s:
        yield l/v
        
def f3(_data):
    d1, v1 = _data
    for d2, v2 in data:
        ans = str((float(d1)/v1)/(float(d2)/v2))
        print int(ans[:-2])
        
def gcd(x, y):
    while y != 0:
        x = x-(x/y)*y
        x, y = y, x
    else:
        return x
    
def lcm(x, y, g):
    return (x*y)/g
