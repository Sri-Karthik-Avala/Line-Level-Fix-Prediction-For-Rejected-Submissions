inline ll solve (ll s, int b) {
	if (s < 0) return 0;
	if (s == 0 && b == 0) {
	//	cout << "was: 1" << '\n';
		return 1;
	}
	if (2ll * (pw2[b] - 1) <= s) {
	//	cout << "was: 2" << '\n';
		return pw3[b];
	}  

	if (s > N || b > LIM) {
		//cout << (b > LIM) << '\n';
		int res = (solve(s, b-1) + solve(s - pw2[b], b-1) + solve(s - pw2[b-1], b-1)) % mod;

		return res;
	}

	int &res = dp[b][s];
	//cout << res << ":\n";
	if (u[b][s]) return res;
	u[b][s] = 1;

	res = (solve(s, b-1) + solve(s - pw2[b], b-1) + solve(s - pw2[b-1], b-1)) % mod;
	return res; 
}
