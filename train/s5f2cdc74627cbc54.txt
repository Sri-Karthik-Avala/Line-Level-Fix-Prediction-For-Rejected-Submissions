ef search(l,r) :
    counter =0;
    j=0
    i=0
    while j < len(r):
        while i < len(l):
            if r[j] > l[i]:
                i+=1
            elif r[j]==l[i]:
                counter +=1
                break
            elif r[j] < l[i]:
                break
        j+=1

    return  counter
