#include <cstdio>
#include <algorithm>
#include <queue>
#define REP(i,n) for(int i=0; i<(int)(n); i++)

using namespace std;

int main(){
  int r, c, q;

  while(scanf("%d%d%d", &r, &c, &q), r + c + q){
    vector<int> a(q);
    vector<int> b(q);
    vector<int> o(q);

    vector<int> rs(r, -1);
    vector<int> cs(r, -1);

    REP(i,q) scanf("%d%d%d", &a[i], &b[i], &o[i]);

    reverse(a.begin(), a.end());
    reverse(b.begin(), b.end());
    reverse(o.begin(), o.end());

    int rr = 0;
    int cc = 0;
    int ans = 0;

    REP(i,q){
      if(a[i] == 0){
	if(rs[b[i]] == -1){
	  rs[b[i]] = o[i];
	  rr++;
	  if(o[i] == 1) ans += c - cc;
	}
      }else{
	if(cs[b[i]] == -1){
	  cs[b[i]] = o[i];
	  cc++;
	  if(o[i] == 1) ans += r - rr;
	}
      }
    }

    printf("%d\n", ans);
  }

  return 0;
}
