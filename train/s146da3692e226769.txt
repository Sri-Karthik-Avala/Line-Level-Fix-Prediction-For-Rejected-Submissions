#include<cstdio>
#include<vector>
std::vector<int> a[100005];
int n,k,cnt,used[100005];
int ans[100005],p[100005],pp[100005],q[100005],qq[100005],g[100005],gg[100005],gk[100005],ggk[100005],f[10][100005],ff[10][10005];
int main()
{
	scanf("%d%d",&n,&k);
	for (int i=1;i<=n;i++) scanf("%d",&p[i]);
	for (int i=1;i<=n;i++) scanf("%d",&q[i]);
	for (int i=1;i<=n;i++) f[0][i]=i;
	for (int i=1;i<=n;i++) ff[0][i]=i;
	for (int i=1;i<=n;i++) f[1][i]=p[i];
	for (int i=1;i<=n;i++) f[2][i]=q[i];
	for (int i=1;i<=n;i++) ff[1][f[1][i]]=i;
	for (int i=1;i<=n;i++) ff[2][f[2][i]]=i;
	for (int r=3;r<=6;r++)
	{
		for (int i=1;i<=n;i++) f[r][i]=f[r-1][ff[r-2][i]];
		for (int i=1;i<=n;i++) ff[r][f[r][i]]=i;
	}
	for (int i=1;i<=n;i++) pp[p[i]]=i;
	for (int i=1;i<=n;i++) qq[q[i]]=i;
	for (int i=1;i<=n;i++) g[i]=q[pp[qq[p[i]]]];
	for (int i=1;i<=n;i++) gg[g[i]]=i;
//	for (int i=1;i<=n;i++) printf("%d ",g[i]);puts("");
	for (int i=1;i<=n;i++)
	    if (!used[i])
	    {
			int x=++cnt;
			used[i]=1;
			a[x].push_back(i);
	    	for (int j=g[i];!used[j];j=g[j])
	    	{
	    		a[x].push_back(j);
	    		used[j]=1;
			}
	    }
	for (int i=1;i<=cnt;i++)
	{
		int sz=a[i].size();
		for (int j=0;j<a[i].size();j++)
		{
			int x=a[i][j],y=a[i][(j+(k-1)/6)%sz];
			gk[x]=y;
		}
	}
//	for (int i=1;i<=n;i++) printf("%d ",gk[i]);puts("");
	for (int i=1;i<=n;i++) used[i]=0;
	for (int i=1;i<=cnt;i++) a[i].clear();cnt=0;
	for (int i=1;i<=n;i++)
	    if (!used[i])
	    {
			int x=++cnt;
			used[i]=1;
			a[x].push_back(i);
	    	for (int j=gg[i];!used[j];j=gg[j])
	    	{
	    		a[x].push_back(j);
	    		used[j]=1;
			}
	    }
	for (int i=1;i<=cnt;i++)
	{
		int sz=a[i].size();
		for (int j=0;j<a[i].size();j++)
		{
			int x=a[i][j],y=a[i][(j+(k-1)/6)%sz];
			ggk[x]=y;
		}
	}
	for (int i=1;i<=n;i++) used[i]=0;
	for (int i=1;i<=cnt;i++) a[i].clear();cnt=0;
	for (int i=1;i<=n;i++) ans[i]=gk[f[(k-1)%6+1][ggk[i]]];
	
	for (int i=1;i<=n;i++) printf("%d ",ans[i]);
	puts("");
	return 0;
}
