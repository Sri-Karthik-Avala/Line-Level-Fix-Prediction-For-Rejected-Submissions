import numpy as np
import scipy.fftpack as fft

mod = 200003
n = int(input())
a = []
for i in map(int, input().split()):
    if i % mod == 0:
        n -= 1
    else:
        a.append(i % mod)

tab = [0] * mod
j = 1
for i in range(mod-2):
    tab[j] = i
    j = (j*2) % mod

b = [0] * (mod-1)
for i in a:
    b[tab[i]] += 1

b = np.array(b, dtype=np.float64)
b = fft.fft(b)
b = b*b
b = fft.ifft(b)

ans = 0
for i, j in enumerate(b):
    ans += int(round(j.real))*pow(2, i, mod)

for i in a:
    ans -= i*i % mod
print(ans//2)
