#include <iostream>

#define REP(i,n) for(int i=0; i<n; i++)
#define debug(x) cerr << #x << " = " << (x) << endl;

using namespace std;

int main(){
	int n, a[100000], q, k, l, r, m, ans;
	cin >> n;
	REP(i,n) cin >> a[i];
	cin >> q;
	REP(i,q){
		cin >> k;
		l = 0;
		r = n;
		m = -1;
		ans = 0;
		if (k<0 || k>n){
			cout << 0 << endl;
			continue;
		}
		while(1){
			m = (r+l)/2;
			//printf("r=%d, l=%d, m=%d\n", r, l, m);
			if (a[m]==k){
				ans = 1;
				break;
			}
			if (m==r || m==l) break;
			if (k<a[m]) r = m;
			else l = m;
		}
		cout << ans << endl;
	}
	return 0;
}
