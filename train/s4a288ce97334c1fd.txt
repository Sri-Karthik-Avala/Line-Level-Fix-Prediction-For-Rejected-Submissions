const int MOD = 1e9 + 7, MAXN = 1e6 + 10, INF = 1e18 + 10, BASE = 37;

void solve() {
    string s;
    cin >> s;
    vec<string> mask = {"dream", "dreamer", "erase", "eraser"};
    vec<int> dp(sz(s) + 1);
    dp[0] = 1;

    for (int i = 1; i <= sz(s); ++i) {
        for (auto msk : mask) {
            if (sz(msk) <= i) {
                string tmp = s.substr(i - sz(s), sz(s));
                if (tmp == msk && dp[i - sz(s)]) {
                    dp[i] = 1;
                }
            }
        }
    }

    cout << (dp[sz(s)] ? "YES" : "NO") << endl;
}
