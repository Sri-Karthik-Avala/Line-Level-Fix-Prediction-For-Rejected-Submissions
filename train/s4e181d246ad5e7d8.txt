#include <bits/stdc++.h>
using namespace std;
double eps = 0.0000001;
struct P {
  double x, y;
};
struct L {
  P p1, p2;
  double x() { return p2.x - p1.x; }
  double y() { return p2.y - p1.y; }
};
double prod(L l1, L l2) { return l1.x() * l2.y() - l1.y() * l2.x(); }
pair<double, double> interP(L l1, L l2) {
  double a = l1.x();
  double b = -l2.x();
  double c = l1.y();
  double d = -l2.y();
  double inv = 1. / (a * d - c * b);
  double e1 = -l1.p1.x + l2.p1.x;
  double e2 = -l1.p1.y + l2.p1.y;
  return make_pair((d * e1 - b * e2) * inv, (-c * e1 + a * e2) * inv);
}
bool intersect(L l1, L l2) {
  if (fabs(prod(l1, l2)) < eps) {
    return false;
  }
  auto p = interP(l1, l2);
  return eps < p.first && p.first < 1 - eps && eps < p.second &&
         p.second < 1 - eps;
}
int main() {

  int n, m;
  cin >> n >> m;
  vector<vector<P>> p(n);
  for (int i = 0; i < n; i++) {
    int l;
    cin >> l;
    for (int j = 0; j < l; j++) {
      double x, y;
      cin >> x >> y;
      p[i].push_back(P{x, y});
    }
  }
  vector<P> hum(m);
  for (int i = 0; i < m; i++) {
    cin >> hum[i].x >> hum[i].y;
  }
  vector<L> line;
  for (int i = 0; i < m; i++) {
    for (auto &poly : p) {
      for (int j = 0; j < poly.size(); j++) {
        line.push_back(L{hum[i], poly[j]});
      }
    }
  }
  vector<P> ps;
  for (int i = 0; i < line.size(); i++)
    for (int j = i + 1; j < line.size(); j++) {
      if (fabs(prod(line[i], line[j])) < eps)
        continue;
      auto rs = interP(line[i], line[j]);
      ps.push_back(P{line[i].p1.x + line[i].x() * rs.first,
                     line[i].p1.y + line[i].y() * rs.second});
    }

  if (n == 0) {
    cout << m << endl;
    return 0;
  }
  if (m == 1) {
    cout << 1 << endl;
    return 0;
  }

  int ans = 0;
  for (int i = 0; i < ps.size(); i++) {
    int score = m;
    for (int j = 0; j < m; j++) {
      L l1{ps[i], hum[j]};
      bool ng = false;

      for (auto &poly : p) {
        for (int k = 0; k < poly.size(); k++) {
          P p1 = poly[k];
          P p2 = poly[(k + 1) % poly.size()];
          L l2{p1, p2};
          if (intersect(l1, l2)) {
            ng = true;
          }
        }
      }

      if (ng) {
        score--;
      }
    }
    ans = max(ans, score);
  }
  cout << ans << endl;

  return 0;
}
