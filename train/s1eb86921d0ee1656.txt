int f[MaxN*MaxN];
signed main(void)
{
	int n;read(n);
	reg int ga,gb,sa,sb,ba,bb;
	read(ga,gb,sa,sb,ba,bb);
	for(int i=ga;i<=n;++i)
		checkmax(f[i],f[i-ga]+gb-ga);
	for(int i=sa;i<=n;++i)
		checkmax(f[i],f[i-sa]+sb-sa);
	for(int i=ba;i<=n;++i)
		checkmax(f[i],f[i-ba]+bb-ba);
	reg int t=0;
	for(int i=0;i<=n;++i)
		checkmax(t,f[i]);
	reg int ans=n+t;
	n=ans;
	mem(f,0);
	for(int i=gb;i<=n;++i)
		checkmax(f[i],f[i-gb]+ga-gb);
	for(int i=sb;i<=n;++i)
		checkmax(f[i],f[i-sb]+sa-sb);
	for(int i=bb;i<=n;++i)
		checkmax(f[i],f[i-bb]+ba-bb);
	t=0;
	for(int i=0;i<=n;++i)
		checkmax(t,f[i]);
	write(ans+t),ln;
	return 0;
}
