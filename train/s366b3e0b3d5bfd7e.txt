using namespace std;

int main() {
	int N, C; cin >> N >> C;
	const int cst = 10003;
	vector<vector<int>> vec(C, vi(2 * cst+1));
	for (int i = 0; i < N; i++) {
		int s, t, c; cin >> s >> t >> c;
		--c;
		for (int ind = 2 * s-1 ; ind < 2 *t; ind++) {
			vec[c][ind] = 1;
		}
	}
	int M = 0;
	for (int pos = 0; pos < vec[0].size(); pos++) {
		int tmp = 0;
		for (int channel = 0; channel < C; channel++) {
			tmp += vec[channel][pos];
		}
		M = max(M, tmp);
	}
	cout << M << endl;
}
