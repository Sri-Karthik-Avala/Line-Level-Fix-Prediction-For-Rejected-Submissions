#include<iostream>
using namespace std;
int main()
{
	int n,m,i,a[100],j,v,r,t;
	cin>>n>>m;
	for(i=0;i<n;i++)
	{
		cin>>a[i];
	}
	for(i=1;i<=m;i++)
	{
		for(j=1;j<n-1;j++)
		{
			if(j<=n)
			{
				v=a[j]%i;
				r=a[j+1]%i;
				if(v>r)
				{
					t=a[j];
					a[j]=a[j+1];
					a[j+1]=t;
				}
			}
		}
	}
	for(i=0;i<n;i++)
	{
		cout<<a[i]<<endl;
	}
				
				
				
	return 0;
}
