#include <iostream>
#include <algorithm>
#include <vector>
#include <cmath>
#include <string>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
const int mod = 1000000007;

int main() {
    ios::sync_with_stdio(false);
    int n, x[102] = {}, y[102] = {}, h[102] = {};
    cin >> n;
    for (int i = 0; i < n; i++) cin >> x[i] >> y[i] >> h[i];
    for (int i = 0; i < n; i++) if (h[i]) {
        swap(x[0], x[i]); swap(y[0], y[i]); swap(h[0], h[i]);
    }
    for (int i = 0; i <= 100; i++) for (int j = 0; j <= 100; j++) {
        int z = h[0] + abs(x[0] - i) + abs(y[0] - j);
        bool w = 0;
        for (int k = 1; k < n; k++) if (h[k] && h[k] + abs(x[k] - i) + abs(y[k] - j) != z) w = 1;
        if (!w) {
            cout << i << ' ' << j << ' ' << z;
            return 0;
        }
    }
    return 1;
}
