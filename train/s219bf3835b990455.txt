struct edge
{
	int from, to, cost;
	bool operator < (const edge& e) const { return cost < e.cost; }
	bool operator >(const edge& e) const { return cost > e.cost; }
};
///*************************************************************************************///
///*************************************************************************************///
///*************************************************************************************///
int n;
vector<vi> t;
int v[105];
int dp[105][1 << 17];

int solve(int i, int j)
{
	if (i == n) return 0;
	if (dp[i][j] >= 0) return dp[i][j];
	int tmp = j;
	bool f = true;
	REP(k, t[i].size())
	{
		if (((tmp >> t[i][k]) & 1) == 0) tmp |= (1 << t[i][k]);
		else f = false;
	}
	int res;
	if (f) res = max(solve(i + 1, j), solve(i + 1, tmp) + v[i]);
	else res = solve(i + 1, j);
	return dp[i][j] = res;
}

int main()
{
	while (cin >> n, n)
	{
		MS(dp, -1);
		t.resize(n);
		REP(i, n)
		{
			int k;
			cin >> k >> v[i];
			REP(j, k)
			{
				int a, b;
				cin >> a >> b;
				while (a != b)
				{
					t[i].push_back(a-6);
					a++;
				}
			}
		}
		cout << solve(0, 0) << endl;
	}
	return 0;
}
