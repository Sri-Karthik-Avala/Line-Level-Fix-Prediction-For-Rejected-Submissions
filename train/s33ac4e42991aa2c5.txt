#include <iostream>
#include <cstdlib>
#include <map>
#include <algorithm>
#define fr first
#define sc second
#define INF (2 << 28)
#define SQR(x) (abs(x) * abs(x))
using namespace std;

pair<int, int> p[500001];

int main(){
    
    int n;
    cin >> n;
    for(int i = 0; i < n; i++) cin >> p[i].sc >> p[i].fr;
    
    sort(p, p + n);
    int ret = INF;
    for(int i  = 0; i < n; i++){
        for(int j = i+1; j < n; j++){
            if(SQR(p[i].sc - p[j].sc) >= ret) break;
            ret = min(SQR(p[i].fr - p[j].fr) + SQR(p[i].sc - p[j].sc), ret);
        }
    }
    cout << ret << endl;
}
