#define rep(i,n) for(int i=0; i<(n); i++)
using ll = long long;
#define int long long
using P = pair<int,int>;
//#########################################
signed main(){
	int n;cin >> n;
	vector<int> a(n);
	rep(i,n)cin >> a[i];
	vector<int> sum(n+1);
	rep(i,n)sum[i+1] = sum[i] + a[i];
	auto get_sum = [&](int l,int r){
		l = max(l,0ll);
		r = min(r,n);
		return sum[r] - sum[l];
	};
	auto max_4 = [](int x,int y,int z,int w){
		return max(max(x,y),max(z,w));
	};
	auto min_4 = [](int x,int y,int z,int w){
		return min(min(x,y),min(z,w));
	};
	int ret = 1e18;
	// l c r
	for(int c = 2;c < n-2;c++){
		vector<int> boundary_l;
		{
			int top = c+1,bot = -1;
			while(top - bot > 1){
				int mid = (top + bot)/2;
				if(get_sum(0,mid) < get_sum(mid,c))bot = mid;
				else top = mid;
			}
			boundary_l = {bot,top};
		}
		vector<int> boundary_r;
		{
			int top = n+1,bot = c-1;
			while(top - bot > 1){
				int mid = (top + bot)/2;
				if(get_sum(c,mid) < get_sum(mid,n))bot = mid;
				else top = mid;
			}
			boundary_r = {bot,top};
		}
		for(auto l:boundary_l)for(auto r:boundary_r){
			int tmp = max_4(get_sum(0,l),get_sum(l,c),get_sum(c,r),get_sum(r,n)) - min_4(get_sum(0,l),get_sum(l,c),get_sum(c,r),get_sum(r,n));
			ret = min(ret,tmp);
		}
		
	}
	cout << ret << endl;
	return 0;
}
