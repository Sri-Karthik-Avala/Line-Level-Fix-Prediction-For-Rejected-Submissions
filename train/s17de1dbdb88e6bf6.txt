typedef struct _edge{
  int to,cost;
  _edge(int ito,int icost):to(ito),cost(icost){};
}edge;

vector<edge> E[60];
int s,d;
bool used[50];

int prim(){
  memset(used,0,sizeof(used));
  priority_queue<PI> Q;//cost,to;

  Q.push(mp(0,0));

  int ret=0;
  
  while(!Q.empty()){
    int ccost=-Q.top().F,from=Q.top().S;
    Q.pop();
    if(used[from])continue;
    used[from]=true;

    ret+=ccost;
    rep(i,E[from].size()){
      int to=E[from][i].to,cost=E[from][i].cost;
      if(used[to])continue;
      Q.push(mp(-cost,to));
    }
  }
  return ret;

}
    

main(){
  while(cin>>s>>d,s|d){
    rep(i,s+d)E[i].clear();
    rep(i,s){
      rep(j,d){
	int co;
	cin>>co;
	if(!co)continue;
	E[i].pb(edge(j+s,co));
      }
    }
    rep(i,d-1){
      rep(j,d-i-1){
	int co;
	cin>>co;
	if(!co)continue;
	int to=i+j+1+s;
	E[i+s].pb(edge(to,co));
	E[to].pb(edge(i+s,co));
      }
    }

    rep(i,s){
      rep(j,i){
	E[i].pb(edge(j,0));
	E[j].pb(edge(i,0));
      }
    }

    cout<<prim()<<endl;
  }
}
