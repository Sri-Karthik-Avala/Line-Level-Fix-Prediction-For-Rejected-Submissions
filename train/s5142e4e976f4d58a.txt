#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define INF 1LL<<62
#define inf 1000000007

ll h,w,k;
bool cut[12];
ll ch[12][1002];
ll ans=INF;
void cal(ll cnt){
	ll left=0;
	for(ll j=1;j<=w;j++){
		ll sum=0;
		for(ll i=1;i<=h;i++){
			sum+=ch[i][j]-ch[i][left];
			if(sum>k){
				cnt++;
				left=j;
				break;
			}
			if(cut[i]){
				sum=0;
			}
		}
	}
	ans=min(ans,cnt);
}
void dep(ll now,ll cnt){
	if(now==h){
		cal(cnt);
		return;
	}
	cut[now]=1;dep(now+1,cnt+1);
	cut[now]=0;dep(now+1,cnt);
}
int main() {
	cin>>h>>w>>k;
	for(ll i=0;i<h;i++){
		string s;
		cin>>s;
		for(ll j=0;j<w;j++){
			if(s[j]=='1'){
				ch[i+1][j+1]=1;
			}
		}
	}
	for(ll i=1;i<=h;i++){
		for(ll j=1;j<=w;j++){
			ch[i][j]=ch[i][j-1]+ch[i][j];
		}
	}
	dep(1,0);
	cout <<ans;
	// your code goes here
	return 0;
}
