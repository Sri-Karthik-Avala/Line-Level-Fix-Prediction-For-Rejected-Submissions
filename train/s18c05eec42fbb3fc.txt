int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    int n, qq;
    string s;
    cin >> n >> s >> qq;
    while ( qq-- ) {
        int k;
        cin >> k;
        int l=0, c1=0, c2=0, res=0, ans=0;
        for ( int i=0; i<n; i++ ) {
            if ( s[i]=='D' ) c1++;
            if ( s[i]=='M' ) {
                res+=c1;
                c2++;
            }
            if ( s[i]=='C' ) {
                while ( l<i && l+k<=i ) {
                    if ( s[l]=='D' ) {
                        res-=c2;
                        c1--;
                    }
                    if ( s[l]=='M' ) {
                        c2--;
                    }
                    l++;
                }
                ans+=res;
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
