bool search(int y,int x){
	if(y>H)

	if(!able(y,x))
		return search(y+x/W,x%W+1);

	int cans=able(y+1,x)+able(y,x+1);
	if(cans+shared[y][x]<2) return false;
	if(shared[y][x]==0){
		shared[y][x+1]++;
		shared[y+1][x]++;
		ans=search(y+x/W,x%W+1);
		shared[y][x+1]--;
		shared[y+1][x]--;
		return ans;
	}
	if(shared[y][x]==1){
		REP(i,2){

				ans|=search(y+x/W,x%W+1);
				shared[y+dy[i]][x+dx[i]]--;
			}
		}
		return ans;
		
	}
	
}
