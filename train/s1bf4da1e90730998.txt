#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

typedef pair <int, int> P;

int main(void)
{
	int n, p;

	while (cin >> n, n){
		vector < pair <P, int> > v, must;
		vector < string > name;
		string tmp;
		int week, start;
		for (int i = 0; i < n; i++){
			cin >> tmp >> week >> start;
			int s = (week * 60 * 60 * 24) + (start / 100 * 60 * 60) + (start % 100);
			int t = s + (60 * 30);
			v.push_back(make_pair(make_pair(t, s), i));
			name.push_back(tmp);
		}
		cin >> p;
		for (int i = 0; i < p; i++){
			cin >> tmp;
			for (int j = 0; j < n; j++){
				if (tmp == name[j]){
					must.push_back(make_pair(v[j].first, j));
					break;
				}
			}
		}

		sort(v.begin(), v.end());
		sort(must.begin(), must.end());

		int ret = -1, last = -1;
		bool done[512] = {false}, flag = true;
		for (int i = 0; i < p; i++){
			if (last > must[i].first.second){
				flag = false;
				break;
			}
			last = must[i].first.first;
		}
		if (!flag){
			cout << -1 << endl;
			continue;
		}

		ret = p;
		for (int i = 0; i < p; i++){
			done[must[i].second] = true;
		}

		last = -1;
		// cout << "*****" << endl;
		for (int i = 0; i < n; i++){
			int s = v[i].first.second, e = v[i].first.first, id = v[i].second;
			// cout << last << " " << name[id] << endl;
			if (done[id]){
				last = e;
				continue;
			}
			if (last > s) continue;
			ret++;
			last = e;
		}

		cout << ret << endl;
	}

	return 0;
}
