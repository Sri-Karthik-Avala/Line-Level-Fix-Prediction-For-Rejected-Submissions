#include <iostream>
using namespace std;
int main()
{
	int q;
	cin >> q;
	for(int i=0;i<q;i++)
	{
		int from,to;
		cin >> from >> to;
		if(from < to)
		{
			for(int j=from;j<to;j++)cout << j << ' ';
			cout << to << endl;
			continue;
		}
		if(from<=5)
		{
			for(int j=from;j>to;j--)cout << j <<  ' ';
			cout << to << endl;
		}
		else if(to>=5)
		{
			for(int j=from;j<=9;j++)cout << j << ' ';
			for(int j=5;j<to;j++)cout << j << ' ';
			cout << to << endl;
		}
		else
		{
			for(int j=from;j<=9;j++)cout << j << ' ';
			for(int j=5;j>to;j--)cout << j << ' ';
			cout << to << endl;
		}
	}
	return 0;
}
