int main(){
	ll n; cin >> n;
	vpl v(n);
	rep(i,n){
		cin >> v[i].first >> v[i].second;
		if(v[i].first > v[i].second) swap(v[i].first, v[i].second);
	}
	sort(all(v));
	ll mxa = 0, mxb = 0, mna = inf, mnb = inf;
	rep(i,n){
		chmax(mxa, v[i].second);
		chmin(mna, v[i].second);
		chmax(mxb, v[i].first);
		chmin(mnb, v[i].first);
	}
	ll ans = (mxa - mna) * (mxb - mnb);
	ll t = mxa - mnb;
	ll mx = mxb, mn = mnb;
	rep(i,n-1){
		chmax(mn, min(v[i+1].first, v[i].second));
		chmax(mx, v[i].second);
		chmin(ans, t * (mx - mn));
		if(v[i+1].first >= v[i].second) break;
	}
	cout << ans << "\n";
}
