#include<iostream>
using namespace	std;

int main() {
  int A, B, n;
  int x, y, z;

  while(1) {
    cin >> A >> B;
    if(A == 0 && B == 0) break;

    n = B - A;
    if(n / 1000 > 0) {
      x = n / 1000;
      n = n % 1000;
    }   
    else
      x = 0;
    if(n / 500 > 0) {
      y = n / 500;
      n = n % 500;
    }
    else
      y = 0;
    if(n / 100 > 0) {
      z = n / 100;
      n = n % 100;
    }
    else
      z = 0;

    cout << x << ' ' << y << ' ' << z << endl;
  }

  return 0;
}
