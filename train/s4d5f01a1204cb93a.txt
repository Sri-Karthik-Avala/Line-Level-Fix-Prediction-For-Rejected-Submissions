#include <iostream>
#include <algorithm>
using namespace std;

int n;
int x[30];

void A(int t, int p){
	if(t == 0 || p <= x[t-1]){
		x[t] = p;
		for(int i = 0; i <= t; i++){
			if(i != 0) cout << " " << x[i];
			else cout << x[i];
		}
		cout << endl;
	}
	for(int i = p-1; i >= 1; i--){
		x[t] = i;
		if(t != 0) if(x[t-1] < x[t]) continue;
		A(t+1,p-i);
	}
}

int main(){
	cin >> n;
	A(0,n);
}
