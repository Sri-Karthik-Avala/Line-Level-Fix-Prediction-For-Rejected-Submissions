#include <bits/stdc++.h>
using namespace std;

int main() {
  int N;
  cin >> N;
  
  int count;
  int ans = 0;

  for (int i = 1; i <= N; i+=2) {
    count = 0;
    for (int j = 2; j <= i / 2.0; j++) {
      if (i % j == 0) {
        count++;
      }
    }
    if (count == 8) {
      ans++;
    }
  }
  
  cout << ans << endl;
}
