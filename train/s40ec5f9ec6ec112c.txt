template <typename Type>
struct KMP{
    Type S;
    Vec A;

    KMP(Type S) : S(S){
        A.resize(S.size()+1);
        A[0] = -1;
        ll j = -1;
        for(int i = 0; i < S.size(); ++i){
            while(j >= 0 && S[i] != S[j]) j = A[j];
            ++j;
            if(S[i + 1] == S[j]) A[i + 1] = A[j];
            else A[i + 1] = j;
        }
    }

    vector<ll> search(Type T, ll pos = 0){
        vector<ll> res;
        ll i = 0, j = pos;
        while(i < S.size() && j < T.size()){
            if(S[i] == T[j]) ++i, ++j;
            else if(!i) ++j;
            else i = A[i];
            if(i == S.size()){
                res.push_back(j - i);
                i = A[i];
            }
        }
        return res;
    }
};

int main(void){
    ll n;
    cin >> n;
    Vec a(n), b(n);
    rep(i, n) cin >> a[i];
    rep(i, n) cin >> b[i];

    Vec c(n), d(n);
    rep(i, n) c[i] = a[i] ^ a[(i + 1) % n];
    rep(i, n) d[i] = b[i] ^ b[(i + 1) % n];
    c.insert(c.end(), all(c));
    // co_vec(c);

    KMP<Vec> kmp(d);
    auto v = kmp.search(c);
    for(ll i : v){
        if (i >= n) break;
        co(i, a[i] ^ b[0]);
    }

    return 0;
}
