struct timer{
	time_t start;

};

typedef istringstream iss;

typedef vector<int> vi;

int N,W,w;
vi d[100010];

int main(){
	cin.tie(0);
	ios_base::sync_with_stdio(0);
	
	cin>>N>>W;
	w=W/2;
	vi ans;
	rep2(i,1,w+2){
		ans.pb(i+W-1);
	}
	rep2(i,w+1,W){
		for(int j=i;j<=N;j+=i){
			d[j].pb(i);
		}
	}
	int res=ans[sz(ans)-1];
	rep2(i,w+2,N-W+2){
		int in=i+W-1,out=i-1;
		int hi=min(i-1,W-1);
		rep(j,sz(d[in]))if(d[in][j]<=hi)res++;
		rep(j,sz(d[out]))if(d[out][j]<=hi)res--;
		if(out<W)res++;
		ans.pb(res);
	}
	rep(i,sz(ans))cout<<ans[i]<<(i==sz(ans)-1?"\n":" ");
}
