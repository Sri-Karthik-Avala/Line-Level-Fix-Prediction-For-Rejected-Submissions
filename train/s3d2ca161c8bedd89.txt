signed main(){
  int n,m;
  cin>>n>>m;
  using P = pair<int, int>;
  vector<P> vp;
  for(int i=0;i<n;i++){
    int v,w;
    cin>>v>>w;
    vp.emplace_back(v,w);
  }
  auto cmp=[](P a,P b){return a.first*b.second>b.first*a.second;};
  sort(vp.begin(),vp.end(),cmp);
  double ans=0;
  for(auto p:vp){
    int w=min(p.second,m);
    ans+=double(p.first)*double(w)/double(p.second);
    m-=w;
  }  
  cout<<ans<<endl;
  return 0;
}
