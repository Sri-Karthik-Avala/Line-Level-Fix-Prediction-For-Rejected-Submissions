void sievetoprime(ll n)
{
	mem(isprime,true);
	for(ll p=2;p*p<=n;p++)
	{
		if(isprime[p]==true)
		{
			for(ll i=p*p;i<=n;i+=p)
				isprime[i]=false;
		}
	}
}

void solve()
{
	ll n;
	cin>>n;
	ll a[n],b[n];
	ll dp1[n],dp2[n];
	rep(i,0,n)
	{
		cin>>a[i];
	}
	rep(i,0,n)
	{
		cin>>b[i];
		if(i==0)
		{
			dp1[0]=a[0];
			dp2[0]=b[0];
		}
		else
		{
			dp1[i]=dp1[i-1]+a[i];
			dp2[i]=dp2[i-1]+b[i];
		}
	}
	ll cnt=0,ans=0;
	for(ll i=0;i<n;i++)
	{
		if(i==0)
		{
			cnt=dp1[i]+dp2[n-1]-dp2[i-1];
		}
		else
		{
			cnt=dp1[i]+dp2[n-1]-dp2[i-1];
		}
		ans=max(ans,cnt);
	}
	cout<<ans<<"\n";
}
