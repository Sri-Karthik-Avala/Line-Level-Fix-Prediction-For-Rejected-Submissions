#include<iostream>
using namespace std;
int main(){
	int n,m;
	while(1){
	cin>>n>>m;
	if(n==0 && m==0){
		break;
	}
	int dat[1100],cnt=0,x,ans=0;
	for(int i=0;i<n;i++){
		cin>>dat[i];
	}
	for(int i=0;i<m;i++){
		cin>>x;
		cnt+=x;
		ans++;
		if(cnt>=n){
			break;
		}
		cnt+=dat[cnt];
		if(cnt>=n){
			break;
		}
	}
	cout<<ans<<endl;
	}
	return 0;


}
