#include <bits/stdc++.h>

#define mp make_pair
#define mt make_tuple
#define pb push_back
#define rep(i, n) for (int i = 0; i < (n); i++)

using namespace std;

typedef long long ll;
typedef unsigned long long ull;
typedef pair<int, int> pii;
typedef pair<long, long> pll;

const int INF = 1 << 29;
const double EPS = 1e-9;
const int MOD = 100000007;

const int dx[] = {1, 0, -1, 0}, dy[] = {0, -1, 0, 1};
const int MAX_HW = 1500;
int H, W;
int board[MAX_HW][MAX_HW];
int dp[MAX_HW][MAX_HW];
int main() {
  scanf("%d %d", &H, &W);
  for (int i = 1; i <= H; i++) {
    for (int j = 1; j <= W; j++) {
      scanf("%d", &board[i][j]);
    }
  }
  for (int j = 1; j <= W; j++) {
    for (int i = 1; i <= H; i++) {
      if (board[i][j] == 1) {
        dp[i][j] = 0;
      } else {
        dp[i][j] = dp[i - 1][j] + 1;
      }
    }
  }
  int res = 0;
  for (int i = 1; i <= H; i++) {
    stack<pii> st;
    for (int j = 1; j <= W; j++) {
      if (st.empty()) {
        st.push(mp(dp[i][j], j));
      } else if (st.top().first < dp[i][j]) {
        st.push(mp(dp[i][j], j));
      } else if (st.top().first > dp[i][j]){
        int target = i;
        while (!st.empty() && st.top().first >= dp[i][j]) {
          pii pre = st.top();
          st.pop();
          int area = pre.first * (j - pre.second);
          res = max(res, area);
          target = pre.second;
        }
        st.push(mp(dp[i][j], target));
      }
    }
  }
  cout << res << endl;
  return 0;
}
