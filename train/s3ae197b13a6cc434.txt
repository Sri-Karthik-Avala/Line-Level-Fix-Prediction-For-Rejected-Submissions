#include <bits/stdc++.h>

#define ll long long
#define llong long long
#define REP(i,n) for(auto i=0;i<n;++i)
#define REPI(itr,v) for(auto itr=v.begin();itr!=v.end();++itr)
#define FOR(i,a,b) for(auto i=a;i<b;++i)
#define ALL(v) v.begin(),v.end()
#define pb push_back

#define AT "kougekida"
#define DF "mamoru"
#define TAME "tameru"

using namespace std;

int main(){

  int k;
  cin >> k;

  vector<string> I(k), N(k);
  REP(i,k){
    cin >> I[i];
  }
  REP(i,k){
    cin >> N[i];
  }

  int at_1 = 0, at_2 = 0;
  REP(i,k){
    if(I[i] == TAME && N[i] == TAME){
      at_1 = min(5, at_1+1);
      at_2 = min(5, at_2+1);
    }else if(I[i] == AT && N[i] == AT){
      if(at_1 > at_2){
        cout << "Isono-kun\n";
        return 0;
      }else if(at_1 < at_2){
        cout << "Nakajima-kun\n";
        return 0;
      }else{
        at_1 = 0; at_2 = 0;
      }
    }else if(I[i] == AT && N[i] == TAME){
      cout << "Isono-kun\n";
      return 0;
    }else if(I[i] == TAME && N[i] == AT){
      cout << "Nakajima-kun\n";
      return 0;
    }else if(I[i] == DF && N[i] == AT){
      if(at_2 == 5){
        cout << "Nakajima-kun\n";
        return 0;
      }
      at_2 = 0;
    }else if(I[i] == AT && N[i] == DF){
      if(at_1 == 5){
        cout << "Isono-kun\n";
        return 0;
      }
      at_1 = 0;
    }else if(I[i] == DF && N[i] == TAME){
      at_2 = min(5, at_2 + 1);
    }else if(I[i] == TAME && N[i] == DF){
      at_1 = min(5, at_1+1);
    }
  }
  
  cout << "Hikiwake-kun\n";
  return 0;
}
