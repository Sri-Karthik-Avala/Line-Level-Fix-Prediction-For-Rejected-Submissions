#include<bits/stdc++.h>
using namespace std;

int N;
int L[50];

bool check(int x,int y){
  map<int,bool> visited;
  queue<int> Q;
  Q.push(0);
  visited[0]=true;
  while(!Q.empty()){
    int p=Q.front();
    Q.pop();
    if(p==N)return true;
    int sum=0;
    for(int to=p+1;to<=N;to++){
      if(p==0 && to==N)continue;
      sum+=L[to-1];
      if(sum<x || y<sum)continue;
      if( !visited[to] ){
        visited[to]=true;
        Q.push(to);
      }
    }
  }
  return false;
}

int main(){
  cin>>N;
  for(int i=0;i<N;i++){
    cin>>L[i];
  }
  for(int i=1;i<=1000;i++){
    for(int j=1;j<=25000;j++){
      if( check(j,i+j) ){
        cout<<i<<endl;
        return 0;
      }
    }
  }
  return 0;
}
