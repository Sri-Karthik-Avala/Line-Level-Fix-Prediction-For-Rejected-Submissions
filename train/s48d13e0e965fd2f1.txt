#define ALL(c) c.begin(),c.end()
#define RALL(c) c.rbegin(),c.rend()
#define REP(type,i,n) for(type i=0;i<(n);++i)
#define MP(a,b) make_pair((a),(b))
#define F_ first
#define S_ second

typedef long long int lli;

const int INF=100000000;
const long long int INF_=1000000000000000000;

int dy[4]={0,1,0,1},dx[4]={0,0,1,1};
int dy_[4]={0,1,1,0},dx_[4]={0,0,-1,-1};

bool IsSame(char c[][2],vector<vector<char> > &g,int y,int x){
	REP(int,i,4){
		int ny=y+dy[i],nx=x+dx[i];
		if(g[ny][nx]!=c[dy[i]][dx[i]]) return false;
	}
	return true;
}

int main(){
	int M,N;
	cin >> M >> N;
	vector<vector<char> > g(M+2,vector<char>(N+2,'.'));
	REP(int,i,M){
		REP(int,j,N){
			cin >> g[i+1][j+1];
		}
	}
	char c[2][2];
	REP(int,i,2) REP(int,j,2) cin >> c[i][j];
	int ans=0,sum=0;
	for(int y=1; y<M; ++y){
		for(int x=1; x<N; ++x){
			if(IsSame(c,g,y,x)) ++sum;
		}
	}
	ans=sum;
	string s="JOI";
	for(int y=1; y<=M; ++y){
		for(int x=1; x<=N; ++x){
			REP(int,i,3){
				char tmp=g[y][x];
				int c1=0,c2=0;
				REP(int,j,4){
					int ny=y+dy_[j],nx=x+dx_[j];
					if(IsSame(c,g,ny,nx)) ++c1;
				}
				g[y][x]=s[i];
				REP(int,j,4){
					int ny=y+dy_[j],nx=x+dx_[j];
					if(IsSame(c,g,ny,nx)) ++c2;
				}
				g[y][x]=tmp;
				ans=max(ans,sum-c1+c2);
			}
		}
	}
	cout << ans << endl;
	return 0;
}
