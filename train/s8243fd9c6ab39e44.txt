import sys

sys.setrecursionlimit(max(1000, 10**9))

n,m = list(map(int, input().split()))
from collections import defaultdict
ns = defaultdict(set)
for i in range(m):
    a,b = map(int, input().split())
    a -= 1
    b -= 1
    ns[a].add(b)
    ns[b].add(a)

cs = [None]*n

    cs[start] = 1
    v1 = 1
    v0 = 0
    while q:
        u = q.pop()
        c = cs[u]
        cc = int(not c)
        bip = True
        for v in ns[u]:
            if cs[v] is None:
                cs[v] = cc
                if cc==0:
                    v0 += 1
                else:
                    v1 += 1
                q.append(v)
            elif cs[v]==c:
                bip = False
    if bip:
        return True, v0, v1
    else:
        return False, v0+v1, 0
