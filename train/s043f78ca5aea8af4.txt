#include <string>
#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;
struct edge { int to, id; };
int n, vc, a[200009], b[200009], c[200009]; long long sd, ans[200009], s[200009], dist[200009]; vector<edge> g[200009];
void dfs(int pos, int pre) {
	c[pos] = 1;
	for (edge e : g[pos]) {
		if (e.to != pre) {
			dfs(e.to, pos);
			c[pos] += c[e.to];
			// e.to -> pos, increased nodes = c[e.to], decreased nodes = n - c[e.to]
			if (n - c[e.to] * 2 != 0) {
				ans[e.id] = (s[e.to] - s[pos]) / (n - c[e.to] * 2);
			}
		}
	}
}
void dfs2(int pos, int pre, int flag) {
	if (flag) vc++; sd += dist[pos];
	for (edge e : g[pos]) {
		if (e.to != pre) {
			dist[e.to] = dist[pos];
			if (s[e.to] != -1) dist[e.to] += ans[e.id];
			dfs2(e.to, pos, (flag | (ans[e.id] == -1 ? 1 : 0)));
		}
	}
}
int main() {
	cin >> n;
	for (int i = 0; i < n - 1; i++) {
		cin >> a[i] >> b[i]; a[i]--, b[i]--;
		g[a[i]].push_back(edge{ b[i], i });
		g[b[i]].push_back(edge{ a[i], i });
	}
	for (int i = 0; i < n; i++) cin >> s[i];
	fill(ans, ans + n - 1, -1);
	dfs(0, -1);
	int zp = -1;
	for (int i = 0; i < n - 1; i++) {
		if (ans[i] == -1) zp = i;
	}
	if (zp != -1) {
		dfs2(0, -1, 0);
		ans[zp] = (s[0] - sd) / vc;
	}
	for (int i = 0; i < n - 1; i++) cout << ans[i] << endl;
	return 0;
}
