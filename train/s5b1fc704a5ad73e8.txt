const int MOD = 1e9+7;

int main(){
  string L;
  cin >> L;
  vector<vector<int>> dp(L.size()+1, vector<int>(2));
  dp[0][0] = 1;
  for(int i=1; i<=L.size(); i++) {
    if(L[i-1] == '0') {
      dp[i][0] = (dp[i][0]+dp[i-1][0])%MOD;
      dp[i][1] = (dp[i][1]+dp[i-1][1]*3)%MOD;
    } else {
      dp[i][0] = (dp[i][0]+dp[i-1][0]*2)%MOD;
      dp[i][1] = (dp[i][1]+dp[i-1][0])%MOD;
      dp[i][1] = (dp[i][1]+dp[i-1][1]*3)%MOD;
    }
  }
  int ans = (dp[L.size()][0]+dp[L.size()][1])%MOD;
  cout << ans << endl;
  return 0;
}
