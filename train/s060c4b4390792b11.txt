#include <bits/stdc++.h>
#define rep(i, n) for(int i = 0; i < (n); i++)
using namespace std;
using ll = long long;
using pii = pair<int, int>;

int h, w, k;

int f(vector<vector<int>> g, int x, int y) {
	g[y][x] = 0;
	
	int ret = 0;
	for (int c = 1; ; c *= 2) {
		rep(j, w) {
			for (int i = h - 1; i > 0; i--) {
				if (g[i][j] == 0) {
					int t = i - 1;
					while (t > 0 && g[t][j] == 0) t--;
					if (g[t][j] != 0) swap(g[i][j], g[t][j]);
				}
			}
		}
		
		bool loop = false;
		rep(i, h) {
			int num = g[i][0];
			int cnt = 1;
			int idx = 0;
			for (int j = 1; j < w; j++) {
				if (g[i][j] == num) cnt++;
				else {
					if (num > 0 && cnt >= k) break;
					num = g[i][j];
					cnt = 1;
					idx = j;
				}
			}
			
			if (num > 0 && cnt >= k) {
				ret += num * cnt * c;
				rep(j, cnt) g[i][idx + j] = 0;
				loop = true;
			}
		}
		
		if (!loop) break;
	}
	
	return ret;
}

int main() {
	cin >> h >> w >> k;
	
	vector<string> s(h);
	rep(i, h) cin >> s[i];
	
	vector<vector<int>> g(h, vector<int>(w));
	rep(i, h) rep(j, w) g[i][j] = s[i][j] - '0';
	
	int ans = 0;
	rep(i, h) rep(j, w) ans = max(ans, f(g, j, i));
	
	cout << ans << endl;
	return 0;
}
