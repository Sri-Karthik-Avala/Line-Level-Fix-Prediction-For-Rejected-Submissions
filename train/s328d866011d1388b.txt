int N;
int A[5500];
int U[5500], V[5500];
vint G[5500];
pair<vint, int> dfs(int cur, int par) {
    vint x = {A[cur]};
    int cut = 0;
    for (int to:G[cur]) {
        if (to == par) { continue; }
        pair<vint, int> tmp = dfs(to, cur);
        vint nxt = tmp.first;
        int ccut = tmp.second;
        vint y(x.size() + nxt.size(), LLONG_MAX);
        if (ccut >= 0) { rep(i, x.size()) { y[i + ccut + 1] = x[i]; }}
        rep(i, x.size()) {
            rep(j, nxt.size()) {
                chmin(y[i + j], x[i] + nxt[j]);
                if (nxt[j] < 0) { chmin(y[i + j + 1], x[i]); }
            }
        }
        x = y;
        reps(i, x.size() - 1) { chmin(x[i], x[i - 1]); }
        int pl = -1;
        for (int i = nxt.size() - 1; i >= 0; i--) { if (nxt[i] < 0) { pl = i; }}
        assert(pl >= 0);
        if (ccut >= 0) { cut += min(ccut, pl + 1); }
        else { cut += pl + 1; }
    }
    if (A[cur] < 0) { cut = -1; }
    return {x, cut};
}

signed main() {
    cin >> N;
    rep(i, N) { cin >> A[i]; }
    rep(i, N - 1) {
        cin >> U[i] >> V[i];
        U[i]--, V[i]--;
        G[U[i]].push_back(V[i]), G[V[i]].push_back(U[i]);
    }
    pair<vint, int> res = dfs(0, -1);
    int ans = N;
    if (A[0] > 0) { ans = res.second; }
    rep(i, res.first.size()) { if (res.first[i] < 0) { chmin(ans, i); }}
    cout << ans << endl;
}
