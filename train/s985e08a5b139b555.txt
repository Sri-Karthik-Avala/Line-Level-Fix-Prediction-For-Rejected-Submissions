N = int(input())
As = list(map(int, input().split()))
Bs = list(map(int, input().split()))

cur = N
dp = list(range(cur + 1))
for i in range(cur + 1):
    if i >= As[0]:
        if dp[i] < dp[i - As[0]] + Bs[0]:
            dp[i] = dp[i - As[0]] + Bs[0]
    if i >= As[1]:
        if dp[i] < dp[i - As[1]] + Bs[1]:
            dp[i] = dp[i - As[1]] + Bs[1]
    if i >= As[2]:
        if dp[i] < dp[i - As[2]] + Bs[2]:
            dp[i] = dp[i - As[2]] + Bs[2]
