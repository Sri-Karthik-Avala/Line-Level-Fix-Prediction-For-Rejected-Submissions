while(1):
    e=int(raw_input())
    if e==0:
        break
    else:
        sum=99999999999999
        zmax=int(e**(1/3.0))
        for z in range(zmax+1):
            edash=e-z**3
            y=int(edash**(1/2.0))
            x=edash-y**2
            if x+y+z<sum:
                sum=x+y+z
        print sum
