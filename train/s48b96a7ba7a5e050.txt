#include <iostream>

typedef long long ll;
using namespace std;

ll dp[1001][1001];

ll solve(ll n, ll k, ll m) {
  dp[0][0] = 1;
  for(int i=0;i<=n;++i) {
    for(int j=0;j<=k;++j) {
      if (i<j) {
        dp[i][j] = dp[i][j-1] % m;
      }
      else {
        dp[i][j] = (dp[i][j-1] + dp[i-j][j]) % m;
      }
    }
  }
  return dp[n][k];
}

int main() {
  ll n,k,mod=1e9+7;
  cin>>n>>k;
  cout<<solve(n,k,mod)<<endl;
  return 0;
}
