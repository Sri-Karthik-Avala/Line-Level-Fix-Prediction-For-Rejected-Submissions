set<string> dfs(map<string,vs> &m,string &member,map<string,set<string> > &dp){
	if(EXIST(dp,member)){
		return dp[member];
	}
	if(!EXIST(m,member)){
		set<string> ret;
		ret.insert(member);
		return ret;
	}
	set<string> ret;
	REP(i,m[member].size()){
		set<string> s=dfs(m,m[member][i],dp);
		ret.insert(ALL(s));
	}
	return ret;
}

		REP(i,n){

				group+=s[pos++];

			m[group]=vs();

				string member;
				while(s[pos]!=','&&s[pos]!='.'){
					member+=s[pos++];
				}
				m[group].push_back(member);
			}while(s[pos++]!='.');
		}
		map<string,set<string> > dp;
