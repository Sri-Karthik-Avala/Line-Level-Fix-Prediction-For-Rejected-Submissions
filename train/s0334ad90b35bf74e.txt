using namespace std;
typedef long long ll;
typedef pair<int, int> P;

const ll MOD = 1E+09 + 7;
const ll INF = 1E18;
const int MAX_N = 1E+05;

int N;

int H[MAX_N + 1];

ll dp[MAX_N + 1], A[MAX_N + 1];

ll getmax(int i);
void setmax(int i, ll x);

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> N;

    for (int i = 1; i <= N; i++) {
        cin >> H[i];
    }
    for (int i = 1; i <= N; i++) {
        cin >> A[i];
    }

    for (int i = 1; i <= N; i++) {
        setmax(H[i], getmax(H[i] - 1) + A[i]);
    }

    cout << getmax(N) << "\n";

    return 0;
}

ll getmax(int i)
{
    ll s = 0;
    while (i > 0) {
        s = max(s, dp[i]);
        i -= i & -i;
    }
    return s;
}

void setmax(int i, ll x)
{
    while (i <= N) {
        dp[i] = max(dp[i], x);
        i += i & -i;
    }
}
