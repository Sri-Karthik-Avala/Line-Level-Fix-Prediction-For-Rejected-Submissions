int binpow(int x, int y){
    int res = 1;
    while(y){
        if (y % 2){
            res = res * x % mod;
        }
        x = x * x % mod;
        y /= 2;
    }
    return res;
}
int c(int k, int n){
    if (k > n) return 0;
    int res = 1;
    for (int i = n - k + 1; i <= n; i++) res = res * i;
    for (int i = 1; i <= k; i++) res = res / i;
    return res % mod;
}
int f(int x){
    if (x < 0 || x % 2){
        return 0;
    }
    x /= 2;
    int res = 1;
    for (int i = 1; i <= 15; i++){
        res = res * binpow(i, mod - 2) % mod;
    }
    for (int i = x + 1; i <= x + 15; i++){
        res = res * i % mod;
    }
    return res;
}
