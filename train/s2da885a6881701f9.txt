#include<iostream>
using namespace std;
int n, minx;
int x[100000];

void palind() {
	for (int i = 0; i < 100000; i++) {
		if (i < 10) { x[i] = 1; }
		else if (i >= 10 && i < 100 && i % 11 == 0) { x[i] = 1; }
		else if (i >= 100 && i < 1000 && i / 100 == i % 10) { x[i] = 1; }
		else if (i >= 1000 && i < 10000 && i / 1000 == i % 10 && i % 11 == 0) { x[i] = 1; }
		else if (i >= 10000 && i / 10000 == i % 10 && (i / 1000) % 10 == (i / 10) % 10) { x[i] = 1; }
	}
}

int main() {
	palind();
	cin >> n;
	for (int i = n; i >= 0; i++) {
		if (x[i] == 1) { minx = i; break; }
	}
	for (int i = n; i < 100000; i++) {
		if (x[i] == 1) {
			if (n - minx > i - n) { minx = i; }; break;
		}
	}
	cout << minx << endl;
	return 0;
}
