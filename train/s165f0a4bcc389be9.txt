#include <string>
#include <math.h>
#include <iostream>
 
using namespace std;
 
int main()
{
  string s;
  cin >> s;
  int n = s.size();
  int x = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'A' && s[i+1] == 'C')
      x = 1; break;
  }
  if (x == 1)
    cout << "Yes";
  else
    cout << "No";
}
