g=[[] for _ in xrange(100)]
i=-1
while 1:
    i+=1
    try:
        g[i]=map(int,raw_input().split(","))
    except:
        dp=[[0]*101 for _ in xrange(101)]
        dp[0][0]=7
        for j in xrange(1,i):
            for k in xrange(len(g[j])):
                if j<=i/2:
                    dp[j][k]=max(dp[j-1][k-1],dp[j-1][k])+g[j][k]
                else:
                    dp[j][k]=max(dp[j-1][k],dp[j-1][k+1])+g[j][k]
        print(dp[i-1][0])
        break
