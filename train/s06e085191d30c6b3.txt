h,w=map(int,input().split())
n=int(input())
lst=[]

for i in range(1,n+1):
  for j in range(i):
    lst.append(str(i))
    
ans=[[0]*w for i in range(h)]

for i in range(h):
  if i%2==0:
    for j in range(w):
      ans[i][j]=lst.pop()

  else:
    for j in range(w-1,-1,-1):
      ans[i][j]=lst.pop()

for i in ans:
  print(' '.join(i))
