#include<bits/stdc++.h>
using namespace std;

#define ll long long int
#define INF 1000000007



int main(){
	string s;
	cin >> s;
	int arr[256] = { 0 };
	int n = s.length();
	if( n == 0 ){
		cout << "Yes";
		return 0;
	}
	if( n == 1 ){
		cout << "No";
		return 0;
	}	
	int ans = 1;
	for ( int i = 0; i < n; i++ )
		arr[ (int)s[i] ]++;
	for ( int i = 0; i < n; i++ ){
		if( arr[i]%2 != 0 ){
			ans = 0;
			break;
		}
	}
	if( ans == 1 )
		cout << "Yes";
	else
		cout << "No";
	return 0;
}
