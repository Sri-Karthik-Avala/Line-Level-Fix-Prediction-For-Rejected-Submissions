#include <iostream>
#include <cstring>
using namespace std;
typedef long long ll;
ll dp[128][128][128];

int main(){
    int n;
    int const mod=1000000007;
    while(cin>>n && n){
        memset(dp,0,sizeof(dp));
        dp[0][0][0]=1;
        for(int k=0;k<n;k++){
            char c;cin>>c;
            if(c=='-'){
                for(int x=0;x<=k;x++){
                    for(int y=0;y<=k;y++){
                        dp[k+1][x][y]+=dp[k][x][y];
                    }
                }
            }

            if(c=='D'){
                for(int x=1;x<=k;x++){
                    for(int y=1;y<=k;y++){
                        dp[k+1][x-1][y-1]+=dp[k][x][y]*x*y;
                        dp[k+1][x-1][y-1]%=mod;
                    }
                }
                for(int x=0;x<=k;x++){
                    for(int y=0;y<=k;y++){
                        dp[k+1][x][y]+=dp[k][x][y]*y;
                        dp[k+1][x][y]%=mod;
                    }
                }
            }
            
            if(c=='U'){
                for(int x=0;x<=k;x++){
                    for(int y=0;y<=k;y++){
                        dp[k+1][x][y]+=dp[k][x][y]*x;
                        dp[k+1][x][y]%=mod;
                    }
                }
                for(int x=0;x<=k;x++){
                    for(int y=0;y<=k;y++){
                        dp[k+1][x+1][y+1]+=dp[k][x][y];
                        dp[k+1][x+1][y+1]%=mod;
                    }
                }            
            }
        }

        // for(int i=0;i<=n;i++){
        //     for(int j=0;j<=n;j++){
        //         cout<<dp[n][i][j]<<" ";
        //         if(j==n)cout<<endl;
        //     }
        // }
        cout<<dp[n][0][0]<<endl;
    }
}
