#include<bits/stdc++.h>
using namespace std;

const int INF = 1 << 30;

int main(){
  int D, N, a[100], e[100], r[100];
  while(cin >> D >> N, D){
    for(int i = 0; i < N; i++){
      cin >> a[i] >> e[i] >> r[i];
    }
    vector< vector< int > > dp(D + 1, vector< int >( 101, INF));
    dp[D][0] = 0;
    for(int i = D; i > 0; i--){
      for(int j = 100; j >= 0; j--){
        if(dp[i][j] == INF) continue;
        for(int k = 0; k < N; k++){
          if(j < r[k]) continue;
          dp[max( 0, i - a[k])][min( 100, j + e[k])] = min( dp[max( 0, i - a[k])][min( 100, j + e[k])], dp[i][j] + 1);
        }
      }
    }
    int ret = INF;
    for(int i = 0; i < 101; i++) ret = min( ret, dp[0][i]);
    if(ret == INF) cout << "NA" << endl;
    else cout << ret << endl;
  }
}
