def solve():
    N = int(input())
    S = [input() for _ in range(N)]
    a_count = 1 if S[0] == "A" else -1
    if N == 1 and a_count == -1:
        return "NO"
    for i in range(1, N):
        if a_count >= 0:
            a_count += 1 if S[i] == "A" else -1
        else:
            return "NO"
    return "YES"


if __name__ == "__main__":
    print(solve())
