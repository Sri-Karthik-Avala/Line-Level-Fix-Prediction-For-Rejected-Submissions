int solve() {
    int N; cin >> N;
    if(N == 0) return 0;

    long double pw; cin >> pw;
    vector<long double> vw(N), pf(N), vf(N), th(N);
    for(int i=0; i<N; i++) {
        cin >> vw[i] >> pf[i] >> vf[i] >> th[i];
    }

    auto calc_cost = [&](long double x) {
        long double cost_sum = x * pw;
        for(int i=0; i<N; i++) {
            long double f = (th[i] - x * vw[i]) / vf[i];
            cost_sum += pf[i] * max(f, (long double)0.0);
        }
        return cost_sum;
    };

    long double ub = 1e10, lb = 0;
    for(int i=0; i<100; i++) {
        long double l = (lb * 2 + ub) / 3;
        long double r = (lb + 2 * ub) / 3;
        if(calc_cost(l) < calc_cost(r)) {
            ub = r;
        }
        else {
            lb = l;
        }
    }

    printf("%.12Lf\n", calc_cost(ub));
    return 1;
}

signed main() {
    while(solve());
    return 0;
}
