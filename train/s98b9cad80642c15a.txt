class Prime():
    def __init__(self, n):
        self.M = m = int(math.sqrt(n)) + 10
        self.A = a = [True] * m
        a[0] = a[1] = False
        self.T = t = []
        for i in range(2, int(math.sqrt(m)) + 1):
            if not a[i]:
                continue
            t.append(i)
            for j in range(i*i,m,i):
                a[j] = False

    def is_prime(self, n):
        return self.A[n]

def main():
    rr = []
    pr = Prime(100000**2)
    tl = len(pr.T)

    while True:
        n,a,b = LI()
        if n == 0:
            break
        m = fractions.Fraction(a,b)
        sn = n ** 0.5
        r = 0
        rs = ''
        for ni in range(tl):
            i = pr.T[ni]
            if i > sn:
                break
            j = n // i
            while not pr.is_prime(j):
                j -= 1
            if i*j <= r or fractions.Fraction(i,j) < m:
                continue
            r = i*j
            rs = '{} {}'.format(i,j)

        rr.append(rs)

    return '\n'.join(map(str, rr))


print(main())
