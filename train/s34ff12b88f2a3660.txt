signed main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    int n, k;
    cin >> n >> k;
    int s = sqrt(n) + 1;
    for (int b = 2; b <= s; b++) {
        int sum = 0;
        int x = n;
        while (x) {
            sum += x % b;
            x /= b;
        }
        if (sum == k) {
            cout << b << "\n";
            return 0;
        }
    }
    int ans = INF;
    if (n == k) {
        ans = n + 1;
    }
    for (int d1 = 1; d1 <= s; d1++) {
        int d2 = k - d1;
        if (d2 < 0) break;
        if ((n - d2) % d1) continue;
        int b = (n - d2) / d1;
        if (d2 >= b) continue;
        chkmin(ans, b);
    }
    if (ans == INF) cout << "-1\n";
    else cout << ans << "\n";
}
