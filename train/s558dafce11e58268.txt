#include<bits/stdc++.h>

typedef long long ll;
#define put(n) cout<<(n)<<endl
#define FOR(i,num,N) for(int(i)=(num);(i)<(N);++(i))
#define RFOR(i,num,N) for(int (i)=(num);(i)>(N);--(i))
#define all(v) (v).begin() , (v).end()
#define rall(v) (v).rbegin() , (v).rend()

int main(){
    int W,H,C,MIN =1e9;
    cin>>W>>H>>C;
   
    for(int i=1;i<=min(W,H);++i){
        if((W*H)%(i*i) == 0){
            MIN = min((W*H)/(i*i),MIN);
        }
    }
    cout<<MIN*C<<endl;
}
