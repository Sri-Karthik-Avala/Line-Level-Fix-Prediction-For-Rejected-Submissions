def maximum(A,l,m,n):
    maxint=A[l][m]
    for i in range(m,n):
        s=sum(A[l][m:i+1])
        if i<>m and s<0:break
        if s<0:continue
        for j in range(l,n-1):
            t=sum(A[j+1][m:i+1])
            if s+t<=0:break
            s+=t
            maxint=max(maxint,s)
    return maxint

n=input()
A=[map(int,raw_input().split()) for i in range(n)]
maxint=A[0][0]
for i in range(n):
    for j in range(n):
        maxint=max(maximum(A,i,j,n),maxint)
print maxint
