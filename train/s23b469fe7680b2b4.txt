#include <iostream>
#include <cmath>
#include <string>

using namespace std;

int main(void){
	double n;
	while(1){
		string ans = "";
		cin >> n;
		if(n < 0){
			break;
		}
		for(int i = 8;i >= -4;i--){
			if(n >= pow((double)2,(double)i)){
				ans += '1';
				n -= pow((double)2,(double)i);
			}else{
				ans += '0';
			}
			if(i == 0){
				ans += '.';
			}
		}
		if(n == 0){
			cout << ans << endl;
		}
		else{
			cout << "NA" << endl;
		}
	}

	return 0;
}
