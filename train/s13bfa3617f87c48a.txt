int in(){int a;scanf("%d",&a);return a;}

constexpr int inf = 1e9;

vector<ll> sum[40 / 2 + 1];

int main()
{
    ll N = in(), K = in(), L = in(), R = in();

    vector<ll> A(N);

    rep(i, N) {
        A[i] = lin();
    }

    int sz1 = N / 2;
    rep(bit, 1 << sz1) {
        int cnt = 0;
        ll s = 0;

        rep(i, sz1) {
            if (bit >> i & 1) {
                cnt++;
                s += A[i];
            }
        }

        sum[cnt].PB(s);
    }

    rep(i, sz1 + 1) sort(all(sum[i]));

    ll ans = 0;

    int sz2 = N - N / 2;
    rep(bit, 1 << sz2) {
        int cnt = 0;
        ll s = 0;

        rep(i, sz2) {
            if (bit >> i & 1) {
                cnt++;
                s += A[i + sz1];
            }
        }

        if (K - cnt < 0 || K - cnt > sz1) continue;

        int pos1 = upper_bound(all(sum[K - cnt]), R - s) - begin(sum[K - cnt]) - 1;
        int pos2 = lower_bound(all(sum[K - cnt]), L - s) - begin(sum[K - cnt]);

        if (sum[K - cnt][pos1] < sum[K - cnt][pos2]) continue;

        ans += max(pos1 - pos2 + 1, 0);
    }

    printf("%lld\n", ans);
}
