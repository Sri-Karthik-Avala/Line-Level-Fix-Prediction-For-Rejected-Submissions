void speed() {

}



int main(void) {
	speed();
	string s; cin >> s;
	int k; cin >> k;
	vector<string> d;
	for (int i = 0; i < (int)s.length(); i++) {
		for (int j = 1; j <= k; j++) {
			if (i + j >= (int)s.length()) continue;
			d.emplace_back(s.substr(i, j));
		}
	}
	sort(d.begin(), d.end());
	d.erase(unique(d.begin(), d.end()), d.end());
	cout << d[k - 1] << endl;

	return 0;
}
