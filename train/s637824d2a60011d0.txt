s=input()
t=input()
S,T=len(s),len(t)
dp=[[0]*(T+1) for _ in range(S+1)]
for i in range(S):
    for j in range(T):
        if s[i]==t[j]:
            dp[i+1][j+1]=dp[i][j]+1
        else:
            dp[i+1][j+1]=max(dp[i][j+1],dp[i+1][j])

len=dp[S][T]
x,y=S-1,T-1
ans=[0]*len
while len>0:
    if s[x]==t[y]:
        ans[len-1]=s[x]
        x-=1
        y-=1
        len-=1
    else:
        if dp[x][y]==dp[x-1][y]: x-=1
        else: y-=1
print("".join(ans))
