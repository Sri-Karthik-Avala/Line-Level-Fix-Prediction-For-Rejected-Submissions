#define rep(i,n) for(int i=0; i<n; i++)

#define int long long

using namespace std;

int prime[200];
bool is_prime[201];

int sieve(int n) {
    int p = 0;
    for (int i=0; i<=n; i++) is_prime[i] = true;
    for (int i=2; i<=n; i++) {
        if (is_prime[i]) {
            prime[p++] = i;
            for (int j=2*i; j<=n; j+=i) is_prime[j] = false;
        }
    }
}

vector<int> primeFactorization(int n)
{
    vector<int> ret;
    int p = 0;
    while (n > 1) {
        while (n % prime[p] == 0) {
            ret.push_back(prime[p]);
            n /= prime[p];
        }
        p++;
    }
    return ret;
}

void printV(vector<int> vi)
{
    rep(i,vi.size()) {
        cout << vi[i] << " ";
    }
    cout << endl;
}

signed main()
{
#ifdef LOCAL
    ifstream in("input.txt");
    cin.rdbuf(in.rdbuf());
#endif

    int N;
    map<int, int> cnt;

    cin >> N;

    sieve(1000);

    if (N == 1) {
        cout << 1 << endl;
        return 0;
    }

    for (int i=2; i<=N; i++) {
        vector<int> factors = primeFactorization(i);
        rep(j,factors.size()) {
            cnt[factors[j]]++;
        }
    }

    int ans = 1;
    for (map<int, int>::iterator it = cnt.begin(); it != cnt.end(); it++) {
        ans *= (it->second + 1);
        ans %= 1000000007;
    }

    cout << ans << endl;
}
