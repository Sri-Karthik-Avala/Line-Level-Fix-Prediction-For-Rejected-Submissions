#include<bits/stdc++.h>
using namespace std;

long long n,a[20005],dp[20005],maks;
int main()
{
	ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
	cin>>n;
	for(int i=1;i<=n;i++)
	{
		cin>>a[i];
	}
	for(int i=1;i<=n;i++)
	{
		dp[a[i]]=dp[a[i]-1]+1;
		maks=max(maks,dp[a[i]]);
	}
	cout<<n-maks<<'\n';
}
