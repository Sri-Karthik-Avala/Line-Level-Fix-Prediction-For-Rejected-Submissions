#include <iostream>
using namespace std;

int main(){
    int H, W;
    cin >> H >> W;
    while ((H != 0) || (W != 0)){
        int c=0;
        for(int i = 0; i < H * W; i++){    
            if(c % 2 == 0) cout << "#";
            else cout << ".";
            if((i + 1) % W == 0){
                cout << endl;
                c++;
            }
            c++;
                
        }
        cout << endl;
        cin >> H >> W;
    }
    return 0;
}
