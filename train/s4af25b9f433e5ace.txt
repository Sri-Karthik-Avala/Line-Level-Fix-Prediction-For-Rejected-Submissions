#include <bits/stdc++.h>
using namespace std;
 
int main() {
  int N;
  cin >> N;
  string A, B, C;
  cin >> A >> B >> C;
  int ans = 0;
  for (int i = 0; i < N; i++) {
    if (A.at(i) == B.at(i) && B.at(i) != C.at(i)) {
      ans++;
    }
    else if (A.at(i) != B.at(i) && B.at(i) != C.at(i)) {
      ans += 2;
    }
  }
  cout << ans << endl;
}
