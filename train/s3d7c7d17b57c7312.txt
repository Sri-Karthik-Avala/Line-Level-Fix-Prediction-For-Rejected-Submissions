#include <string>
#include <iostream>
using namespace std;
int n, m, a[10], b[2], c[10], dp[10][100]; string s;
int r[10] = { 0, 2, 4, 6, 8, 1, 3, 5, 7, 9 };
int main() {
	cin >> n >> s >> m;
	for (int i = 0; i < m; i++) cin >> a[i];
	int v = 0;
	for (int i = 0; i < n; i++) {
		if (s[i] == '*') b[i & 1]++;
		else v += (((n - i - 1) & 1) ? r[s[i] - 48] : (s[i] - 48));
	}
	v = (n * 10 - v) % 10; dp[0][0] = 1;
	for (int i = 0; i < b[0] + b[1]; i++) {
		for (int j = 0; j < 100; j++) {
			for (int k = 0; k < m; k++) {
				if (i < b[0]) dp[i + 1][j] += (j >= a[k] ? dp[i][j - a[k]] : 0);
				else dp[i + 1][j] += (j >= r[a[k]] ? dp[i][j - r[a[k]]] : 0);
			}
		}
	}
	int ret = 0;
	for (int i = v; i < 100; i += 10) ret += dp[b[0] + b[1]][i];
	cout << ret << endl;
	return 0;
}
