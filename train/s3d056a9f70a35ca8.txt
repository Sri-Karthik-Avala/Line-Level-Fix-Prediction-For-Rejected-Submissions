#include <iostream>
#include <string>

using namespace std;

int main() {
    int n;  cin >> n;
    string s;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (i % 2 == 0 && s[i] == 'p')
            ans--;
        else if (i % 2 && s[i] == 'g')
            ans++;
    }

    cout << ans << endl;
    return 0;
}
