#include <bits/stdc++.h>
using namespace std;

int main(){
  int N;
  cin >> N;
  
  int a;
  
  vector<int> v;
  for(int i=0; i<3*N; i++){
    cin >> a;
    v.push_back(a);
  }
  
  sort(v.begin(),v.end());
  reverse(v.begin(), v.end());
  
  int sum =0;
  for(int i=1;i<2*N; i+=2){
    sum += v.at(i);
  }
  
  cout << sum << endl;
 
}
