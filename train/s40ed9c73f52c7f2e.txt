int solve(int bit, ll L, ll R){
    ll r = (R>>bit)<<bit, l = (L>>bit)<<bit;
    if((1LL<<bit) > r) return 0;
    if((1LL<<bit) < l) return 0;

    ll dp[bit+1][2][2]; REP(i,bit+1)REP(j,2)REP(k,2) dp[i][j][k] = 0;

    if((1LL<<bit)==r && (1LL<<bit)==l) dp[0][0][0] = 1;
    else if((1LL<<bit)==r) dp[0][0][1] = 1;
    else if((1LL<<bit)==l) dp[0][1][0] = 1;
    else dp[0][1][1] = 1;

    REP(i,bit){
        r = (R>>(bit-i-1)) & 1; l = (L>>(bit-i-1)) & 1;

        // from 0 0
        (dp[i+1][0][0] += dp[i][0][0]) %= MOD;
        if(r==1 && l==0){
            (dp[i+1][0][1] += dp[i][0][0]) %= MOD;
            (dp[i+1][1][0] += dp[i][0][0]) %= MOD;
        }

        // from 0 1
        if(r==1){
            (dp[i+1][0][1] += dp[i][0][1]*2) %= MOD;
            (dp[i+1][1][1] += dp[i][0][1]) %= MOD;
        }
        else{
            (dp[i+1][0][1] += dp[i][0][1]) %= MOD;
        }

        // from 1 0
        if(l==1){
            (dp[i+1][1][0] += dp[i][1][0]) %= MOD;
        }
        else{
            (dp[i+1][1][0] += dp[i][1][0]*2) %= MOD;
            (dp[i+1][1][1] += dp[i][1][0]) %= MOD;
        }

        // from 1 1
        (dp[i+1][1][1] += dp[i][1][1]*3) %= MOD;
    }

    int ret = 0; REP(i,2)REP(j,2) (ret += dp[bit][i][j]) %= MOD;
    return ret;
}
void Main(){
    ll L,R; cin >> L >> R;
    int ans = 0;
    REP(i,61) (ans += solve(i,L,R)) %= MOD;
    cout << ans << en;
    return;
}
