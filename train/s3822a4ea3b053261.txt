#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define all(x) (x).begin(),(x).end()
const int mod=998244353,MAX=6005,INF=1<<30;
ll cnt[MAX][MAX];
ll inv[MAX],fac[MAX],finv[MAX];

void make(){
    
    fac[0]=fac[1]=1;
    finv[0]=finv[1]=1;
    inv[1]=1;
    
    for(int i=2;i<MAX;i++){
        inv[i]=mod-inv[mod%i]*(mod/i)%mod;
        fac[i]=fac[i-1]*(ll)i%mod;
        finv[i]=finv[i-1]*inv[i]%mod;
    }
    
}

ll comb(ll a,ll b){
    if(a<b) return 0;
    return fac[a]*(finv[b]*finv[a-b]%mod)%mod;
}


int main(){
    
    std::ifstream in("text.txt");
    std::cin.rdbuf(in.rdbuf());
    cin.tie(0);
    ios::sync_with_stdio(false);
    
    make();
    int N,X;cin>>N>>X;
    ll sum=0;
    
    for(int i=0;i<X;i++){
        for(int j=0;j<=N;j++){
            if(i<j) continue;
            
            int rem=i-j;
            cnt[i][j]=comb(j,rem);
            
            sum+=cnt[i][j]*comb(N,j);
            sum%=mod;
        }
    }
    
    for(int i=X+1;i<=2*N;i+=2){
        if(i<4) continue;
        int rd=(i-(X-1))*2;
        if(i-rd<0){
            if(X%2==1){
                int c=(i+1)/2;
                sum+=comb(N,c);
                sum%=mod;
            }
        }else{
            for(int j=0;j+rd/2<=N;j++){
                sum+=cnt[i-rd][j]*comb(N,j+rd/2);
                sum%=mod;
            }
        }
    }
    
    cout<<sum<<endl;
}
