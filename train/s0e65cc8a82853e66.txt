#include <iostream>
#include <algorithm>

using namespace std;
typedef long long ll;

int main() {
	ll a, b, n;
	cin >> a >> b >> n;

	ll ans = (1<<30);
	for(ll i = 1; i <= n; i++) {
		for(ll j = 1; j <= n; j++) {
			if((b * i) % j == 0) {
				ll nA = (b * i) / j;
				ans = min( abs(a - nA), ans);
			}
		}
	}

	cout << ans << endl;

	return 0;
}
