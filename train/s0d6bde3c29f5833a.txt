const ll INF = 1LL<<60;
const double EPS = 1e-8;
const int MOD = 1000000007;


int n;

main(){
	scanf("%d", &n);
	vi A(n+1);
	vector< set<plp> > vs(n+1);
	REP(i, n) scanf("%lld", &A[i+1]);
	vs[0].insert(plp(-1000000, pii(0, 0)));
	REP(i, n+1){
		ll sum = 0;
//		cout << vs[i] << endl;
		for(int j=i+1;j<=n;j++){
			sum += A[j];
			auto it = vs[i].lower_bound(plp(sum, pii(0, 0)));
			if(it == vs[i].begin()) continue;
			-- it;
			int c = it->second.first + 1;
			auto jt = vs[j].upper_bound(plp(sum, pii(0, 0)));
			if(jt != vs[j].begin()){
				-- jt;
				if(jt->second.first >= c) continue;
				++ jt;
			}
			while(jt != vs[j].end()){
				if(jt->second.first <= c) jt = vs[j].erase(jt);
				else break;
			}
			vs[j].insert(plp(sum, pii(c, i)));
		}
	}
	auto it = vs[n].end();
	--it;
	int m = it->second.first;
	printf("%d\n", m);
	vi ans;
	RREP(i, m-1){
		ans.push_back(it->second.second);
		it = vs[it->second.second].lower_bound(plp(it->first, pii(0, 0)));
		-- it;
	}
	reverse(ALL(ans));
	REP(i, ans.size()) printf("%d%s", ans[i], i<ans.size()-1?" ":"");
	puts("");
	
	
	return 0;
}
