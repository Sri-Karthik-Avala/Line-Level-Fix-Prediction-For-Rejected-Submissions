import sys
input=sys.stdin.buffer.readline
inputs=sys.stdin.buffer.readlines
sys.setrecursionlimit(10**9)
n=int(input())
from collections import Counter as co
ds=co(list(map(int,input().split() ))+[0]).most_common()
if ds[0][1]>3:print(0)
else:
    nows=[]
    nexts=[]
    for k,v in ds:
        if v==1:nexts.append(k)
        else:
            nows.append(-k)
            nows.append(k)
    def f(k):
        res=24
        for i in range(len(k)-1):
            for j in range(i+1,len(k)):
                #x y  0 y-x 
                res=min(res,-abs(k[i]-k[j])%24 ,abs(k[i]-k[j])%24)
        return res
    from itertools import product as pr
    print(max(f(nows+[v*j  for v,j in zip(nexts,i)])for i in pr((1,-1),repeat=len(ds))))
