#include<iostream>
using namespace std;

int main()
{
    int n;
    cin>>n;
    int a[n];
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    for(int i=0;i<n;i++){
        for(int j=i+1;j<n;j++){
            if(a[j]>a[i]){
                int c;
                c=a[i];
                a[i]=a[j];
                a[j]=c;
            }
        }
        cout<<a[i];
        (i==n-1) ? cout<<'\n' : cout<<' ';
    }
}
