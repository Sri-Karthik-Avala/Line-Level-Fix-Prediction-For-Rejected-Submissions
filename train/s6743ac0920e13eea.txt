    int n;
    cin>>n;
    for(int i=1;i<=n;++i){
        cin>>a[i].f>>a[i].s;
    }
    sort(a+1,a+n+1,[](const pii&p1,const pii&p2){return p1.f+p1.s<p2.f+p2.s;});
    forn(i,n+1)forn(j,n+1)dp[i][j]=OO;
    dp[0][0]=0;
    for(int i=1;i<=n;++i){
        for(int j=0;j<=i;++j){
            dp[i][j]=dp[i-1][j];
            if(j&&a[i].f>=dp[i-1][j-1])
                upn(dp[i][j],dp[i-1][j-1]+a[i].s);
        }
    }

        if(dp[n][i]<OO){
