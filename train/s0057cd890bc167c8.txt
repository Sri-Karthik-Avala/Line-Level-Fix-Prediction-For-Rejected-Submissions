#include <iostream>
#include <complex>
#include <vector>
#include <cmath>
#define EPS 1.0e-10
using namespace std;

#define X real()
#define Y imag()
typedef complex<double> P;
 
vector<P> p;
int check(P a){
	int ret=0;
	for(int i=0;i<p.size();i++) if(abs(a-p[i]) < 1+EPS) ret ++;
	return ret;
}

int adjcircle(P a, P b){
	P dir = a-b, pos=(a+b)*0.5;
	dir = P(dir.Y, -dir.X);
	dir = dir*(1/abs(dir))*sqrt(1-abs(pos-a)*abs(pos-a));
	return max(check(pos+dir), check(pos-dir));
}

main(){
	int i,j,N;
	while(cin >> N, N){
		p = vector<P>(N);
		int ans = 0;
		for(i=0;i<N;i++) cin >> p[i].X >> p[i].Y;
		for(i=1;i<N;i++)
			for(j=0;j<i;j++)
				if(abs(p[i]-p[j])<=2.0) ans = max(ans, adjcircle(p[i], p[j]));
		cout << ans << endl;
	}
}
