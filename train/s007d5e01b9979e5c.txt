#include <bits/stdc++.h>
using namespace std;

vector<int> sieve(int n) {
  vector<int> p(n);
  for(int i=2; i<n; i++) p[i]=i;
  for(int i=2; i*i<n; i++) {
    if(p[i]) {
      for(int j=i*i; j<n; j+=i) p[j]=0;
    }
  }
  p.erase(remove(p.begin(),p.end(),0),p.end());
  return p;
}

int main() {
  vector<int> p=sieve(1000000);
  int a[2];
  while(cin >> a[0] >> a[1]&&a[0]) {
    int b[2]={0,0};
    for(int j=0; j<2; j++) {
      int k=1;
      for(int i=p.size()-1; i>=0; i--) {
        if(a[j]==p[i]) continue;
        if(a[j]%p[i]==0) {
          b[j]+=p[i]*k;
          k=-1;
        }
      }
    }
    if(b[0]>b[1]) cout << 'a' << endl;
    else cout << 'b' << endl;
  }
  return 0;
}
