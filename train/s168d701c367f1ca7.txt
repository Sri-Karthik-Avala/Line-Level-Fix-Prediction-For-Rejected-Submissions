#include <bits/stdc++.h>
using namespace std;
#define rep(i,n) for(int i=0; i<(n); i++)
using ll = long long;
using P = pair<int,int>;
const int INF = 1001001001;
int dp[1100][3000];//dp[iこまで鍵をつかう][状態]とできるか
int main()
{
    int n,m;
    cin >> n >> m;
    vector<P> keys(m);
    rep(i,m){ 
        int a,b;
        int res = 0;
        cin >> a >> b;
        rep(j,b){
            int c;
            cin >> c;
            res |= 1<<(c-1);
        }
        keys[i] = P(a,res);//こすと、あける箱の集合

    }
    
    rep(i,m+1)rep(j,1<<(n+1))dp[i][j] = INF;
    dp[0][0] = 0;
    rep(i,m)rep(j,1<<n){
        int ni = i+1;
        int nj = j | keys[i].second;
        dp[i+1][j] = min(dp[i+1][j],dp[i][j]);
        dp[ni][nj] = min(dp[ni][nj],dp[i][j]+keys[i].first);
    }
    int ans = INF;
    rep(i,m+1)ans = min(ans,dp[i][(1<<n)-1]);
    if(ans == INF)ans = -1;
    cout << ans << endl;


    return 0;
}
