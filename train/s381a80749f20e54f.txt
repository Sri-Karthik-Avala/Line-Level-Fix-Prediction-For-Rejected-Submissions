#include<iostream>
#include<algorithm>
#include<cstring>
using namespace std;


int main(){
	int W,Q;
	int a[100];
	while(cin>>W>>Q,W|Q){
		memset(a,-1,sizeof(a));
		for(int i=0;i<Q;i++){
			char c;
			cin>>c;
			if(c=='s'){
				int id,w;
				cin>>id>>w;
				int pos=-1;
				for(int j=0;j<W-w;j++){
					bool ok=true;
					for(int k=0;k<w;k++)
						if(a[j+k]!=-1)ok=false;
					if(ok){
						pos=j;
						break;
					}
				}
				if(pos==-1)cout<<"impossible"<<endl;
				else{
					cout<<pos<<endl;
					for(int j=0;j<w;j++)a[pos+j]=id;
				}
			}else{
				int id;
				cin>>id;
				for(int j=0;j<W;j++)
					if(a[j]==id)a[j]=-1;
			}
		}
		cout<<"END"<<endl;
	}
    return 0;
}
