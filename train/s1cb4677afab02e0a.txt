using namespace std;

void rot(char (*d)[8]){
  char t[7][8]={};
  for(int i=0;i<7;i++){
    for(int j=0;j<7;j++){
      t[j][6-i]=(d[i][j]=='-')?'|':
	(d[i][j]=='|')?'-':
	d[i][j];
    }
  }
  copy(t[0],t[7],d[0]);
}

void lr(char (*d)[8]){
  for(int i=0;i<7;i++){
    for(int j=0;j<3;j++){
      swap(d[i][j],d[i][6-j]);
    }
  }
}

void ud(char (*d)[8]){
  for(int i=0;i<7;i++){
    for(int j=0;j<3;j++){
      swap(d[j][i],d[6-j][i]);
    }
  }
}
