#include <bits/stdc++.h>
#define pb push_back
#define mp make_pair
#define int long long
#define inf 1000000007
#define Endl endl
#define LINF 100000000000000007LL
#define ll long long
using namespace std;
signed main(){
	int n;
	cin>>n;
	int ans = 0;
	vector<int> a(n);
	for(int i=0;i<n;i++){
		cin>>a[i];
	}
	for(int i=30;i>=0;i--){
		int cnt = 0;
		for(int j=0;j<n;j++){
			if( (a[j]>>i)%2 ) cnt++;
		}
		if( cnt % 2 == 1 ){
			bool fl = false;
			for(int j=0;j<n;j++){
				if( a[j] == (1<<i) && fl == false ){
					a[j]--;
					ans++;
					fl = true;
				}
			}
			if( fl == false ){
				cout<<-1<<Endl;
				return 0;
			}
		}
		for(int j=0;j<n;j++){
			a[j] = a[j] % (1<<i);
		}
	}
	cout<<ans<<endl;
	return 0;
}
