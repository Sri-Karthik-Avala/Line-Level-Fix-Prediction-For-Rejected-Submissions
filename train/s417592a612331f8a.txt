#include <iostream>
#include <vector>
#include <cstdio>
#include <string>
using namespace std;

int main(){
	
	int s[3][3];
	int f[3][3];
	int w[3][3];
	
	for(int i=0;i<3;i++){
		for(int j=0;j<3;j++)
			cin>>s[i][j];
		for(int j=0;j<3;j++)
			cin>>f[i][j];
		if(f[i][2]-s[i][2]<0){
			if(f[i][2]==0){
				f[i][0]--;
				f[i][1]+=59;
				f[i][2]+=60;
			}
			else{
				f[i][1]--;
				f[i][2]+=60;
			}
		}
			w[i][2]=f[i][2]-s[i][2];
		if(f[i][1]-s[i][1]<0){
			f[i][0]--;
			f[i][1]+=60;
	}
		w[i][1]=f[i][1]-s[i][1];
		w[i][0]=f[i][0]-s[i][0];
	}
	for(int i=0;i<3;i++){
			cout<<w[i][0];
		for(int j=1;j<3;j++){
			cout<<" "<<w[i][j];
		}
		cout<<endl;
	}
	
}
