#include <bits/stdc++.h>
typedef long long ll;
using namespace std;

int a[100005], b[100005];

int main()
{
	int n, p;
	scanf("%d", &n);
	for (int i = 1; i <= n; i++)
	{
		scanf("%d", &p);
		if (i&1)
			a[p]++;
		else
			b[p]++;
	}
	int mx = 0, mx_freq = 0;
	int mx_freq2 = 0;
	for (int i = 1; i < 1e5 + 1; i++)
	{
		if (a[i] >= mx_freq)
		{
			mx = i;
			mx_freq = a[i];
		}
	}
	for (int i = 1; i < 1e5 + 1; i++)
		if (i != mx)
			mx_freq2 = max(mx_freq2, a[i]);

	int m = 0, m_freq = 0;
	int m_freq2 = 0;
	for (int i = 1; i < 1e5 + 1; i++)
	{
		if (b[i] >= m_freq)
		{
			m = i;
			m_freq = b[i];
		}
	}
	for (int i = 1; i < 1e5 + 1; i++)
		if (i != m)
			m_freq2 = max(m_freq2, b[i]);


	cout<<m_freq2<<" "<<mx_freq2<<endl;
	if (mx != m)
		printf("%d", n - m_freq - mx_freq);
	else
		printf("%d", n - max(mx_freq + m_freq2, mx_freq2 + m_freq));
}
