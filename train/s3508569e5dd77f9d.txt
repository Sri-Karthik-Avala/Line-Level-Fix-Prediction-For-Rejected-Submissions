int f(const string &s){
	if(s[0]=='*')return -1;
	int res=0;
	rep(i,s.size()){
		if(s[i]!='/'){
			res=res*10+s[i]-'0';
		}
	}
	return res;
}

bool contain(const string &s,const string &t){
	rep(i,s.size()){
		if(s.compare(i,t.size(),t)==0)
			return true;
	}
	return false;
}
