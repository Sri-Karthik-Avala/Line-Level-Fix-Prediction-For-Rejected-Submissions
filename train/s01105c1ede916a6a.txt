#include<iostream>
#include<string>
using namespace std;

long long a[10], n, t, sum;
string S;

int main() {
	a[0] = 1;
	cin >> n >> t;
	cin >> S;
	for (int i = 1; i < 10; i++) {
		if (a[i - 1] * n > 1000000000) {
			a[i] = 2000000000;
		}
		else {
			a[i] = a[i - 1] * n;
		}
	}
	for (int i = 0; i <= S.size() / 4; i++) {
		sum += t*a[S[i * 4 + 2] - 48];
	}
	if (sum > 1000000000) { cout << "TLE" << endl; }
	else { cout << sum << endl; }
	return 0;
}
