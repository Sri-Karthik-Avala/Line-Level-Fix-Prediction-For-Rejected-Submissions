#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); ++i)
using namespace std;
using ll = long long;
using P = pair<int,int>;

//vector<int> to[100005], co[100005]; 

int main() {
    int n;
    cin >> n;
    vector<vector<int>> to(n), co(n);
    rep(i,n-1) {
      int a, b, w;
      cin >> a >> b >> w;
      a--; b--;
      cin >> a >> b >> w;
      to[a].push_back(b); co[a].push_back(w);
      to[b].push_back(a); co[b].push_back(w);
    }
    vector<int> ans(n, -1);
    queue<int> q;
    ans[0] = 0;
    q.push(0);
    while (!q.empty()) {
      int v = q.front(); q.pop();
      rep(i,to[v].size()) {
        int u = to[v][i];
        int w = co[v][i];
        if (ans[u] != -1) continue;
        ans[u] = (ans[v]+w)%2;
        q.push(u);
      }
    }
    rep(i,n) cout << ans[i] << endl;
    return 0;
}
