template<class Abel> class WeightedUnionFind{
    vector<ll> par; 
    vector<ll> r_size; 
    vector<ll> rank; 
    vector<Abel> diff_weight;
    public:
        WeightedUnionFind(ll N, Abel SUM_UNITY = 0){
            init(N, SUM_UNITY);
        }

        //Nの大きさで初期化
        void init(ll N, Abel SUM_UNITY = 0){
            par.resize(N); r_size.resize(N); rank.resize(N); diff_weight.resize(N);
            for(int i=0; i<N; ++i){ par[i] = i; r_size[i] = 1; rank[i] = 0; diff_weight[i] = SUM_UNITY; }
        }

        ll root(ll x){
            if(par[x] == x)return x;
            int r = root(par[x]);
            diff_weight[x] += diff_weight[par[x]];
            return par[x] = r;
        }

        Abel get_weight(int x){
            root(x);
            return diff_weight[x];
        }

        ll diff(int x, int y){
            return get_weight(y) - get_weight(x);
        }

        //xとyとが所属する集合を併呑する
        //なお，weight(y) - weight(x) = wとなるようにmerge
        void merge(ll x, ll y, Abel w){
            w += get_weight(x); w -= get_weight(y);
            ll rx = root(x); ll ry = root(y);
            if(rx == ry) return;
            if(rank[rx] < rank[ry]) swap(rx,ry), w = -w;
            if(rank[rx] == rank[ry]) ++rank[rx];
            r_size[rx] += r_size[ry];
            par[ry] = rx;
            diff_weight[ry] = w;
        }

        bool same(ll x, ll y){
            return root(x) == root(y);
        }

        //素集合の大きさを返す
        ll size(ll x){
            return r_size[root(x)];
        }
};

int main(){
    int N, M, l, r, d;
    cin >> N >> M;
    WeightedUnionFind<int> uf(N,0);
    bool ans = true;
    rep(_,M){
        cin >> l >> r >> d;
        --l; --r;
        if(uf.same(l,r) && abs(uf.diff(l,r)) != d)ans = false;
        uf.merge(l,r,d);
    }
    rep(i,N)cout << uf.get_weight(i) << endl;
    cout << (ans ? "Yes" : "No") << endl;
    return 0;
}
