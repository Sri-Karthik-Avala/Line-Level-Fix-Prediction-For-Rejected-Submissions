#include<stdio.h>
#include<algorithm>
#include<queue>
using namespace std;
char str[20];
int ijk[1<<16][16];
int v[1<<16][16];
int ABS(int a){return max(a,-a);}
int main(){
	int a;
	while(scanf("%d",&a),a){
		scanf("%s",str);
		int A=0;
		for(int i=0;i<a;i++){
			if(str[i]=='B')A+=(1<<i);
		}
		scanf("%s",str);
		int B=0;
		for(int i=0;i<a;i++){
			if(str[i]=='B')B+=(1<<i);
		}
		for(int i=0;i<(1<<a);i++)for(int j=0;j<a;j++)
			ijk[i][j]=999999999;
		priority_queue<pair<int,pair<int,int> > >Q;
		for(int i=0;i<a;i++){
			Q.push(make_pair(0,make_pair(A,i)));
			ijk[A][i]=0;
		}
		while(Q.size()){
			int cost=-Q.top().first;
			int bit=Q.top().second.first;
			int at=Q.top().second.second;
			Q.pop();
			if(v[bit][at])continue;
			v[bit][at]=1;
			for(int i=0;i<a;i++){
				for(int j=0;j<a;j++){
					if((bit&(1<<i))&&!(bit&(1<<j))){
						int to=bit-(1<<i)+(1<<j);
						if(!v[to][j]&&ijk[to][j]>cost+ABS(at-i)+ABS(i-j)){
							ijk[to][j]=cost+ABS(at-i)+ABS(i-j);
							Q.push(make_pair(-ijk[to][j],make_pair(to,j)));
						}
					}else if(!(bit&(1<<i))&&(bit&(1<<j))){
						int to=bit-(1<<j)+(1<<i);
						if(!v[to][j]&&ijk[to][j]>cost+ABS(at-i)+ABS(i-j)){
							ijk[to][j]=cost+ABS(at-i)+ABS(i-j);
							Q.push(make_pair(-ijk[to][j],make_pair(to,j)));
						}
					}
				}
			}
		}
		int ret=9999999;
		for(int i=0;i<a;i++)ret=min(ret,ijk[B][i]);
		printf("%d\n",ret);
	}
}
