ll low_bs(int l,int r,ll val,vector<ll>v)
{
    ll ans=1e17;
    while(l<=r)
    {
        int mid=(l+r)>>1;
        if(v[mid]<=val)
        {
            ans=v[mid];
            l=mid+1;
        }
        else
            r=mid-1;
    }
    return ans;
}
ll high_bs(int l,int r,ll val,vector<ll>v)
{
    ll ans=1e17;
    while(l<=r)
    {
        int mid=(l+r)>>1;
        if(v[mid]>=val)
        {
            ans=v[mid];
            r=mid-1;
        }
        else
            l=mid+1;
    }
    return ans;
}
