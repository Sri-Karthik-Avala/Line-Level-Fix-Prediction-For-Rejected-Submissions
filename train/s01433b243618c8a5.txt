int M, N;
int d[256][256];
int op[256];

int main(){
  while(scanf("%d", &M) && M){
    scanf("%d", &N);
    memset(d, 0, sizeof(d));
    memset(op, 0, sizeof(op));
    REP(i, 1, N+1) scanf("%d", op+i);

    rep(i, N+1) REP(j, 1, M+1){
      int next = i + j;
      if(op[next] == 0){
	d[i][next] = 1;
      }else{
	d[i][max(0, min(N+1, next+op[next]))] = 1;
      }
    }

    rep(k, N+2) rep(i, N+2) rep(j, N+2){
      if(d[i][k] && d[k][j]) d[i][j] = 1;
    }

    /*    rep(i, N+2){
      rep(j, N+2) printf("%d ", d[i][j]); puts("");
      }*/

    int p;
    for(p = 0; p < N+1; p++) if(d[p][N+1] == 0){ puts("NG"); break;}
    if(p == N+1) puts("OK");
  }
  return 0;
}
