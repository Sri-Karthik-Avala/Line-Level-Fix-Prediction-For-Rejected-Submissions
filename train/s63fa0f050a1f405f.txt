#include <bits/stdc++.h>
using namespace std;


int main(){
  int n,a[10];
  vector<int> ans;
  cin>>n;
  for(int i=0;i<n;i++)cin>>a[i];
  sort(a,a+n);
  for(int i=1;i<=a[0]/i+1;i++){
    if(a[0]%i==0){
      bool f1=1,f2=1;
      for(int j=1;j<n;j++){
        if(a[j]%i) f1=0;
        if(a[j]%(a[0]/i)) f2=0;
      }
      if(f1) ans.push_back(i);
      if(f2) ans.push_back(a[0]/i);
    }
  }
  sort(ans.begin(),ans.end());
  for(int i=0;i<(int)ans.size();i++)cout<<ans[i]<<endl;
  return 0;
}
