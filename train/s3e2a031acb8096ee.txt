ll lcm(ll a,ll b){return a/gcd(a,b)*b;}

double d[100][10000];
int main(){
	int n,m,k;
	while(1){
		cin >> n >> m >> k;
		if(n==0&&m==0&k==0) break;
		REP(i,100) REP(j,10000) d[i][j] = 0;
		d[0][0] = 1.0;
		REP(i,n){
			REP(j,10000){
				if(d[i][j]<0.00001) continue;
				FOR(p,1,m+1){
					d[i+1][j+p] += d[i][j]*(1.0/double(m));
				}
			}
		}
		double res = 0;
		REP(i,10000){
			 // if(d[n][i]!=0) cout << i << "  " << d[n][i] << endl;
			int num = max(i-k,1);
			res += d[n][i]*num;
		}
		cout << shosu(8) << res << endl;
	}
}
