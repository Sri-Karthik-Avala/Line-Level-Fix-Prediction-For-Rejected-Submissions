#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

const int MAX_N = 1121;
const int MAX_K = 15;
const int MAX_P = 188;

bool isP[MAX_N];
vector<int> P; // size:187

int T[MAX_K][MAX_P][MAX_N];

void make() {
  fill(isP, isP+MAX_N, true);
  isP[0] = isP[1] = false;
  for(int i = 2; i < MAX_N; ++i) {
    if(isP[i]) {
      P.push_back(i);
      for(int j = i*2; j < MAX_N; j += i) isP[j] = false;
    }
  }
}

int rec(int k, int p, int n) {
  int &res = T[k][p][n];
  if(res == -1) {
    res = 0;
    for(int i = p; i < MAX_P && n-P[i] >= 0; ++i) {
      res += rec(k-1, i+1, n-P[i]);
    }
  }
  return res;
}

int main() {
  make();
  for(int i = 0; i < MAX_K; ++i) {
    for(int j = 0; j < MAX_P; ++j) {
      for(int k = 0; k < MAX_N; ++k) {
	if(i == 0 && k == 0) T[i][j][k] = 1;
	else if(i == 0 || k == 0) T[i][j][k] = 0;
	else T[i][j][k] = -1;
      }
    }
  }
  int n, k;
  while((cin >> n >> k) && (n || k)) {
    cout << rec(k, 0, n) << endl;
  }
  return 0;
}
