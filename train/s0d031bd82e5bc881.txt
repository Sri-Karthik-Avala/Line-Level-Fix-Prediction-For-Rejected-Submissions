#include <iostream>
#include <vector>

using namespace std;
  
int sum[20][100000]; // sum[c][i]: ???c???i???????????§??????????????????
int count[20]; // count[c]: ???c??????????????????
int dp[1 << 20]; // dp[i]: i????????£???????????????????????´????????????????????????????????????????????¨???????§£
const int INF = 1 << 29;

int main() {
    int N, M;
    cin >> N >> M;
    vector<int> C(N);

    for(int i = 0; i < N; i++) {
        cin >> C[i];
        C[i]--;
        count[C[i]]++;
    }

    // ?´???????????¨????
    // O(NM)
    for(int c = 0; c < M; c++) {
        for(int i = 0; i < N; i++) {
            if (C[i] == c) sum[c][i] = 1;
        }
    }
    for(int c = 0; c < M; c++) {
        int s = 0;
        for(int i = 0; i < N; i++) {
            s += sum[c][i];
            sum[c][i] = s;
        }
    }

    // BitDP?????????
    for(int i = 0; i < 1 << M; i++) {
        dp[i] = INF;
    }
    dp[0] = 0;

    // BitDP
    // ?????¨?????£???????????????????????¨?´????????????????????????£???????????????????????????????????¬?????????????????°????±????????????????
    // O(2^M M)
    for(int i = 0; i < 1 << M; i++) {
        int s = 0;
        for(int j = 0; j < M; j++) {
            if (i >> j & 1) {
                s += count[j];
            }
        }
        for(int j = 0; j < M; j++) {
            if (!(i >> j & 1)) {
                dp[i | (1 << j)] = min(dp[i | (1 << j)], dp[i] + count[j] - (sum[j][s + count[j] - 1] - sum[j][s >= 1 ? s - 1 : 0]));
            }
        }
    }

    cout << dp[(1 << M) - 1] << endl;
    return 0;
}
