#include "bits/stdc++.h"
#include<unordered_map>
#include<unordered_set>
#include<random>
#pragma warning(disable:4996)
using namespace std;
using ld=long double;
int main() {
	int T; ld E;cin>>T>>E;
	vector<int>cs(2*T+1);
	for (int i = 0; i < 2 * T + 1; ++i) {
		cin>>cs[i];
	}
	ld ans=0;
	ans=E/cs[T];
	int l=T-1,r=T+1;
	{
		ld sum=0;
		ld pre=1e18;
		for (int i = T + 1; i < 2 * T + 1; ++i) {
			ld now=sum+(2*T+1-i)*cs[i];
			if (now < pre) {
				r=i;
				pre=now;
			}
			sum+=cs[i];
		}
	}
	{
		ld sum = 0;
		ld pre = 1e18;
		for (int i = T - 1; i >=0; --i) {
			ld now = sum + (i+1)*cs[i];
			if (now < pre) {
				l = i;
				pre = now;
			}
			sum += cs[i];
		}
	}
	ld sum=0;
	assert(l<r);
	for (int i = l; i <= r; ++i) {
		if(i==l)sum+=(i+1)*cs[i];
		else if(i==r)sum+=(2*T+1-i)*cs[i];
		else sum+=cs[i];
	}
	ans=max(ans,E/sum*(T+1));
	cout<<setprecision(20)<<fixed<<ans<<endl;
	return 0;
}
