#include <iostream>
#include <cmath>
#include <vector>
#include <algorithm>
#include <string>
using namespace std;

#define loop(i,a,b) for(int i=(a); i<int(b); i++)
#define rep(i,b) loop(i,0,b)

int main(){
    int n;
    while(cin>>n && n){
        typedef pair<double,string> data;
        vector<data> v(n);
        rep(i,n){
            string l;
            int p,a,b,c,d,e,f,s,m;
            cin>>l>>p>>a>>b>>c>>d>>e>>f>>s>>m;
            double lifetime = a+(b+c+d+e)*m;
            double benefit = f*m*s-p;
            // cout << lifetime << " " << benefit << endl;
            v[i] = make_pair(benefit/lifetime,l);
        }
        sort(v.begin(), v.end(), [](const data& a, const data& b){
                if(abs(a.first-b.first)>1e-8) return a.first > b.first;
                return a.second < b.second;
            });
        rep(i,n){
            cout << v[i].second << endl;
        }
        cout << "#" << endl;
    }
}
