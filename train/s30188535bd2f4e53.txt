void solve() {
    ll n, a, b;
    cin >> n >> a >> b;
    if(a * b < n || a + b - 1 > n) {
        cout << -1 << endl;
        return;
    }
    ll rem = a * b - n;
    vll v;
    rep(i, b) v.push_back(b - i);
    for(int i = 1; i < a; ++i) {
        v.push_back(b * (i + 1));
        if(rem >= b - 1) {
            rem -= b - 1;
        }
        else if(rem > 0) {
            rep(j, rem) v.push_back(b * (i + 1) - j - 1);
            rem = 0;
        }
        else {
            rep(j, b - 1) v.push_back(b * (i + 1) - j - 1);
        }
    }
    vll v2 = v;
    sort(v2.begin(), v2.end());
    for(ll i : v) {
        ll j = lower_bound(v2.begin(), v2.end(), i) - v2.begin() + 1;
        cout << j << ' ';
    }
    cout << endl;
    return;
}
