using namespace std;
using namespace atcoder;
typedef long long ll;

int main()
{
   int N, K;
   cin >> N >> K;
   mcf_graph<ll, ll> graph(2 * N + 2); // start 2N, end 2N+1
   ll m = 1000000009;
   for (int i = 0; i < N; i++)
   {
      graph.add_edge(2 * N, i, K, 0);
      graph.add_edge(N + i, 2 * N + 1, K, 0);
   }
   for (int i = 0; i < N; i++)
   {
      for (int j = 0; j < N; j++)
      {
         ll a;
         cin >> a;
         graph.add_edge(i, N + j, 1, m - a);
      }
   }
   mcf_graph<ll, ll> graph2 = graph;
   auto slope = graph2.slope(2 * N, 2 * N + 1, N * K);
   ll f = -1;
   ll score = -1;
   for (auto point : slope)
   {
      ll curScore = point.first * m - point.second;
      //cout << point.first << " " << point.second << " " << curScore << endl;
      if (score < curScore)
      {
         f = point.first;
         score = curScore;
      }
   }
   auto result = graph.flow(2 * N, 2 * N + 1, f);
   cout << score << endl;
   vector<string> ans(N, string(N, '.'));
   auto edges = graph.edges();
   for (auto edge : edges)
   {
      if (edge.from < N && edge.cap == edge.flow)
      {
         ans[edge.from][edge.to - N] = '#';
      }
   }
   for (int i = 0; i < N; i++)
   {
      cout << ans[i] << endl;
   }
   return 0;
}
