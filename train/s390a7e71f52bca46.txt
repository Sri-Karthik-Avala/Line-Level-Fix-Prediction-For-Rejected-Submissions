#include<bits/stdc++.h>
using namespace std;
const int mod = 1e9+7;
const int maxN = 3e5+10;
long long int fact[maxN]={0};
long long int power(long long int a,long long int b){
	if(a==1||b==0){
		return 1;
	}
	long long int ans = 1;
	while(b){
		if(b&1){
			ans = (ans%mod)*(a%mod)%mod;
		}
		b/=2;
		a=((a%mod)*(a%mod))%mod;
	}
	return ans;
}
void init(){
	fact[0]=fact[1]=1;
	for(int i=2;i<maxN;i++){
		fact[i]=(fact[i-1]%mod)*(i%mod)%mod;
	}
}
long long int modInverse(long long int n){
	return power(n,mod-2);
}
long long int nCr(long long int n,long long int r){
	return (((fact[n]%mod)*(modInverse(fact[n-r]%mod))%mod)%mod)*(modInverse(fact[r]%mod))%mod;
}
int main(){
	init();
	long long int h,w,a,b;
	cin>>h>>w>>a>>b;
	//let L[k] be the number of ways from (0,0) to (k,b-1)
	//R[k] be the number of ways from (k,b) to (h-1,w-1)
	//ans will be sigma L[k]*R[k]
	
	long long int  ans = 0;
	for(long long int k=0;k<=h-a-1;k++){
		ans = (ans%mod) + (nCr(k+b-1,b-1)%mod)*(nCr(h-1-k+w-b-1,w-b-1)%mod);
	}
	cout<<ans<<"\n";
}
