const int M = 1e9 + 7;
int bp(int a, int b) {
    if (b == 0) return 1;
    int x = bp(a, b/2);
    x = (x*x) %M;
    if (b % 2) x = x * a % M;
    return x;
}
void solve() {
    int n, k;
    cin >> n >> k;
    vector<vi> dp(n + 1, vi(1e4, 0));
    rep(i, 1, n + 1) rep(j, 0, dp[i].size()) {
        rep(m, 0, k + 1) {
            int incr = (m > i?0:(m + j)/i);
            if (j + incr < dp[i].size()) {
                dp[i][j] += dp[i - 1][j + incr] + (m > i?m + j:(m + j) % i) * bp(k + 1, i - 1);
                dp[i][j] %= M;
            }
        }
    }
    put(dp[n][0]);
}

    ios_base::sync_with_stdio(false);

    while (q--) solve();
