#include <stdio.h>
#include <algorithm>
using namespace std;
const int N=1e3+10;
int main(){
    int n,q,l,r,a[N];
    scanf("%d",&n);
    for(int i=0;i<n;i++)scanf("%d",&a[i]);
    scanf("%d",&q);
    while(q--){
        scanf("%d%d",&l,&r);
        reverse(a+l,a+r);
    }
    printf("%d",a[0]);
    for(int i=1;i<n;i++)printf(" %d",&a[i]);
    printf("\n");
    
}
