#include<iostream>
using namespace std;

int main(){
	long A,B;
	cin>>A>>B;
	long num;
	long ans;
	if(A%2==0){
		if(B%2==0){
			num=B-A;
			if(num%4==0){
				ans=0;
			}
			else{
				ans=1;
			}
			ans=ans^B;
		}
		else{
			num=B-A-1;
			if(num%4==0){
				ans=0;
			}
			else{
				ans=1;
			}
		}
	}
	else{
		if(B%2==0){
			num=B-A-1;
			if(num%4==0){
				ans=0;
			}
			else{
				ans=1;
			}
			ans=ans^A;
			ans=ans^B;
		}
		else{
			num=B-A;
			if(num%4==0){
				ans=0;
			}
			else{
				ans=1;
			}
			ans=ans^A;
		}
	}
	cout<<ans<<endl;
}
