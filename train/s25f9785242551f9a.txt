#include<stdio.h>
int main(void)
{
	int a,b,s,i,max,f,flg;
	flg=0;
	max=-1;
	while(1){
	for(i=0;i<5;i++){
		scanf("%d %d",&a,&b);
		s=a+b;
		if(a==0 && b==0){
			flg=1;
			break;
		}
		if(max<s){
			max=s;
			f=i;
		}
	}
		if(flg==1){
			break;
		}
	if(f==0){
		printf("A %d\n",max);
	}
	if(f==1){
		printf("B %d\n",max);
	}
	if(f==2){
		printf("C %d\n",max);
	}
	if(f==3){
		printf("D %d\n",max);
	}
	if(f==4){
		printf("E %d\n",max);
	}
	}
	return 0;
}
