#include<iostream>
#include<algorithm>

using namespace std;

int main(){
  int M,N;
  cin>>M>>N;
  int a[100][100];
  for(int i=0;i<M;i++){
    for(int j=0;j<N;j++){
      cin>>a[i][j];
    }
  }
  int ans=0;
  for(int i=0;i<M;i++){
    int l=0;
    for(int j=0;j<N;j++){
      l+=a[i][j]^!!i^(j!=0&&j!=N-1);
    }
    ans+=((N>1)?min(N,l+1):l);
  }
  cout<<ans<<endl;
}
