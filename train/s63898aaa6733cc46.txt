using tp = tuple<int, int, int>;

int n;
tp a[1000];
int dp[1000][366][366];

int calc(int i, int j, int k)
{
	if (i == n) return 0;
	int &res = dp[i][j][k];
	if (~res) return res;
	
	int x, y, z;
	tie(x, y, z) = a[i];

	chmax(res, calc(i + 1, j, k));
	if (x >= j) chmax(res, z + calc(i + 1, y, k));
	if (x >= k) chmax(res, z + calc(i + 1, j, y));

	return res;
}

int main()
{
	while (cin >> n, n){
		rep(i, n){
			int x, y, z;
			cin >> x >> y >> z;
			a[i] = tie(--x, y, z);
		}
		sort(a, a + n);

		memset(dp, -1, sizeof(dp));
		cout << calc(0, 0, 0) << endl;
	}
}
