#include <bits/stdc++.h>
#include <string>
using namespace std;
typedef long long ll;

int main(){
    ll n;
    cin>>n;
    ll sum=0;
    vector<ll> a(n);
    for(ll i=0;i<n;++i){
        cin>>a[i];
        sum+=a[i];
    }
    ll t = (sum>=0?(sum+n-1):sum)/n;
    ll ans=0;
    for(ll i=0;i<n;++i){
        ans+=(t-a[i])*(t-a[i]);
    }
    cout<<ans;
}
