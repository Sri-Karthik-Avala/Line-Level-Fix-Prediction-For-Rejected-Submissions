const ll inf=1e9+7;

ll rui(ll a,ll b){
    ll res=1;
    ll x=a;
    while(b){
        if(b&1){
            res=res*x%inf;
        }
        x=x*x%inf;
        b/=2;
    }
    return res;
}
signed main(){
    ll n,x;cin>>n>>x;
    vector<ll>s(n);
    rep(i,n)cin>>s[i];
    vector<vector<ll> >dp(n+1,vector<ll>(x+1));
    sort(rall(s));
    ll p=1;
    REP(i,1,n+1){
        p=p*i%inf;
    }
    dp[0][x]=p;
    rep(i,n){
        rep(j,x+1){
            ll b=dp[i][j]*rui(n-i,inf-2)%inf;
            if(i<n-1)dp[i+1][j]=(dp[i+1][j]+b*(n-i-1)%inf)%inf;
            dp[i+1][j%s[i]]=(dp[i+1][j%s[i]]+b)%inf;
            //cout<<dp[i][j]<<' ';
        }
        //cout<<endl;
    }
    ll ans=0;
    rep(i,min(s[n-1],x+1)){
        ans=(ans+i*dp[n][i]%inf)%inf;

    }
