#include<iostream>
#include<string>
#include<algorithm>
using namespace std;
int main(){

  int N,A,B,C,D;
  cin>>N>>A>>B>>C>>D;
  string S;
  cin>>S;
  bool flag1=0;
  for(int i=A-1;i<=max(C,D)-2;i++){
  
    if(S[i]=='#'&&S[i+1]=='#'){
    
      cout<<"No";
      return 0;
    }
  }
  if(C>D){
  
    for(int i=B-2;i<=N-2;i++){
    
      if(S[i]=='.'&&S[i+1]=='.'&&S[i+2]=='.')flag1=1;
    }
    if(flag1==0)cout<<"No";
    else cout<<"Yes";
  }
  else cout<<"Yes";
  return 0;
}
