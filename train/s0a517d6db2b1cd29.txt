#include <bits/stdc++.h>

using namespace std;

const int M = 262144;

int A[M], tr[M << 1];
int n, q;

void update(int p, int x) {
    tr[p | M] = x;
    for (p = (p | M) >> 1; p != 0; p >>= 1)
        tr[p] = max(tr[p << 1], tr[p << 1 | 1]);
}

int query(int l, int r) {
    int res = -1;
    for (l += M - 1, r += M + 1; (l ^ r) != 1; l >>= 1, r >>= 1) {
        if ((~l & 1) == 1)
            res = max(res, tr[l ^ 1]);
        if ((r & 1) == 1)
            res = max(res, tr[r ^ 1]);
    }
    return res;
}

int main() {
    scanf("%d%d", &n, &q);
    for (int i = 1; i <= n; ++i) {
        scanf("%d", &A[i]);
        tr[i | M] = A[i];
    }
    for (int i = M - 1; i != 0; --i)
        tr[i] = max(tr[i << 1], tr[i << 1 | 1]);
    while (q-- != 0) {
        int t, x, y;
        scanf("%d%d%d", &t, &x, &y);
        if (t == 1)
            update(x, y);
        if (t == 2)
            printf("%d\n", query(x, y));
        if (t == 3) {
            int l = x, r = n, res = n + 1;
            while (l <= r) {
                int mid = (l + r) >> 1;
                if (y <= query(mid, r)) {
                    res = mid;
                    r = mid - 1;
                } else
                    l = mid + 1;
            }
            printf("%d\n", res);
        }
    }
    return 0;
}
