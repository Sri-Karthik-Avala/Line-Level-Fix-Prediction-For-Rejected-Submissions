#include <iostream>
#include <vector>
#include <algorithm>


using std::cin;

int main()
{
    int from_num;
    cin >> from_num;
    std::vector<int> from(from_num);
    for (auto i = 0; i < from_num; i++) {
        cin >> from.at(i);
    }

    int target_num;
    cin >> target_num;
    std::vector<int> target(target_num);
    for (auto i = 0; i < target_num; i++) {
        cin >> target.at(i);
    }

    int cnt = 0;
    for (auto i = 0; i < target_num; i++) {
        if (std::count(from.begin(), from.end(), target.at(i)) > 0) {
            cnt++;
        }
    }

    std::cout << cnt << std::endl;

    return 0;
}
