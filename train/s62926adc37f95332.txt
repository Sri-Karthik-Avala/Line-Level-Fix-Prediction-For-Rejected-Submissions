#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int n;
    vector<int> time(n);

    while (cin >> n) {
        time.clear();
        for (int i = 0; i < n; i++) {
            int t; cin >> t;
            time.push_back(t);
        }
        sort(time.begin(), time.end());
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += time[i] * (n-i);
        }
        cout << sum << endl;
    }
    return 0;
}
