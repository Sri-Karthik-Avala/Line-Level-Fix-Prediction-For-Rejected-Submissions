import math
def sieve (n):
	prime = [0,0]
	prime += [1 for i in range(n-1)]
	ub = math.sqrt(n) + 1
	d = 2
	while d <= ub:
		if prime[d] == 0:
			d += 1
			continue
		prod = 2
		while d * prod <= n:
			prime[d*prod] = 0
			prod += 1
		d += 1
	return prime
prime = sieve(2*10**5)

while 1:
	N,P = map(int,raw_input().split())
	if (N,P) == (0,0):
		break
	prime_ub = P
	prime_list = []
	prime_num = 0
	inc = N + 1
	while prime_num < prime_ub :
		if prime[inc] == 1:
			prime_num += 1
			prime_list.append(inc)
		inc += 1
	sum_list = []
	for i in range(prime_num):
		sum_list += [prime_list[i] + p for p in prime_list[i:]]
	sum_list.sort()
	print sum_list[P-1]
