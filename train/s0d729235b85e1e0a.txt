#include<iostream>
#include<vector>
using namespace std;

struct mes{
    mes* pre;
    vector<mes*> fol;
    string m;
};
vector<mes>vec;

void dfs(mes m, int d){
    for(int i=0;i<d;i++)
        cout<<".";
    cout<<m.m<<endl;
    for(int i=0;i<m.fol.size();i++)
        dfs(*m.fol[i],d+1);
}
int main(){
    int n, k;
    string str;
    cin>>n;
    for(int i=0;i<=n;i++){
        mes me;
        vec.push_back(me);
    }
    for(int i=1;i<n;i++){
        cin>>k>>str;
        vec[i].pre = &vec[k];
        vec[i].m = str;
        vec[k].fol.push_back(&vec[i]);
    }
    dfs(vec[1],0);
}
