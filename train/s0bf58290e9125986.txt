while True:
    name,type = map(str, raw_input().split())
    ans = ""
    if (type == 'X'):
        break
    names = []
    if (name.find("_") != -1):
        names = name.split("_")
    elif (name[0].islower() == True):
        tmp = name[0]
        for i in range(1, len(name)):
            if (name[i].isupper() == True):
                names.append(tmp)
                tmp = name[i]
            else:
                tmp += name[i]
        if (len(tmp) > 1):
            names.append(tmp)
    else:
        tmp = name[0]
        for i in range(1, len(name)):
            if (name[i].isupper() == True):
                names.append(tmp)
                tmp = name[i]
            else:
                tmp += name[i]
        if (len(tmp) > 1):
            names.append(tmp)
    if (type == "U"):
        for x in names:
            ans += x[0].upper()
            ans += x[1:].lower()
    elif (type == "L"):
        cnt = 0
        for x in names:
            if (cnt == 0):
                ans += x.lower()
            else:
                ans += x[0].upper()
                ans += x[1:].lower()
            cnt += 1
    else:
        ans = "_".join(names)
        ans = ans.lower()
    print ans
