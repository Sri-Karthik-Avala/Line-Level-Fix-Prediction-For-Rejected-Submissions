int n,m,q;
ll b[1010];

inline void output(int arg){
	if(__builtin_popcount(b[arg])!=1)
		cout << '?';
	else{
		int bit;
		for(bit=0;bit<n;bit++)
			if(b[arg]&(1LL<<bit))
				break;
		if(bit>=10)
			cout << (char)(bit-10+'A');
		else
			cout << bit;
	}
}

int main(void){
	while(cin >> n >> m >> q){
		if(n==0)
			break;
		ll sw=0LL;
		rep(i,m) b[i]=((1LL<<n)-1LL);
		rep(i,q){
			ll nsw=0;
			string tmp;
			cin >> tmp;
			rep(j,n){
				if(tmp[j]=='1')
					nsw|=(1LL<<j);
			}
			sw^=nsw;
			cin >> tmp;
			rep(j,m){
				if(tmp[j]=='1')
					b[j]&=sw;
				else
					b[j]&=(~sw);
			}
		}
		rep(i,m) output(i);
		cout << endl;
	}

	return 0;
}
