#include <bits/stdc++.h>
using namespace std;

void solve() {
  int n, h, w; cin >> n >> h >> w;
  int ans = 0;
  while(n--) {
    int a, b; cin >> a >> b;
    ans += a <= h && b <= w;
  }
  cout << ans << endl;
}

signed main() {
  cin.tie(nullptr);
  ios::sync_with_stdio(false);
  cout << fixed << setprecision(10);
  solve();
  return 0;
}
