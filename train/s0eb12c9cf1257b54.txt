template<class Type>
Type solve(Type res = Type()){
    int N; cin >> N;
    vector<string> C(N);
    vector<int> D(N);
    for(int i = 0; i < N;i++) cin >> C[i] >> D[i];
    int M; cin >> M;
    vector<string> O(M);
    for(auto& in:O) cin >> in;
    
    
    map<string,vector<int>> mp;
    for(int i = 0; i < N;i++){
        mp[C[i]].push_back(D[i]);
    }
    for(auto& p:mp){
        sort(p.second.rbegin(),p.second.rend());
    }
    
    int pred = -1;
    for(int i = 0; i < M;i++){
        vector<int>& v = mp[O[i]];
        if(v.empty()) return "No";
        if(pred >= v.back()) return "No";
        pred = v.back();
        v.pop_back();
    }
    return "Yes";
    // return res;
}
