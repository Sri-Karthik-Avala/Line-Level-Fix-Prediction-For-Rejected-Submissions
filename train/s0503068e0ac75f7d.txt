#include <bits/stdc++.h>

using namespace std;
const int MAXN = 1e5+2;

int arr[MAXN];
int main()
{
    int N;
    cin >> N;

    for(int i = 2; i <= N; i++){
        int pos; cin >> pos;
        arr[pos]++;
    }
    for (int i = 1; i <= N; i++){
        cout << arr[i] << "\n";
    }
    return 0;
}
