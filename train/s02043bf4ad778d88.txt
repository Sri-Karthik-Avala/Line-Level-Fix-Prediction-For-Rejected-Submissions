#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;
typedef long long ll;
const ll INF = 1000000000000000000;
ll dp[301][301];

int main(){
	ios_base::sync_with_stdio(false);
	int n;
	cin >> n;
	vector<int> a(n);
	for(int i = 0; i < n; ++i){ cin >> a[i]; }
	vector<int> b(n + 1);
	for(int i = 0; i < n; ++i){ b[i + 1] = b[i] + a[i]; }
	for(int i = 0; i <= n; ++i){
		for(int j = 0; j <= n; ++j){ dp[i][j] = INF; }
	}
	for(int i = 1; i <= n; ++i){
		for(int j = 0; j + i <= n; ++j){
			const int head = j, tail = i + j;
			if(i == 1){ dp[head][tail] = 0; }
			for(int k = head + 1; k < tail; ++k){
				int x = b[k] - b[head], y = b[tail] - b[k], c = 0;
				ll cost = dp[head][k] + dp[k][tail];
				while(x > 0 || y > 0){
					const int z = x % 10, w = y % 10;
					cost += (x * w) + c;
					c = (z + w + c >= 10 ? 1 : 0);
					x /= 10; y /= 10;
				}
				cost += c;
				dp[head][tail] = min(dp[head][tail], cost);
			}
		}
	}
	cout << dp[0][n] << endl;
	return 0;
}
