def intread():

def mapline(t=int):
    return map(t, sysread().split())
def mapread(t=int):
    return map(t, read().split())



def run():
    H, W = mapline()
    ans = INF
    for i in range(2):
        H, W = W, H
        for h in range(1, H-1):
            _ans = 0
            A = h * W
            # pat1
            B1 = math.ceil(W/2) * (H-h)
            C1 = math.floor(W/2) * (H-h)
            _ans = max([abs(A - B1), abs(B1 - C1), abs(A - C1)])
            ans = min(_ans, ans)
            # pat2
            _ans = 0
            if H*W <= 3 * A:
                w = math.ceil(A // (H-h))
                B2 = (H-h) * w
                C2 = (H-h) * (W-w)
                _ans = max([abs(A - B2), abs(B2 - C2), abs(A - C2)])
                ans = min(_ans, ans)
            # pat3
            _ans = 0
            h2 = math.ceil((H-h) / 2)
            if h + h2 != H:
                C3 = W * (H - h2 - h)
                B3 = W * h2
                _ans = max([abs(A - B3), abs(B3 - C3), abs(A - C3)])
                ans = min(_ans, ans)

    print(ans)
