	int walk = 0, running = 0;

	FOR(i, 1, l + 1)
	{
		if (!d.empty() && d.front() == i)
		{
			d.pop();
			if (run == 0) run += t*v;
			else

				if (car < k) car++;
				else run = t*v;
			}
		}

		if (run == 0 && car>0)

			car--;
			run = t*v;
		}

		{
			running++;
			run--;
		}
		else walk++;
