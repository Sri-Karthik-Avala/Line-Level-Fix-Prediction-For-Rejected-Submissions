#include<iostream>
#include<cstring>
#define mod 998244353
using namespace std;
int n, i, j, ii, k, t, nr, m, sol, jj, x;
char s[305];
int d[305][305][305];
void adauga(int &x, int y){
    x += y;
    if(x >= mod){
        x -= mod;
    }
}
int main(){
    cin>> s + 1;
    cin>> k;
    n = strlen(s + 1);
    s[++n] = '0';
    d[0][0][0] = 1;
    for(i = 1; i <= n; i++){
        if(s[i] == '1'){
            m++;
        }
    }
    for(i = 1; i <= n + 1; i++){
        if(s[i] == '1'){
            nr++;
            continue;
        }
        t++;
        for(j = 0; j <= m; j++){
            for(ii = 0; ii <= m; ii++){
                for(jj = 0; jj <= m; jj++){
                    if(jj >= nr){
                        x = jj - nr;
                        adauga(d[t][j + x][ii + x], d[t - 1][j][ii]);
                    }
                    else{
                        if(nr - jj <= j){
                            x = nr - jj;
                            adauga(d[t][j - x][ii], d[t - 1][j][ii]);
                        }
                    }
                }
            }
        }
        nr = 0;
    }
    for(i = 0; i <= min(m, k); i++){
        adauga(sol, d[t][0][i]);
    }
    cout<< sol;
}
