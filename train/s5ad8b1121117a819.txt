#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
typedef vector<int> vint;
typedef pair<int, int> pii;

#define abs(a) ((a) < (0)? -(a):(a))
#define rep(i, n) REP(i, 0, (n))
#define REP(i, a, n) for(int i=(a); i<(n); i++)
#define pb push_back

int N = 110000;
bool is_prime[110000];

void set_prime() {
    rep(i, N) is_prime[i] = true;
    is_prime[0] = is_prime[1] = false;
    REP(i, 2, sqrt(N)) {
        if(is_prime[i]) {
            for(int j=0; i*(j+2)<N; j++) {
                is_prime[i*(j+2)] = false;
            }
        }
    }
}

int main() {
    set_prime();
    int n, p;
    while(cin>>n>>p, n+p>0) {
        vint v;
        for(int i=n+1, j = 0; j<20; i++) {
            if(is_prime[i]) j++, v.pb(i);
        }
        vint w;
        rep(i, v.size()) REP(j, i, v.size()) {
            w.pb(v[i]+v[j]);
        }
        sort(w.begin(), w.end());
        cout << w[p-1] << endl;
    }
}
