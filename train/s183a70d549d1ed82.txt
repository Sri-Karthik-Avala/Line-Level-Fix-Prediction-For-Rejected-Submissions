int power(int n, int k)
{
    if(k == 0) return 1;
    int tmp = power(n,k/2); tmp *= tmp; tmp %= mod;
    return (k&1 ? (tmp*n)%mod : tmp);
}

int binomial(int n,int k){
    return ((f[n]*vf[k] % mod)*vf[n - k]) % mod;
}

int invoker(int n, int m)
{
    return binomial(n+m-2,m-1);
}

int bad_path(int id, int i) //
{
    int a = v[id].first; int b = v[id].second;
    int x = v[i].first; int y = v[i].second;
    return (invoker(a-x+1,b-y+1) * dp[i]) % mod;
}

void calc(int id)
{
    int tmp = 0;
    for(int i = 0; i < id; i++){
        tmp += bad_path(id,i);
        tmp %= mod;
    }
    int ans = invoker(v[id].first, v[id].second);
    ans -= tmp; ans %= mod; ans += mod; ans %= mod;
    dp[id] = ans;
}
