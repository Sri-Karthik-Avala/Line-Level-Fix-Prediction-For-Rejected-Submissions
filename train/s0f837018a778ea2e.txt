ll pw(ll x, ll p){
	ll ret=1;
	while(p>0){
		if (p&1){
			ret*=x;
			ret%=mod;
		}
		p>>=1;
		x*=x;
		x%=mod;
	}
	return ret;
}
int op(int a, int qq){
	if (qq==0){
		return a;
	}
	else if (qq==1){
		return (a*3+1)%mod;
	}
	int x=a*pw(3,qq)%mod;
	int y=pw(3,qq-1)*qq%mod;
	return (x+y)%mod;
}
