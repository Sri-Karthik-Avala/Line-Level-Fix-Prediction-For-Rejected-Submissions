#include <bits/stdc++.h>

#define FOR(i,a,b) for(int i=a;i<b;i++)
#define REP(i,b) FOR(i,0,b)
#define PB push_back

using namespace std;

int read(){
	int i;
	scanf("%d",&i);
	return i;
}

typedef long long ll;
const int mod=1000000003;
vector<int> to[50000];

void Merge(int a[2][2],int b[2][2],int (&c)[2][2]){
	REP(i,2)
		REP(j,2)
			c[i][j]=(ll(a[i][0])*b[1][j]+ll(a[i][1])*b[0][j])%mod;
}

void Calc(int l,int r,int (&dst)[2][2]){
	dst[0][0]=1;
	dst[0][1]=0;
	dst[1][0]=0;
	dst[1][1]=1;
	if(l+1==r)
		return;
	int buf[2][2],tmp[2][2];
	FOR(i,1,(int)to[l].size()){
		Calc(to[l][i-1],to[l][i],buf);
		Merge(dst,buf,tmp);
		memcpy(dst,tmp,sizeof(dst));
		dst[1][1]=(dst[1][1]+dst[0][0])%mod;
	}
	if(to[l].back()<r){
		Calc(to[l].back(),r,buf);
		Merge(dst,buf,tmp);
		memcpy(dst,tmp,sizeof(dst));
	}
}

int main(){
	while(1){
		int n=read(),m=read();
		if(!n)
			return 0;
		REP(i,n)
			to[i].clear();
		REP(i,n-1)
			to[i].PB(i+1);
		to[0].PB(n-1);
		REP(i,m){
			int a=read()-1,b=read()-1;
			to[min(a,b)].PB(max(a,b));
		}
		REP(i,n)
			sort(to[i].begin(),to[i].end());
		int ans[2][2];
		Calc(0,n-1,ans);
		printf("%d\n",ans[1][1]);
	}
}
