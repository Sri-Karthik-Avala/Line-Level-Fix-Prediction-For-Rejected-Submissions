int myfind(int a, vector<int>& relation) {
    if (relation[a] == a) return a;
    int res = myfind(relation[a], relation);
    relation[a] = res;
    return res;
}

void myunion(int a, int b, vector<int>& relation) {
    int ta = myfind(a, relation);
    int tb = myfind(b, relation);
    if (ta == tb) return;
    relation[ta] = tb;
}

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> v(n, 0);
    vector<int> relation(n, 0);
    for (int i = 0; i < n; i++) {
        cin >> v[i];
        relation[i] = i;
    }
    int a, b;
    for (int i = 0; i < m; i++) {
        cin >> a >> b;
        myunion(a, b, relation);
    }
    unordered_map<int, vector<int>> um;
    for (int i = 0; i < n; i++) {
        um[myfind(i, relation)].push_back(v[i]);
    }
    if (um.size() <= 1) {
        cout << 0 << endl;
        return 0;
    }
    long long int res = 0;
    int c = 0;
    vector<int> other;
    for (auto it = um.begin(); it != um.end(); it++) {
        sort(it->second.begin(), it->second.end());
        res += it->second[0];
        if (it->second.size() == 1) {
            c++;
        } else {
            for (int i = 1; i < it->second.size(); i++) {
                other.push_back(it->second[i]);
            }
        }
    }
    if (c > 2) {
        cout << "Impossible" << endl;
        return 0;
    }
    sort(other.begin(), other.end());
    res += accumulate(other.begin(), other.begin() + um.size() - 2, (long long int)0);
    cout << res << endl;
}
