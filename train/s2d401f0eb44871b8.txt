#include <string.h>
#include <math.h>
#include<map>
#include<string.h>
#define ll long long
using namespace std;
ll a[60];
int main()
{
    int n,i,j;
    ll ans=0,op=0;
    scanf("%d",&n);
    for(i=1; i<=n; i++)
    {
        scanf("%lld",&a[i]);
    }
    bool flag=1;
    while(flag)
    {
        flag=0;
        for(i=1;i<=n;i++)
        {
            op=a[i]/n;
            if(a[i])
            {
                a[i]%=n;
                ans+=op;
            }
            for(j=1;j<=n;j++)
            {
                if(i!=j)
                {
                    a[i]+=op;
                }
            }
        }
        ll minn=0;
        for(i=1;i<=n;i++)
        {
            minn=max(minn,a[i]);
        }
        if(minn>=n)
        {
            flag=1;
        }
    }
    printf("%lld\n",ans);
    return 0;
}
