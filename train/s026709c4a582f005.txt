// 4近傍（右, 下, 左, 上）
const std::vector<int> dy = { 0, -1, 0, 1 };

using namespace std;

int main() {
    cin.tie(0);
    ios::sync_with_stdio(false);

    while (true) {
        int n, s;
        cin >> n >> s;
        if (n == 0 and s == 0) { break; }

        // dp[i][j] = i個からjを作る組み合わせの数
        VVI dp(n + 2, VI(s + 1, 0));
        dp[0][0] = 1;

        FOR(v, 0, 101) {
            for(int i = n; i >= 0; --i) {
                FOR(j, 0, s + 1) {
                    if (dp[i][j] == 0) { continue; }

                    if (j + v < dp[0].size()) {
                        dp[i + 1][j + v] += dp[i][j];
                    }
                }
            }
        }

        cout << (dp[n][s] == -1 ? 0 : dp[n][s]) << endl;
    }

    return 0;
}
