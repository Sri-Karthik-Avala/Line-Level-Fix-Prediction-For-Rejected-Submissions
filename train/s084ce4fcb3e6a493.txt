int main(){
    string s;
    cin>>s;
    int n=s.length();
    vector<int> dp(n,0);
    int streak=0;
    for(int i=0;i<n; ){
        if(s[i]=='A'){
            streak++;
            if(i-1>=0)dp[i]=dp[i-1];
            i++;
        }else if(i+1<n&&s[i]=='B'&&s[i+1]=='C'){
            if(i-1>=0)dp[i+1]=streak+dp[i-1];
            else dp[i+1]=streak;
            i+=2;
        }else{
            streak=0;
            if(i-1>=0)dp[i]=dp[i-1];
            i++;
        }
    }
    out(dp[n-1]);
}
