#include<iostream>
using namespace std;
int main(){
	int x,y;
	cin>>x>>y;
	if(x<=y){
		for(int i = x;i>=1;i--){
			if(x%i == 0 && y%i == 0){
				cout<<i<<endl;
				break;
			}
		}
	}
	else if(x>=y){
		for(int i = y;i>=1;i--){
			if(x%i == 0 && y%i == 0){
				cout<<i<<endl;
				break;
			}
		}
	}
	return 0;
}
