long long int powint(long long int a, long long int b) {
	assert(b >= 0);
	if (b == 0)return 1;
	if (b == 1)return a;
	long long int ans = 1;
	long long int aa = powint(a, b / 2);
	ans *= aa*aa;
	if (b % 2)ans *= a;
	return ans;
}


int main() {
	
	while (1) {
		vector<int>reads;
		vector<int>writes;
		int N; cin >> N;
		if (!N)break;
		for (int i = 0; i < N; ++i) {
			int a, b; cin >> a >> b;
			reads.push_back(a);
			writes.push_back(b);
		}
		int ans;
		const int readsum = accumulate(reads.begin(), reads.end(), 0);
		int readtime;
		const auto maxid = (max_element(reads.begin(), reads.end()))-reads.begin();
		if (reads[maxid] * 2 > readsum) {
			readtime = reads[maxid] * 2;
			const int resttime = readtime - readsum;//余った時間で感想文書く//maxの以外

			bool dp[1001];
			memset(dp, false, sizeof(dp));
			dp[0] = true;
			for (int j = N - 1; j >= 0; --j) {
			for (int i = 0; i < resttime; ++i) {
				if (dp[i]) {

						if (j != maxid) {
							if (i + writes[j] <= resttime) {
								dp[i + writes[j]] = true;
							}
						}
					}
				}
			}
			int maxtime;
			for (int i = resttime; i >= 0; --i) {
				if (dp[i]) {
					maxtime = i;
					break;
				}
			}
			ans = readtime + accumulate(writes.begin(), writes.end(), 0) - maxtime;

		}
		else {
			readtime = readsum;

			ans = readtime + accumulate(writes.begin(), writes.end(), 0);

		}
		cout << ans << endl;
	}
	
	return 0;
}
