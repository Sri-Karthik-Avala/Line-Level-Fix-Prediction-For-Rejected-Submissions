#include<iostream>
#include<algorithm>
#include<vector>
#include<list>
#define int long long
using namespace std;
list<int>l;
signed main() {
	auto i = l.end();
	int m; cin >> m;
	for (int p = 0; p < m; p++) {
		int c; cin >> c;
		if (c == 0) {
			int x; cin >> x;
			i = l.insert(i, x);
		}
		else if (c == 1) {
			int d; cin >> d;
			if (d > 0)for (int j = 0; j < d; j++)i++;
			else for (int j = 0; j < d; j++)i--;
		}
		else {
			i = l.erase(i);
		}
	}
	for (int p : l)cout << p << endl;
}
