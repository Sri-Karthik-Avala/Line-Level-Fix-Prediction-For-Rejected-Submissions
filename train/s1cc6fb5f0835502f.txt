#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
	ll n, k;
    cin >> n >> k;
    ll w[n], d[n];
    for (int i = 0; i < n; ++i) {
        cin >> w[i] >> d[i];
    }

    ll lb = 1, rb = 2e18;
    while(rb - lb > 1) {
        ll mid = lb + (rb - lb) / 2;
        ll cnt = 0;
        for (int i = 0; i < n; ++i) {
            if(w[i] > mid) continue;
            ll num = (mid - w[i]) / d[i];
            num++;
            cnt += num;
        }
        if(cnt < k) lb = mid;
        else rb = mid;
    }
    cout << rb << endl;
}
