inline ll read(){
    ll x=0,f=1;char c=getchar();
    while ((c<'0'||c>'9')&&(c!='-')) c=getchar();
    if (c=='-') f=-1,c=getchar();
    while (c>='0'&&c<='9') x=x*10+c-'0',c=getchar();
    return x*f;
}

const int N = 1e6+10;
int n,a[N],m,b[N],p[3],cnt[3];
char s[N];
bool flag[N];

int main(){
	scanf("%s",s+1),m=strlen(s+1);
	a[n=1]=s[1]-'A';
	For(i,2,m) if (s[i]-'A'!=a[n]) a[++n]=s[i]-'A';
	For(i,1,n) cnt[a[i]]++;
	FOR(i,0,3) p[i]=i;
	sort(p,p+3,[](int a,int b){
		return cnt[a]<cnt[b];
	});
	a[0]=a[n+1]=-1;
	if (cnt[p[2]]>cnt[p[1]]){
		m=0;
		For(i,1,n){
			if (a[i]==p[2]&&a[i-1]!=a[i+1]) --cnt[p[2]];
				else b[++m]=a[i];
			if (cnt[p[2]]==cnt[p[1]]){
				For(j,i+1,n) b[++m]=a[j];
				break;
			}
		}
		n=m;
		For(i,1,m) a[i]=b[i];
		a[0]=a[n+1]=-1;
	}
	if (cnt[p[2]]>cnt[p[1]]){
		For(i,1,n){
			if (a[i]==p[2]&&a[i-1]==p[0]&&a[i+1]==p[0]) flag[i-1]=flag[i]=1,--cnt[p[0]],--cnt[p[2]];
			if (cnt[p[2]]==cnt[p[1]]) break;
		}
		m=0;
		For(i,1,n) if (!flag[i]) b[++m]=a[i];
		n=m;
		For(i,1,m) a[i]=b[i];
		a[0]=a[n+1]=-1;
	}
	if (cnt[p[2]]>cnt[p[1]]) return 0;
	if (cnt[p[1]]>cnt[p[0]]){
		fil(flag,0);
		int l=1+(a[1]==p[0]),r=l;
		while (a[r+1]!=p[0]&&r<n) ++r;
		for (;cnt[p[0]]<cnt[p[1]]&&l<=n;){
			for (int i=l;i<r-((r-l+1)%2==0);i+=2){
				flag[i]=flag[i+1]=1;
				cnt[p[1]]--,cnt[p[2]]--;
				if (cnt[p[1]]==cnt[p[0]]) break;
			}
			l=r=r+2;
			while (a[r+1]!=p[0]&&r<n) ++r;
		}
		m=0;
		For(i,1,n) if (!flag[i]) b[++m]=a[i];
		n=m;
		For(i,1,m) a[i]=b[i];
	}
	if (cnt[p[1]]==cnt[p[0]]) For(i,1,n) printf("%c",a[i]+'A');
}
