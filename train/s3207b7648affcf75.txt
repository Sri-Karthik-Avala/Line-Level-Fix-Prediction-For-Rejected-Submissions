#include <bits/stdc++.h> 
using namespace std; 
typedef long long LL;
const LL INF = 1e18; 
const int MAX_N = 3e5 + 5; 
int N, M, a[MAX_N]; 
LL s[MAX_N]; 
int stk[MAX_N], top; 
LL F[MAX_N], G[MAX_N], H[MAX_N], y[MAX_N]; 
long double slope(int i, int j) { return 1.0 * (y[j] - y[i]) / (j - i); } 
void Init(LL f[]) { 
    for (int i = 1; i <= N; i++) s[i] = s[i - 1] + a[i];
    top = 0; 
    for (int i = 1; i <= N; i++) { 
        while (top && slope(stk[top - 1], stk[top]) <= 1.0 * i) --top; 
        int j = stk[top]; 
        f[i] = f[j] - s[i - 1] + s[j] + 1ll * (i - j) * (i - j - 1) / 2; 
        y[i] = f[i] + s[i] + 1ll * (i + 1) * i / 2; 
        while (top && slope(i, stk[top]) >= slope(stk[top], stk[top - 1])) --top;
        stk[++top] = i; 
    } 
} 
void chkmax(LL &x, LL y) { if (x < y) x = y; } 
LL t[MAX_N];
//57766
void Div(int l, int r) { 
    if (l == r) return chkmax(H[l], max(1ll - a[l], F[l] + G[l] - a[l] + 1)); 
    int mid = (l + r) >> 1; 
    Div(l, mid), Div(mid + 1, r);
    stk[top = 1] = 0; 
    for (int i = l - 1; i <= mid; i++) { 
        y[i] = F[i] + s[i] + 1ll * (i - 1) * i / 2; 
        while (top > 1 && slope(stk[top], i) >= slope(stk[top - 1], stk[top])) --top; 
        stk[++top] = i; 
    } 
    for (int i = mid + 1; i <= r; i++) { 
        while (top > 1 && slope(stk[top - 1], stk[top]) <= 1.0 * i) --top; 
        int j = stk[top]; 
        t[i] = F[j] - s[i] + s[j] + 1ll * (i - j + 1) * (i - j) / 2; 
    } 
    t[r] += G[r + 1]; 
    for (int i = r - 1; i > mid; i--) 
        t[i] += G[i + 1], chkmax(t[i], t[i + 1]); 
    for (int i = mid + 1; i <= r; i++) chkmax(H[i], t[i]); 
    stk[top = 1] = N + 1; 
    for (int i = mid + 1; i <= r + 1; i++) { 
        y[i] = G[i] - s[i - 1] + 1ll * (i + 1) * i / 2; 
        while (top > 1 && slope(stk[top], i) >= slope(stk[top - 1], stk[top])) --top;
        stk[++top] = i; 
    } 
    for (int i = l; i <= mid; i++) { 
        while (top > 1 && slope(stk[top - 1], stk[top]) <= 1.0 * i) --top; 
        int j = stk[top]; 
        t[i] = G[j] - s[j - 1] + s[i - 1] + 1ll * (j - i) * (j - i + 1) / 2;
    } 
    t[l] += F[l - 1]; 
    for (int i = l + 1; i <= mid; i++) 
        t[i] += F[i - 1], chkmax(t[i], t[i - 1]); 
    for (int i = l; i <= mid; i++) chkmax(H[i], t[i]); 
} 
int main() { 
    scanf("%d", &N); 
    for (int i = 1; i <= N; i++) scanf("%d", a + i); 
    Init(F); 
    reverse(&a[1], &a[N + 1]); 
    Init(G); 
    reverse(&G[1], &G[N + 1]); 
    reverse(&a[1], &a[N + 1]); 
    for (int i = 1; i <= N; i++) s[i] = s[i - 1] + a[i]; 
    for (int i = 1; i <= N; i++) H[i] = -INF; 
    Div(1, N); 
    scanf("%d", &M);
    while (M--) { 
        int p, x; scanf("%d %d", &p, &x); 
        printf("%lld\n", max(F[p] + G[p], H[p] + a[p] - x)); 
    } 
    return 0; 
}
