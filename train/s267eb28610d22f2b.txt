#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int x, y;
    cin >> x >> y;
    vector<int> a = {1, 3, 5, 7, 8, 10, 12};
    vector<int> b = {4, 6, 9, 11};
    for (int i = 0; i < 7; i++) {
        if (x == a[i]) {
            for (int j = 0; j < 7; j++) {
                if (y == a[j]) {
                    cout << "Yes";
                }
            }
        }
    }
    for (int i = 0; i < 4; i++) {
        if (x == b[i]) {
            for (int j = 0; j < 4; j++) {
                if (y == b[j]) {
                    cout << "Yes";
                }
            }
        }
    }

    if (x == 2 && y == 2) {
        cout << "Yes";
    }
    return 0;
    cout << "No";
}
