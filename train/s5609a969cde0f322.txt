#include <bits/stdc++.h>
using namespace std;

const long double INF = 1e9;

int n;
long double x[1005], y[1005], dp[1005][1005];

long double dist(int i, int j) {
    return sqrt((x[i] - x[j]) * (x[i] - x[j]) + (y[i] - y[j]) * (y[i] - y[j]));
}

int main() {

    cin >> n;
    for (int i = 0;  i < n; i++) {
        cin >> x[i] >> y[i];
    }

    dp[1][0] = dist(0, 1);
    for (int i = 2; i < n; i++) {
        for (int j = 0; j + 1 < i; j++) {
            dp[i][j] = dp[i-1][j] + dist(i - 1, i);
            dp[i][i-1] = dp[i-1][j] + dist(j, i);
        }
    }

    long double ans = INF;
    for (int j = 0; j + 1 < n; j++) {
        ans = min(ans, dp[n-1][j] + dist(n-1, j));
    }

    cout << fixed << setprecision(10) << ans << endl;

    return 0;
}
