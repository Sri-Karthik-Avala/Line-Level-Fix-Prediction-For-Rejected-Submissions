int search(int now)
{
	//vis[now] = true;
	
	int res = 1;
	for (int i = 0; i < v[now].size(); i++){
		vis[now] = true;
		if (!vis[v[now][i]]) res = max(res, search(v[now][i]) + 1);
		vis[now] = false;
	}
	
	//vis[now] = false;
	
	return res;
}
