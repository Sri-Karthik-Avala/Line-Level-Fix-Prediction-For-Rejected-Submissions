#include <vector>
#include <queue>
#include <string>
#include <iostream>
#include <cstdio>
#include <stack>
using namespace std;

int gen;
int vis[1 << 24];
int dpt[1<<24];
char ch[1<<24];
queue<int> q;
inline void addq(int d2, int d1, int y2, int x2, int y1, int x1){
	int st = (d2 & 3) << 22 | (d1 & 3) << 20 | y2 << 16 | x2 << 10 |
	         y1 << 6 | x1;
	if(vis[st] != gen){
		vis[st] = gen;
		q.push(st);
	}
}

int main(){
	ios::sync_with_stdio(false);

	int dy[] = {-1, 0, 1, 0};
	int dx[] = {0, -1, 0, 1};
	int h, w;
	while(cin >> w >> h, h){
		++gen;
		vector<string> v(h);
		int kd = -1, ky = -1, kx = -1, md = -1, my = -1, mx = -1;
		for(int i = 0; i < h; ++i){
			cin >> v[i];
		}
		for(int i = 0; i < h; ++i)
		for(int j = 0; j < w; ++j){
			for(int j = 0; j < w; ++j){
				if(v[i][j] == 'K'){
					for(kd = 0; v[i + dy[kd]][j + dx[kd]] == '#'; ++kd);
					ky = i;
					kx = j;
				}
				else if(v[i][j] == 'M'){
					for(md = 0; v[i + dy[md]][j + dx[md]] == '#'; ++md);
					my = i;
					mx = j;
				}
			}
		}
		int d2 = kd, d1 = kd, y2 = ky, x2 = kx, y1 = ky, x1 = kx;
		addq(d2 + 1, d1, y2, x2, y1, x1);
		addq(d2 + 3, d1, y2, x2, y1, x1);
		while(v[y2][x2] != '#'){
			addq(d2, d1, y2, x2, y1, x1);
			y2 += dy[d2];
			x2 += dx[d2];
		}
		addq(0, d1, 0, 0, y1, x1);
		int ans = 0;
		while(!q.empty()){
			int st = q.front();
			q.pop();

			d2 = st >> 22 & 3;
			d1 = st >> 20 & 3;
			y2 = st >> 16 & 15;
			x2 = st >> 10 & 63;
			y1 = st >> 6 & 15;
			x1 = st & 63;

			while(v[y1 + dy[d1]][x1 + dx[d1]] != '#'){
				y1 += dy[d1];
				x1 += dx[d1];
			}

			if(y1 == my && x1 == mx){
				ans |= 1;
				if(y2 == my && x2 == mx && (d2 + 1 & 3) == md){
					ans = 3;
				}
			}

			if(y2){
				if(v[y2 + dy[d2 + 1 & 3]][x2 + dx[d2 + 1 & 3]] == '#'){
					int ny2 = y2, nx2 = x2;
					while(v[ny2][nx2] != '#'){
						addq(d2 + 3, d1 + 3, ny2, nx2, y1, x1);
						ny2 += dy[d2 + 3 & 3];
						nx2 += dx[d2 + 3 & 3];
					}
				}
				if(v[y2 + dy[d2 + 3 & 3]][x2 + dx[d2 + 3 & 3]] == '#'){
					int ny2 = y2, nx2 = x2;
					while(v[ny2][nx2] != '#'){
						addq(d2 + 1, d1 + 1, ny2, nx2, y1, x1);
						ny2 += dy[d2 + 1 & 3];
						nx2 += dx[d2 + 1 & 3];
						if(y1 == my && x1 == mx && ny2 == my && nx2 == mx && (d2 + 3 & 3) == md){
							ans = 3;
						}
					}
				}
			}
			else{
				addq(0, d1 + 1, 0, 0, y1, x1);
				addq(0, d1 + 3, 0, 0, y1, x1);
			}
		}
		
		if(ans == 3){
			puts("He can accomplish his mission.");
		}
		else if(ans == 1){
			puts("He cannot return to the kitchen.");
		}
		else{
			puts("He cannot bring tea to his master.");
		}
	}
}
