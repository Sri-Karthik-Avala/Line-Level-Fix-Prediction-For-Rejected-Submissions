#include<bits/stdc++.h>
using namespace std;
#define int long long
int n, S, M[22];
int dp[1<<15][22];
signed main(){
  while(cin >> n, n) {
    cin >> S;
    for(int i = 0; i < 2*n; i++) cin >> M[i];
    memset(dp, -1, sizeof(dp));
    for(int i = 0; i < 2*n; i++) dp[1][i] = 0;
    for(int i = 2; i <= S; i++){
      bool flag = false;
      for(int j = 0; j < 2*n; j++) {
	for(int k = 1; k <= min(i, M[j]); k++) {
	  if(!dp[i-k][(j+1)%(2*n)]) flag = true;
	}
	dp[i][j] = (int)flag;
      }
    }
    cout << dp[S][0] << endl;
  }
  return 0;
}
