#include<iostream>
#include<algorithm>
#include<cstdio>
using namespace std;
int main(void)
{
	int H=0,W=0,a=0,c=0;
	cin>>H>>W;
	char d=0;
	for(int i=0;i<H;i++){
		for(int j=0;j<W;j++){
			cin>>d;
			if(d=='.'&&c==0){cout<<"-1";}
			else if(d=='c'){cout<<"0";c=1;a=0;}
			else if(d=='.'){a++;cout<<a;}
			if(j==W-1){cout<<endl;}
		}
	}
	return 0;
}
