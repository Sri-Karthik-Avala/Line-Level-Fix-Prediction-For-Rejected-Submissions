#include <iostream>
#include <algorithm>
using namespace std;
int C, D, W, X, a[33][17], b[33][17], dp[33][55][5];
int solve(int d, int s, int x) {
	if (d == D) return 0;
	if (dp[d][s][x]) return dp[d][s][x] - 1;
	int ret = solve(d + 1, s, x);
	for (int i = 0; i < C; i++) {
		if (s + b[d][i] <= W && a[d][i] != 0) {
			int res = solve(d + 1, s + b[d][i], x);
			ret = max(ret, res + a[d][i]);
		}
	}
	if (x != X) {
		for (int i = 0; i < C; i++) {
			int sum = 0, prof = 0;
			for (int j = i; j < C; j++) {
				prof += a[d][j];
				sum += b[d][j];
				if (a[d][j] == 0 || s + sum > W) break;
				int res = solve(d + 1, s + sum, x + 1);
				ret = max(ret, res + prof);
			}
		}
	}
	dp[d][s][x] = ret + 1;
	return ret;
}
int main() {
	while (cin >> C >> D >> W >> X, C) {
		for (int i = 0; i < C; i++) {
			for (int j = 0; j < D; j++) {
				cin >> a[j][i];
			}
		}
		for (int i = 0; i < C; i++) {
			for (int j = 0; j < D; j++) {
				cin >> b[j][i];
			}
		}
		for (int i = 0; i <= D; i++) {
			for (int j = 0; j <= W; j++) {
				for (int k = 0; k <= X; k++) {
					dp[i][j][k] = 0;
				}
			}
		}
		int ret = solve(0, 0, 0);
		cout << ret << endl;
	}
	return 0;
}
