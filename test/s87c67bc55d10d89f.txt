#include <iostream>

using namespace std;

int main(){
  int min,minc,n,m,a,t,mini,minqd,minqdc,minqdi;

  cin >> n >> m;

  while (n !=0 || m !=0){
    cin >> a;
    min=m - m % a;
    minc=1;
    mini=1;
    minqd=a;
    minqdi=1;
    minqdc=1;
    for (int i = 1;i<n;i++){
      cin >> a;
      if ((min/a+1)*a > m) t = (min/a)*a;else t=(min/a+1)*a;
      if (t == min) {
	minc ++;
	if (minqd==a) minqdc ++;
	else if(minqd>a){
	  	minqd=a;
		minqdi=i+1;
		minqdc=1;
	}
      } else if(t<min){
	min=t;
	mini=i+1;
	minc=1;
	minqd=a;
	minqdi=i+1;
	minqdc=1;
      }
    }
    if (minc>=2){
      if (minqdc>=2) cout << m << endl;
      else cout << minqdi << endl;
    }else cout << mini << endl;

    cin >> n >> m;
  }

  return 0;
}
