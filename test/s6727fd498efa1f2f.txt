int N, A[101010];
//--------------------------------------------------------------------------------------------------
void _main() {
    cin >> N;
    rep(i, 0, N) cin >> A[i];

    int st = 0; // =1:??????, =2:????°?
    int cn = 1;
    int ans = 1;
    rep(i, 1, N) {
        if (A[i - 1] < A[i]) {
            if (st == 1) {
                cn++;
            }
            else {
                ans = max(ans, cn);
                st = 1;
                cn = 2;
            }
        }

        if (A[i - 1] > A[i]) {
            if (st == 2) {
                cn++;
            }
            else {
                ans = max(ans, cn);
                st = 1;
                cn = 2;
            }
        }
    }

    ans = max(ans, cn);
    cout << ans << endl;
}
