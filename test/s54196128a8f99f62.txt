const int N=110;

int gi() {
	int w=0;bool q=1;char c=getchar();
	while ((c<'0'||c>'9') && c!='-') c=getchar();
	if (c=='-') q=0,c=getchar();
	while (c>='0'&&c <= '9') w=w*10+c-'0',c=getchar();
	return q? w:-w;
}

int f[N][N*N];
const int mod=1e9+7;
int main()
{
	int n=gi(),k=gi(),i,j,t,ans=k*(k+1)/2*n;
	for (i=1;i<n;i++) ans=1LL*ans*(k+1)%mod;
	f[n][0]=1;
	for (i=n;i;i--)
		for (j=0;f[i][j];j++)
			for (t=0;t<=k;t++)
				(f[i-1][j+(t<=i?(j+t)/i:0)]+=f[i][j])%=mod;
	for (j=0;f[0][j];j++) (ans-=f[0][j])%=mod;
	cout<<(ans+mod)%mod<<endl;
	return 0;
}
