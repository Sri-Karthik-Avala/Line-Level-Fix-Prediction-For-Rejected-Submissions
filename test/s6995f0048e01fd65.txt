#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, a, b;
  a =0, b=0;
  cin>>n;
  
  vector<int>vect(n);
  for(int i = 0; i<n; i++){
    cin>> vect.at(i);
  }
  
  sort(vect.begin(), vect.end());  
  reverse(vect.begin(), vect.end());
  
  for(int i = 0; i<n; i +=2){
    a += vect.at(i);
    b += vect.at(i+1);
  }
  
  cout<< a - b<< endl;

}
