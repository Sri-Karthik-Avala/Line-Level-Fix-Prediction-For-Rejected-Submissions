#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (int)(n); i++)
using ll = long long;
const int M = 1e7;
int si[] = {-1, 0, 1, 0};
int sj[] = {0, -1, 0, 1};

int main(){
  int n;
  cin >> n;

  int ans = M;
  for(int a = 1; a < n-1; a++){
    int b = n-a;
    int k = 0;
    string x = to_string(a);
    rep(i,x.size()) k += (int)x[i] - (int)'0';
    string y = to_string(b);
    rep(i,y.size()) k += (int)y[i] - (int)'0';
    ans = min(k,ans);
  }

  cout << ans << endl;
  return 0;
}
