#include <cstdio>
using namespace std;

int dx[]={1,0,2,2,0,1,1,0,2};
int dy[]={1,0,2,0,2,0,2,1,1};
int f[3][3],cnt;

bool check(int p, int val, int mod)
{
	if(p==4) return (val==(f[2][0]+f[1][1]+f[0][2])%mod);
	if(p==5) return (val==(f[0][0]+f[1][0]+f[2][0])%mod);
	if(p==6) return (val==(f[0][2]+f[1][2]+f[2][2])%mod)&&(val==(f[1][0]+f[1][1]+f[1][2])%mod);
	if(p==7) return (val==(f[0][0]+f[0][1]+f[0][2])%mod);
	if(p==8) return (val==(f[2][0]+f[2][1]+f[2][2])%mod)&&(val==(f[0][1]+f[1][1]+f[2][1])%mod);
	
	return true;
	
}

void dfs(int p, int mod, int val, int used)
{
	if(p==3)
	{
		val=(f[0][0]+f[1][1]+f[2][2])%mod;
	}
	if(p==9)
	{
		 cnt++;
		 return;
	}
	
	int x=dx[p], y=dy[p];
	if(f[x][y]!=0) dfs(p+1, mod, val, used);
	else
	{
		for(int i=1; i<=10; i++)
		{
			if(used&(1<<i)) continue;
			f[x][y]=i;
			if(check(p,val,mod)) dfs(p+1, mod, val, used|(1<<i));
			f[x][y]=0;
		}
	}
}

int main()
{
	while(1)
	{
		cnt=0;
		int mod, used=0;
		for(int i=0; i<3; i++)
		for(int j=0; j<3; j++)
		{
			scanf("%d", &f[j][i]);
			if(f[j][i]!=-1) used|=(1<<f[j][i]);
		}
		
		if(f[0][0]==-1) break;
		
		scanf("%d", &mod);
		if(mod==0)
		{
			for(int i=1; i<=10; i++)
			{
				if(used&(1<<i)) continue;
				dfs(0, i, -1, used|(1<<i));
			}
		}
		else dfs(0, mod, -1, used|(1<<mod));
		printf("%d\n", cnt);
	}
}
