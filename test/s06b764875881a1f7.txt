    while (1) {
        cin >> n >> h;
        if (n == 0 && h == 0) break;
        set<cube_t> cube;
        for (int i = 0; i < h; i++) {
            cin >> s >> a >> b;
            for (int j = 0; j < n; j++)
            if (s == "xy") {
                cube.insert(make_pair(a, make_pair(b, j)));
            } else if (s == "xz") {
                cube.insert(make_pair(a, make_pair(j, b)));

            }

    }
