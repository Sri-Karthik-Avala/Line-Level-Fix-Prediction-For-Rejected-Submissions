#include<bits/stdc++.h>
using namespace std;

const int N = 100005;

int n, ans;
bool f[N];
vector<int> e[N];

void dfs(int x, int fa)
{
	int s = 0;
	for (int to : e[x]) if (to != fa)
	{
		dfs(to, x);
		s += !f[x];
		f[x] |= f[to];
	}
	if (s > 1) f[x] = 1, ans += s - 1;
}

int main()
{
	scanf("%d", &n);
	for (int i = 1; i < n; i++)
	{
		int x, y; scanf("%d%d", &x, &y);
		x++; y++;
		e[x].push_back(y); e[y].push_back(x);
	}
	int rt = 0;
	for (int i = 1; i <= n; i++) if (e[i].size() > 2) {rt = i; break;}
	if (!rt) {printf("%d\n", 1); return 0;}
	dfs(rt, 0);
	printf("%d\n", ans);
	return 0;
}
