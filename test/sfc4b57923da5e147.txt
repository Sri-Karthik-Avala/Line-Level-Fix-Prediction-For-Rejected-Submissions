int main(){
    ll n,h,w;
    cin>>n>>h>>w;
    vector<ll> a(w*n+1,0);
    for(int i=0;i<n;i++){
        ll x;
        cin>>x;
        if(i&1){
            a[i*w-x]++;
            a[i*w-x+w]--;
        }
        else{
            a[i*w+x]++;
            a[i*w+x+w]--;
        }
    }
    for(int i=1;i<w*n;i++){a[i]+=a[i-1];}
    ll count=-1;
    for(auto &i:a){if(i==0){count++;}}
    cout<<count*h<<endl;
    
    return 0;
}
