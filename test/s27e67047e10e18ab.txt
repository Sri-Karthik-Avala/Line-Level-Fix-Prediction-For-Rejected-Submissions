#include<bits/stdc++.h>
using namespace std;

typedef long long ll; 
ll m=1e9+7;
ll dp[1002][1002];


int main(){
  ll n,k;
  cin>>n>>k;
  n-=k;
  dp[0][0]=1;
  for(int i=0;i<=n;i++)
    for(int j=0;j<=k;j++){
      if(j)dp[i][j]+=dp[i][j-1];
      if((i||j)&&i-j>=0)dp[i][j]+=dp[i-j][j];
      dp[i][j]%=m;
    }
  
  cout<<dp[n][k]<<endl;
  return 0;
}
