# coding: utf-8

while True:
	n, m = map(int, raw_input().split())
	if n == 0:
		break
	else:
		data = []
		for i in range(n):
			student = map(int, raw_input().split())
			data.append(student)
		result = []
		for j in range(m):
			sum = 0
			for i in range(n):
				sum += data[i][j]
			result.append(sum)

		ans = []
		for i in range(1,m+1):
			ans.append(i)
		for i in range(1,m)[::-1]:
			for j in range(1,i)[::-1]:
				if result[j-1] < result[j]:
					tmp = ans[j-1]
					ans[j-1] = ans[j]
					ans[j] = tmp
		ans_text = ""
		for i in range(m-1):
			ans_text += str(ans[i]) + " "
		ans_text += str(ans[m-1])
		print(ans_text)
