#include <cstdio>
#include <cstring>
#include <vector>
#include <map>
#include <algorithm>
#include <utility>

using namespace std;
map<int,map<int,int> > data;
int n;
int x[100001],y[100001],w[100001];
int dx[4]={0,-1,0,-1};
int dy[4]={0,0,-1,-1};
int main(void){
	scanf("%d",&n);
	for(int i=0;i<n;i++){
		scanf("%d %d %d",&x[i],&y[i],&w[i]);
		data[x[i]][y[i]]+=w[i];
	}
	int res=0;
	for(int i=0;i<n;i++){
		for(int j=0;j<4;j++){
			int cnt=0,nx=x[i]+dx[j],ny=y[i]+dx[j];
			for(int k=0;k<=1;k++){
				for(int l=0;l<=1;l++){
					cnt+=data[nx+k][ny+l];
				}
			}
			res=max(res,cnt);
		}
	}
	printf("%d / 1\n",res);
	return 0;
}
