double calc(double a,double b,double c){
	double s = (a+b+c)/2;
	return sqrt(s*(s-a)*(s-b)*(s-c));
}

void func(double a,double l,double x){

	double tmp = (l+x)/2;

	printf("%.10lf\n",calc(a,l,l)+calc(tmp,tmp,l)*2);
}


int main(){

	double a,l,x;

	while(scanf("%lf %lf %lf",a,l,x) != EOF){
		func(a,l,x);
	}
}
