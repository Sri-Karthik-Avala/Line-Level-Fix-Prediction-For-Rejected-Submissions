#include<iostream>
using namespace std;
int n,k,ans,a[100005]; 
int main(){
	ios::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);
	cin>>n>>k;
	ans=1e9;
	for(int i=1;i<=n;i++){
		cin>>a[i];
	}
	for(int i=1,x,y;i<=n;i++){
		if(i+k-1>n) break;
		x=a[i];
		y=a[i+k-1];
		if(y<0){
			ans=min(ans,0-x);
		}
		else if(x<0&&y>0){
			ans=min(ans,min(y-x-x,y-x+y));
		}
		else if(x>0){
			ans=min(ans,y);
		}
	}
	cout<<ans;
}
