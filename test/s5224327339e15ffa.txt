typedef complex<double> P;
typedef long long int Int;
const Int N = 3;
const double EPS = 1e-8;

struct L : public vector<P> {
  L(){}
  L(const P &a, const P &b) {
    push_back(a); push_back(b);
  }
};
typedef vector<P> G;

double cross(const P& a, const P& b) { return imag(conj(a)*b); }
double dot(const P& a, const P& b) { return real(conj(a)*b); }

Int ccw(P a, P b, P c) {
  b -= a; c -= a;
  if (cross(b, c) > 0)   return +1;       // counter clockwise
  if (cross(b, c) < 0)   return -1;       // clockwise
  if (dot(b, c) < 0)     return +2;       // c--a--b on line
  if (norm(b) < norm(c)) return -2;       // a--b--c on line
  return 0;
}

bool intersectSS(const L &s, const L &t) {
  return ccw(s[0],s[1],t[0])*ccw(s[0],s[1],t[1]) <= 0 &&
    ccw(t[0],t[1],s[0])*ccw(t[0],t[1],s[1]) <= 0;
}

pair<bool, P> crosspointLL(const L &l, const L &m) {
  double A = cross(l[1] - l[0], m[1] - m[0]);
  double B = cross(l[1] - l[0], l[1] - m[0]);
  if (abs(A) < EPS && abs(B) < EPS) return pair<bool, P>(0, m[0]); // same line
  if (abs(A) < EPS) assert(false); // !!!PRECONDITION NOT SATISFIED!!!
  return pair<bool, P>(1, m[0] + B / A * (m[1] - m[0]));
}

bool input(L &l){
  Int x1, y1, x2, y2; cin >>x1 >>y1 >>x2 >>y2;
  l = L(P(x1, y1), P(x2, y2));
  return (x1 != 0 || y1 != 0 || x2 != 0 || y2 != 0);
}

double area(const G &P) {
  double A = 0.0;
  for (Int i = 0; i < P.size(); ++i) A += cross(curr(P, i), next(P, i));
  return abs(A / 2.0);
}

string solve(L ls[N]){
  G g;
  REP(i, N){
    if(!intersectSS(ls[i], ls[(i + 1) % N])) continue;
    else{
      pair<bool, P> res = crosspointLL(ls[i], ls[(i + 1) % N]);
      if(res.first) g.push_back(res.second);
    }
  }
  double r = area(g);
  if(g.size() < 3) return "kyo";
  if(r < 100000) return "syo-kichi";
  if(r < 1000000) return "kichi";
  if(r < 1900000) return "chu-kichi";
  return "dai-kichi";
}

int main() {
  L ls[N];
  while(input(ls[0])){
    FOR(i, 1, N) input(ls[i]);
    cout <<solve(ls) <<endl;
  }
  return 0;
}
