#include<bits/stdc++.h>
using namespace std;
struct edge{
  int cost, to;
};

void hasInputEdge(vector<vector<edge> > &ing, edge e, int v){
  for (int j = 0; j < ing[e.to].size(); j++) {
    if(ing[e.to][j].to == v){
      ing[e.to].erase(ing[e.to].begin() + j);
    }
  }
}


vector<int> tpSort(vector<vector<edge> > ing,
                   vector<vector<edge> > outg){
  queue<int> que;
  for (int i = 0; i < ing.size(); i++) {
    if(ing[i].size() == 0) que.push(i);
  }
  vector<int> res;
  while(que.size()){
    int v = que.front(); que.pop();
    res.push_back(v);
    for (int i = 0; i < outg[v].size(); i++) {
      edge e = outg[v][i];
      hasInputEdge(ing, e, v);
      if(ing[e.to].size() == 0) que.push(e.to);
    }
  }
  return res;
}
  
int main(int argc, char *argv[]){
  int n, m;
  cin >> n >> m;
  vector<vector<edge> >outg(n); 
  vector<vector<edge> >ing(n);
  for (int i = 0; i < m; i++) {
    int from;
    edge e;
    cin >> from >> e.to >> e.cost;
    outg[from].push_back(e);
    ing[e.to].push_back((edge){e.cost, from});
  }
  vector<int> tp = tpSort(ing, outg);
  // std::cout << "tp:" << std::endl;
  // for (int i = 0; i < n; i++) {

  // for (int i = 0; i < n; i++) {

  //     std::cout << outg[i][j].to << " ";
  //   }
  //   std::cout << std::endl;
  // }
  // std::cout << std::endl;
  int dp[n];
  memset(dp, 0, sizeof(dp));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < outg[i].size(); j++) {
      int nv = outg[tp[i]][j].to;
      dp[nv] = max(dp[nv],
                   dp[tp[i]] + outg[tp[i]][j].cost);
    }
  }

}
