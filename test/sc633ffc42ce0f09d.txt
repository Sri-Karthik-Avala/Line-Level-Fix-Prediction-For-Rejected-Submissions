#include<stdio.h>
#include<algorithm>
using namespace std;
double dp[20][1<<16];
double c[20][20];
int m,n;
double solve(int a,int b){
	if(dp[a][b]>-0.5)return dp[a][b];
	if(a==n)return 1;
	if(b==0)return 0;
	double ret=0;
	for(int i=0;i<m;i++){
		if(b&(1<<i)){
			ret=max(ret,c[i][a]*solve(a+1,b)+(1.0-c[i][a])*solve(a,b-(1<<i)));
		}
	}
	return dp[a][b]=ret;
}
int main(){
	int a,b;scanf("%d%d",&a,&b);
	m=a;n=b;
	for(int i=0;i<a;i++)for(int j=0;j<b;j++)scanf("%lf",&c[i][j]);
	for(int i=0;i<=b;i++)for(int j=0;j<(1<<a);j++)dp[i][j]=-1;
	printf("%.12f\n",solve(0,(1<<a)-1));
}
