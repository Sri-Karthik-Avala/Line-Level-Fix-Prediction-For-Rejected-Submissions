#include<iostream>
#include<string>
int main() {
	std::string str;
	std::getline(std::cin, str);
	int k = 0;
	int u = 0;
	int p = 0;
	int c = 0;
	int kanban = 0;
	for (int i = 0;i < str.size();++i) {
		if (str[i] == 'K') 
			k++;
		if (str[i] == 'U')
			++u;
			if (str[i] == 'P')
				p++;
			if (str[i] == 'C')
				++c;
	}
	while (1) {
		if ((k > 0)&&(u > 0)&&(p > 0)&&(k > 0)) {
			k--;u--;p--;c--;kanban++;
		}
		else {
			break;
		}
	}
	std::cout << kanban << std::endl;
	return 0;
}
