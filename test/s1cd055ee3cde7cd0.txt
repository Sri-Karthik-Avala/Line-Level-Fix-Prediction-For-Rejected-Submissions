#include<stdio.h>
int main()
{
	int t1[100][100], t2[100][100], n, i,j,k,c=0;
	scanf("%d", &n);
	for (i = 0; i < n; i++){
		for (j = 0; j < 3; j++){
			scanf("%d", &t1[j][i]);
		}
	}
	for (i = 0; i < n; i++){
		for (j = 0; j < 3; j++){
			t2[j][i] = t1[j][i];
		 }
	}
	for (j = 0; j < 3; j++){
		for (i = 0; i < n; i++){
			for (k = 0; k < n; k++){
				if (t1[j][i] == t1[j][k] && i != k){
					t2[j][i] = 0;
					t2[j][i] = 0;
				}
			}
		}
	}
	for (i = 0; i < n; i++){
		for (j = 0; j < 3; j++){
			c += t2[j][i];
		}
		printf("%d\n", c);
		c = 0;
	}
	return 0;
}
