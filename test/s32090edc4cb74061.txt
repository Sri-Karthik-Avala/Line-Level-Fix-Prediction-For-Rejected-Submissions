bool isin(int a,int b,int c,int d,int p,int q)
{
	//printf("%d %d %d %d %d %d\n",a,b,c,d,p,q);
	ld s,t;
	ld A=a,B=b,C=c,D=d,P=p,Q=q;
	if(b==0)
	{
		t=Q/D;
		s=P/A-(C*Q)/(A*D);
		//cout << s << ' ' << t << endl;
		if(s>0&&t>0&&s+t<1)return true;
		else return false;
	}
	if(a==0)
	{
		t=P/C;
		s=Q/B-(D*Q)/(B*C);
		//cout << s << ' ' << t << endl;
		if(s>0&&t>0&&s+t<1)return true;
		else return false;
	}
	t=(P*B-A*Q)/(B*C-A*D);
	s=(P-C*t)/A;
	//cout << s << ' ' << t << endl;
	if(s>0&&t>0&&s+t<1)return true;
	else return false;
}
