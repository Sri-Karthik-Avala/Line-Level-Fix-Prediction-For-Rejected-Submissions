double rec(double high,double low){
	if(high-low<0.001)return high;
	double maxp=1000000;
	double minp=0;
	double mid=(high+low)/2;
	rep(i,n){
		double s=r[i]*r[i]-mid*mid;
		s=sqrt(s);
		maxp=min(maxp,x[i]+s);
		minp=max(minp,x[i]-s);
	}
	if(maxp<minp){
		return rec(mid,low);
	}else{
		return rec(high,mid);
	}
}
