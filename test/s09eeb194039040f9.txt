#include<stdio.h>
#include<algorithm>
using namespace std;
int p[20];
int q[20];
int dp[1<<14];
int main(){
	int a,b;
	while(scanf("%d%d",&a,&b),a){
		for(int i=0;i<a;i++)scanf("%d%d",p+i,q+i);
		for(int i=0;i<(1<<a);i++)dp[i]=99999999;
		dp[0]=0;
		for(int i=0;i<(1<<a);i++){
			int now=(1<<a)-1-i;
			int fr=b;
			for(int j=0;j<a;j++)if(i&(1<<j))fr+=p[j]-q[j];
			while(now){
				int tmp=0;
				for(int j=0;j<a;j++)if(now&(1<<j))tmp+=q[j];
				if(fr>=tmp){
					dp[i+now]=min(dp[i+now],dp[i]+1);
				}
				now=(now-1)&((1<<a)-1-i);
			}
		}
		if(dp[(1<<a)-1]>99999)printf("Impossible\n");
		else printf("%d\n",dp[(1<<a)-1]);
	}
}
