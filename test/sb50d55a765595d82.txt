int main() {

	cin >> N;
	vector<long long int>p(N);
	vector<long long int>s(N);
	for (int i = 0; i < N; i++) {
		cin >> p[i] >> s[i];
	}
	long long int ans = s.back();
	long long int box = s.back() - p.back();
	for (int i = N - 2; i >= 0; i--) {
		box = max(box + s[i], s[i]-p.back());
		ans = max(ans, box + p[i]);
	}
	cout << ans << endl;
