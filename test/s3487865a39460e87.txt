
#include<bits/stdc++.h>
using namespace std;

int main(){
  int M,N,ans=0;;
  int f[100][100]={0};
  string s[100];

  cin>>M>>N;
  for(int i=0;i<M;i++)
    cin>>s[i];

  for(int i=0;i<M;i++){
    for(int j=0;j<N;j++){
      if(s[i][j]=='x'){
	ans++;
	for(int c=-1;c<=1;c++){
	  for(int r=-1;r<=1;r++){
	    if(i+c>=0&&i+c<N&&j+r>=0&&j+r<M){
	      if(s[i+c][j+r]=='-'&&f[i+c][j+r]==0){
		f[i+c][j+r]=1;
		ans++;
	      }
	    }
	  }
	}
      }else if(s[i][j]=='o'){
	ans++;
	if(j-1>=0&&s[i][j-1]=='-'&&f[i][j-1]==0){
	  f[i][j-1]=1;
	  ans++;
	}
	if(j+1<N&&s[i][j+1]=='-'&&f[i][j+1]==0){
	  f[i][j+1]=1;
	  ans++;
	}
      }else if(i==0){
	if(s[i][j]='-'&&f[i][j]==0){
	  f[i][j]=1;
	  ans++;
	}
      }
    }
  }

  cout<<M*N-ans<<endl;
  
  return 0;
}
