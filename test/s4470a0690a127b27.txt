const int MAX_N = 100005;
const double PI = 3.141592653589793;

bool possible(double cri,vi& r){
    int n = (int)r.size();
    double sm = 0;
    rep(i,n){
        sm += acos(2*cri/(r[i]*r[(i+1)%n]));
    }
    return (sm < 2*PI);
}

int main()
{
    cin.tie(0);
    ios::sync_with_stdio(false);
    int n;
    cin >> n;
    vi r(n);
    rep(i,n){
        cin >> r[i];
    }
    double ans = 0;
    srep(i,3,n+1){
        vector<bool> v(n,false);
        fill(v.end() - i, v.end(), true);
        do{
            vector<int> vec;
            rep(j,n){
                if(v[j]){
                    vec.pb(r[j]);
                }
            }
            sort(all(vec));
            do{
                double mn = INF;
                rep(j,i){
                    cmn(mn,(double)vec[j]*vec[(j+1)%i]);
                }
                double l = -mn/2;
                double r = mn/2;
                while(r-l > EPS){
                    double mid = (l+r)/2;
                    if(possible(mid,vec)){
                        r = mid;
                    }else{
                        l = mid;
                    }
                }
                vd st(i);
                double sm = 0;
                rep(j,i){
                    st[j] = acos(2*l/(vec[j]*vec[(j+1)%i]));
                    sm += vec[j]*vec[(j+1)%i]*sin(st[j])/2.0;
                }
                cmx(ans,sm);
            }while(next_permutation(all(vec)));
        } while (next_permutation(all(v)));
    }
    printf("%.12lf\n",ans);
    return 0;
}
