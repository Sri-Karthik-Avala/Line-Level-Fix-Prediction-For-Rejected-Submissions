vector<vector<int>>memo(2, vector<int>((1 << 20)));
int main() {
	int N; cin >> N;
	for (int num = 0; num < N;++num){
		int W, H; cin >> W >> H;
		vector<vector<int>>field(H, vector<int>(W));
		for (int y = 0; y < H; ++y) {
			for (int x = 0; x < W; ++x) {
				cin >> field[y][x];
			}
		}
		memo = vector<vector<int>>(2, vector<int>(1 << W));
		for (int y = 0; y < H; ++y) {
			for (int x = 0; x < W; ++x) {
				const int i = y*W + x;
				int tar = i & 1, cur = tar ^ 1;
				if (i == 0)assert(tar == 0 && cur == 1);
				for (int pre = 0; pre < (1 << W); ++pre) {
					if (((x==0)||!(pre & 1)) && !(pre&(1 << (W-1)))) {
						memo[cur][((pre << 1) + 1) & ((1 << W) - 1)] = max(memo[cur][((pre << 1) + 1) & ((1 << W) - 1)], memo[tar][pre] + field[y][x]);
					}
					memo[cur][(pre << 1)&((1<<W)-1)] = max(memo[cur][(pre << 1)&((1 << W) - 1)], memo[tar][pre]);
				}
			}
		}
		int ans=0;
		for (int i = 0; i < (1 << W); ++i) {
			ans = max(ans, memo[(W*H) % 2][i]);
		}
		cout << "Case " << num+1 << ": " << ans << endl;
		
	}
	return 0;
}
