#include <algorithm>
#include <iostream>
#include <set>
#include <vector>

using namespace std;

typedef long long ll;

int main() {
    int n, m;
    while (cin >> n >> m, n || m) {
        vector<ll> k(n);
        for (int i = 0; i < n; ++i) {
            cin >> k[i];
            k[i] %= m;
        }

        vector<ll> sum(n + 1);
        for (int i = 0; i < n; ++i) {
            sum[i + 1] = (sum[i] + k[i]) % m;
        }

        ll ans = 0;
        set<ll> s;
        s.insert(0);
        for (int i = 1; i <= n; ++i) {
            auto it = s.lower_bound(sum[i] + 1);
            if (it != s.end()) {
                ans = max(ans, (sum[i] + (*it)) % m);
            } else {
                ans = max(ans, sum[i]);
            }

            s.insert(sum[i]);
        }

        cout << ans << endl;
    }
}
