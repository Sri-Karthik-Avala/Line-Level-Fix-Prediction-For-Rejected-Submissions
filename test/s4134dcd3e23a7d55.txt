using namespace std;

int main(){
	bool is_die[1000]={};
	int N,M;
	while(cin>>N>>M,N||M){
		for(int i=0,cntM=1,cntH=N;;i=++i%N){
			if(is_die[i]==true) continue;
			if(cntH==1){
				printf("%d\n",i+1);
				break;
			}
			if(cntM==M){
				is_die[i] = true;
				cntH--;
				cntM = 0;
			}
			cntM++;
		}
	}
	return 0;
}
