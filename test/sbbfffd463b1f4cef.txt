char in[10][10];

ll MOD = 100000007;
ll dp[2][1<<22];

vector<int> state;
int r,c;
void dfs(int dep,int cu){
  if(dep==c+1){
    state.pb(cu);
    return;
  }

  rep(i,3)
    dfs(dep+1,cu*4+i);
}

main(){

  cin >> r >> c;
  rep(i,r)rep(j,c){
    cin >> in[i][j];
    if(isdigit(in[i][j])) in[i][j]-='0';
    else in[i][j]=0;
  }

  dfs(0,0);
  
  int p=0;
  dp[0][0] = 1;
  rep(i,r)rep(j,c){
    //memset(dp[!p],0,sizoef(dp[!p]));
    FOR(it,state) dp[!p][*it]=0;
    p=!p;
    FOR(it,state){
      //cout << i << ' ' << j << ' ' << ' ' << *it << ' ' << dp[!p][*it] << endl;
      int up=(*it>>(2*c))&3;
      int left=(*it)&3;
      int ed=in[i][j];
      int cst = *it;
      if(j==0 && left) continue;
      if(ed && up+left>ed) continue;
      if(ed==0){
        if(up && left) continue;
        cst <<= 2;
        cst &= (1<<(c+1)*2)-1;
        cst &= ~15;        
        if(up) cst |= up<<2;
        else cst |= left;
        dp[p][cst] =(dp[p][cst]+dp[!p][*it])%MOD;
      }else{
        if(ed>4) continue;
        ed -= up+left;
        cst <<= 2;
        cst &= (1<<(c+1)*2)-1;
        cst &= ~15;
        for(int nleft=0;nleft<=min(2,ed);++nleft){
          if(ed-nleft>2) continue;
          int nst=cst;
          nst |= nleft|((ed-nleft)<<2);
          dp[p][nst] =(dp[p][nst]+dp[!p][*it])%MOD;
        }
      }
      
    }
  }

  cout << dp[p][0] << endl;
}
