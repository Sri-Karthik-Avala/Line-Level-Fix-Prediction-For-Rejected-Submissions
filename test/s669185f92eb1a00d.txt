#include <cstdio>
#include <cstring>
#include <iostream>
#include <string>
#include <cmath>
#include <bitset>
#include <vector>
#include <map>
#include <set>
#include <queue>
#include <deque>
#include <algorithm>

using namespace std;
typedef long long int ll;
typedef pair<int, int> P;
typedef pair<string, P> PS;

struct edge{
	int to;
	string lab;
};

int main()
{
	while(1){
		int n, a, s, g;
		cin>>n>>a>>s>>g;
		if(n==0 && a==0 && s==0 && g==0) return 0;
		vector<edge> v[40];
		for(int i=0; i<a; i++){
			int x, y;
			string l;
			cin>>x>>y;
			cin>>l;
			edge e;
			e.to=y, e.lab=l;
			v[x].push_back(e);
		}
		string d[40][501];
		for(int i=0; i<n; i++){
			for(int j=0; j<501; j++){
				if(i==s && j==0) continue;
				d[i][j]+='0';
			}
		}
		priority_queue<PS, vector<PS>, greater<PS> > que;
		que.push(PS(d[s][0], P(s, 0)));
		while(!que.empty()){
			PS p=que.top();
			que.pop();
			int x=p.second.first, lx=p.second.second;
			if(d[x][lx]<p.first) continue;
			for(auto e:v[x]){
				int y=e.to;
				string str=e.lab;
				int m=str.size();
				if(lx+m<=500 && (d[y][lx+m]=="0" || d[y][lx+m]>(d[x][lx]+str))){
					d[y][lx+m]=(d[x][lx]+str);
					que.push(PS(d[y][lx+m], P(y, lx+m)));
				}
			}
		}
		string ans;
		ans+='0';
		for(int i=0; i<=500; i++){
			if(d[g][i]=="0") continue;
			if(ans=="0" || ans>d[g][i]) ans=d[g][i];
		}
		if(ans=="0" || ans.size()==500){
			cout<<"NO"<<endl;
		}else{
			cout<<ans<<endl;
		}
	}
	return 0;
}
