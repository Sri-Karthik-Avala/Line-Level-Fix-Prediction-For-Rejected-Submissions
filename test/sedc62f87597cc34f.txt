#include <bits/stdc++.h>
using namespace std;

int N, M;
int A[3];
vector<int> G[51];
bool used[51];
int cnts[51];
bool dp[101][51][51];  // a step, b start, c goal

int main(){

  cin>>N>>M;
  for(int i=0;i<3;i++)cin>>A[i];
  for(int i=0;i<M;i++){
    int f,t;
    cin>>f>>t;
    f--;t--;
    G[f].push_back(t);
  }
  int ite = 0;

  for(int i=0;i<N;i++)
    dp[0][i][i] = true;
  for(int i=1;i<101;i++){
    for(int j=0;j<N;j++){
      // ??????step??§??°?????????????????´??????????????????, ???????????????????????´????????????
      for(int k=0;k<N;k++){
        if(dp[i-1][j][k]){
          // k?????????????????´???
          for(int to : G[k]){
            dp[i][j][to] = true;
          }
        }
      }
    }
  }
  
  used[N-1] = true;
  cnts[N-1] = 0;
  // N step??§???????????´????????????
  while(1){
    bool ok = false;
    for(int i=0;i<N;i++){
      if(used[i])continue;
      // ???????????????????????????
      int cnt = 0;
      int max_step = -1;
      for(int j=0;j<3;j++){
        // ???????????????step??§, goal?????´????????°?????§??????????????????
        int a = A[j];
        bool t = false;
        int min_step = 100000000;
        for(int k=0;k<N;k++){
          if(dp[a][i][k] && used[k]){
            // cnt++;
            min_step = min(min_step, cnts[k] + 1);
            t = true;
          }
        }
        if(t){
          cnt++;
          max_step=max(max_step, min_step);
        }
      }
      // OK
      if(cnt == 3){
        cnts[i] = max_step;
        used[i]=true;
        ok=true;
        break;
      }
    }
    if(ok){
      ite++;
      if(used[0]){
        cout<<ite<<endl;
        return 0;
      }
    }
    else{
      break;
    }
  }
  cout<<"IMPOSSIBLE"<<endl;
  
  return 0;
}
