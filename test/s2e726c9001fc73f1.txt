inline string getstr(string &s,int l,int r){string ret="";for(int i=l;i<=r;i++)ret.push_back(s[i]);return ret;}
int modpow(int x,int y,int md=mod){int ret=1;do{if(y&1)ret=(ll)ret*x%md;x=(ll)x*x%md;}while(y>>=1);return ret;}
inline int Rand(){return rand()*32768+rand();}
#define SIZE 131072
int f[SIZE<<1],g[SIZE<<1];
int n,k,a[SIZE],b[SIZE],l[SIZE],r[SIZE];
int inv[SIZE<<1];
map<int,int>mp;
void rc(int l,int r,int k,const int &ql,const int &qr,const int &val){
	if(ql>qr)return;
	if(r<ql||qr<l)return;
	if(ql<=l&&r<=qr){
		g[k]=min(g[k],val);
		return;
	}
	rc(l,(l+r)/2,k<<1,ql,qr,val);
	rc((l+r)/2+1,r,k<<1|1,ql,qr,val);
}
int query(int l,int r,int k,const int &pos){
	if(l==r)return g[k];
	int mid=(l+r)/2;
	if(pos<=mid)return min(g[k],query(l,mid,k<<1,pos));
	else return min(g[k],query(mid+1,r,k<<1|1,pos));
}
signed main(){
	memset(g,inf,sizeof g);
	ios::sync_with_stdio(0);
	cin.tie(0);cout.tie(0);
	cin>>n>>k;
	for(int i=1;i<=n;i++){
		cin>>a[i]>>b[i];
		if(b[i]==1&&a[i]*2>k){
			puts("-1");
			return 0;
		}
		a[i]+=a[i-1];
		if(b[i]==2)l[i]=0,r[i]=k-1;
		else l[i]=(-a[i-1]*2ll%k+k)%k,r[i]=(-a[i]*2ll%k+k)%k;
		mp[l[i]];mp[r[i]];
	}
	int tot=0;for(auto &p:mp)p.second=++tot;
	for(int i=1;i<=n;i++){
		inv[mp[l[i]]]=l[i];inv[mp[r[i]]]=r[i];
		l[i]=mp[l[i]];r[i]=mp[r[i]];
	}
	for(int i=n;i>=1;i--){
		int pos=query(0,SIZE-1,1,l[i]);
		if(pos<=n){
			f[i]=f[pos]+(inv[l[pos]]-inv[l[i]]+k)%k;
		}else{
			f[i]=0;
		}
		if(l[i]<=r[i]){
			rc(0,SIZE-1,1,r[i]+1,tot,i);
			rc(0,SIZE-1,1,1,l[i]-1,i);
		}else{
			rc(0,SIZE-1,1,r[i]+1,l[i]-1,i);
		}
	}
	ll ans=(ll)inf*inf;
	for(int i=1;i<=tot;i++){
		int pos=query(0,SIZE-1,1,i);
		if(pos>tot)ans=0;
		else ans=min(ans,(ll)f[pos]+(inv[l[pos]]-inv[i]+k)%k);
	}
	ans+=a[n]*2ll;
	cout<<ans<<endl;
	return 0;
}
