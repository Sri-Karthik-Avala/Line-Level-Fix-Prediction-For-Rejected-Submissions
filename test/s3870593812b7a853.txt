const int MAXN = 2e3 + 6;
string s;
int r[MAXN];
int p[MAXN];
int n, m;

ll dp[MAXN][MAXN];

int ones(int l, int r){
	return p[r] - p[l - 1];
}

ll solve(int l, int c){
	if(l == n && c <= 1){
		return dp[l][c] = 1;
	}

	if(dp[l][c] != 0) return dp[l][c];

	int nc = c + ones(r[l] + 1, r[l + 1]);
	int len = r[l] - l + 1;
	if(len > c) {
		dp[l][c] += solve(l + 1, nc);
	}
	if(c > 0) dp[l][c] += solve(l + 1, nc - 1);

	return dp[l][c] % MOD;
}

int main() {
	#ifdef LOCAL
	setIn("in.txt");
	setOut("out.txt");
	#endif

	cin >> n >> m;

	cin >> s;
	s = "#" + s;

	p[1] = (s[1] == '1');
	for(int i = 2; i <= n; i++){
		p[i] = p[i - 1] + (s[i] == '1');
	}

	for(int i = 1; i <= n; i++){
		r[i] = i;
	}

	for(int i = 0; i < m; i++){
		int l, x;
		cin >> l >> x;

		r[l] = max(r[l], x);
	}

	for(int i = 2; i <= n; i++){
		r[i] = max(r[i], r[i - 1]);
	}

	memset(dp, 0, sizeof dp);
	cout << solve(1, ones(1, r[1]));
}
