using namespace std;
using namespace boost;
int main() {
    int age = 0;
    while (true) {
        int n; cin >> n;
        if (n == 0) break;
        vector<point<double> > p(n); vector<double> r(n), m(n);
        for (int i : irange(0,n)) cin >> p[i] >> r[i] >> m[i];
        if (n <= 2) {
            cout << n << endl;
            continue;
        }
        vector<circle<double> > cs(2*n);
        for (int i : irange(0,n)) {
            cs[2*i].p = p[i];
            cs[2*i].r = r[i];
            cs[2*i+1].p = p[i];
            cs[2*i+1].r = r[i]+m[i];
        }
        vector<line<double> > ls;
        for (int i : irange(0,2*n)) {
            for (int j : irange(i+1,2*n)) {
                for (auto l : common_tangents(cs[i], cs[j])) {
                    ls.push_back(l);
                }
            }
        }
        int result = 0;
        for (auto l : ls) {
            int jewels = 0;
            for (int i : irange(0,n)) {
                double d = distance(l, p[i]);
                if (r[i] < d + eps and d < r[i]+m[i] + eps) {
                    jewels += 1;
                }
            }
            result = max(result, jewels);
        }
        cout << result << endl;
    }
    return 0;
}
