#include<bits/stdc++.h>
using namespace std;

int a[8][1000];
int main(){
  int b;
  while(cin>>b,b){
      memset(a,0,sizeof(a));
      for(int c=0;c<b;c++){
	  a[c][0]=1;
	  int d,e;cin>>d>>e;
	  for(int f=0;f<=e;f++){
	      for(int g=max(f*d,1);g<=1000;g++){
		  a[c+1][g]+=a[c][g-f*d];
		  }
	      }
	  }
	  int s;cin>>s;
	  for(int n=0;n<s;n++){
	      int w;cin>>w;
	      cout<<a[b][w]<<endl;
	      }
      }
}
