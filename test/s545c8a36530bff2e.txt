#include<iostream>
using namespace std;
int x[25],y[25],n,m,p,c;
int main(){
	while(true){
		cin>>n>>m;if(n==0)break;
		for(int i=0;i<25;i++)x[i]=0;
		x[0]=n;p++;c=0;
		while(x[m]<1){
			c++;for(int i=0;i<m;i++)y[i]=0;
			for(int i=0;i<m;i++){
				y[i]+=x[i]%2;
				y[0]+=x[i]/2;
				y[i+1]+=x[i]/2;
			}
			for(int i=0;i<=m;i++){x[i]=y[i];}
		}
		cout<<"Case "<<p<<": "<<c<<endl;
	}
	return 0;
}
