#include<cstdio>
#include<algorithm>
#define mod 998244353ll
int k,n;
long long inv[100005],fac[100005];
long long fp(long long x,int y){return (y&1?x:1ll)*(y/2?fp(x*x%mod,y/2):1ll)%mod;}
void pre(int n){inv[0]=fac[0]=1;for (int i=1;i<=n;i++) fac[i]=fac[i-1]*i%mod,inv[i]=fp(fac[i],mod-2);}
long long c(int x,int y){return y>x?0:fac[x]*inv[y]%mod*inv[x-y]%mod;}
int main()
{
	scanf("%d%d",&k,&n);
	pre(10000);
	for (int i=2;i<=2*k;i++)
	{
		int tot=0;long long ans=0;
		for (int j=1;i-j>=j;j++) tot=tot+(j<=k&&i-j<=k);
		for (int j=0;j<=std::min(tot,n/2);j++) ans=(ans+1ll*(1-(j&1)*2)*c(tot,j)%mod*c(n-2*j+k-1,k-1)%mod);
		printf("%lld\n",(ans+mod)%mod);
	}
	return 0;
}
