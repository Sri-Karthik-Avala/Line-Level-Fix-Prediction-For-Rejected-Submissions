//最大公約数
ll gcd(ll a, ll b){
    if(a % b == 0)return b;
    else return gcd(b, a % b);
}

ll lcm(ll a, ll b){
    return a * b / gcd(a, b);
}

int main(){
    int n, m;
    cin >> n >> m;
    string s, t;
    cin >> s >> t;
    int len = lcm(n, m);
    
    int nn = n / gcd(n, m), mm = m / gcd(n, m);

        }
    }

}
