#include <bits/stdc++.h>
#define REP(i,n) for(int i=0; i<(int)(n); ++i)

using namespace std;

typedef long long LL;

LL extgcd(LL a, LL b, LL& x, LL& y){
    LL d = a;
    if(b != 0){
        d = extgcd(b, a % b, y, x);
        y -= (a / b) * x;
    }else{
        x = 1; y = 0;
    }
    return d;
}

LL mul_mod(LL a, LL b, LL mod){
    if(b == 0) return 0;
    LL res = mul_mod((a + a) % mod, b / 2, mod);
    if(b & 1) res = (res + a) % mod;
    return res;
}

LL pow_mod(LL a, LL b, LL mod){
    if(b == 0) return 1;
    LL res = pow_mod(mul_mod(a, a, mod), b / 2, mod);
    if(b & 1) res = mul_mod(res, a, mod);
    return res;
}

LL inv_mod(LL a, LL mod){
    LL x, y;
    extgcd(a, mod, x, y);
    return (x % mod + mod) % mod;
}

vector<LL> inverse_list(int n, int p){
    vector<LL> inv(n + 1);
    inv[1] = 1;
    for (int i = 2; i <= n; ++i){
        inv[i] = inv[p % i] * (p - p / i) % p;
    }
    return inv;
}

pair<LL, LL> linear_congruence(const vector<LL>& A, const vector<LL>& B, const vector<LL>& M) {
    LL x = 0, m = 1;

    for(int i = 0; i < A.size(); i++) {
        LL a = A[i] * m, b = B[i] - A[i] * x, d = __gcd(M[i], a);
        if(b % d != 0) return make_pair(0, -1); // no solution
        LL t = b / d * inv_mod(a / d , M[i] / d) % (M[i] / d);
        x = x + m * t;
        m *= M[i] / d;
    }
    return make_pair(x % m, m);
}

int main(){
    LL N, M, D;
    while(cin >> N >> M >> D) {

        LL A[10];
        REP(i, M) cin >> A[i];

        for(int t = 0; t < D; t++) {
            vector<LL> a, b, m;
            REP(i, M) {
                LL y;
                cin >> y;
                if(y == -1) continue;
                a.push_back(1);
                b.push_back(y);
                m.push_back(A[i]);
            }

            pair<LL, LL> p = linear_congruence(a, b, m);

            if(p.first > N || p.second == -1) {
                N = -1;
            } else {
                LL k = (N - p.first) / p.second;
                assert(p.first + p.second * k <= N);
                assert(p.first + p.second * (k + 1) > N);
                N = p.first + p.second * k;
            }
            // cout << N << endl;
            
        }

        cout << N << endl;
    }
    return 0;

}
