#include <cstdio>

typedef long long ll;
ll n,l;
int len;
int k[100],pl[100];

ll abs(ll a){
	if(a<0)return -a;
	return a;
}

int main(void){
	while(1){
		scanf("%lld",&n);
		if(n==0)break;

		int i=0;
		while(n!=0){
			if(i%2==0){
				if(n>0)k[i]=abs(n)%10;
				else{
					k[i]=10-abs(n)%10;
					if(abs(n)%10==0)k[i]=0;
				}
				n-=k[i];
			}else{
				if(n<0)k[i]=abs(n)%10;
				else k[i]=10-abs(n%10);
				n+=k[i];
			}
			i++;
			n/=10;
		}
		for(int j=i-1;j>=0;j--){
			printf("%d",k[j]);
		}
		printf("\n");
	}
	return 0;
}
