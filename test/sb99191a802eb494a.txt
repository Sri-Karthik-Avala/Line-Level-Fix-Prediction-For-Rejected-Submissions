#!/usr/bin/python

import re

def main():
    key = raw_input().split('.')[1:]
    indent = 0
    level = 0
    while True:
        try:
            s = raw_input()
        except EOFError:
            break

        if re.match(' *'+' '*indent+key[level]+':', s):
            indent = len(re.match(' *', s).group())
            level += 1
            if len(key) == level:
                t = s[len(key[level-1])+indent+2:]
                if not t:
                    print 'object'
                else:
                    print 'string "'+t+'"'
                return

            t = s[len(key[level-1])+indent+2:]
            if t:
                break

        elif not s.startswith(' '*indent):
            break

        space = len(re.match(' *', s).group())
        if space < indent:
            break

    print 'no such property'

if __name__ == '__main__':
    main()
