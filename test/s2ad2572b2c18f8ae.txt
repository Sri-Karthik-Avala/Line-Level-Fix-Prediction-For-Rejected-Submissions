#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
	
	int n;
	cin>>n;
	vector<ll> A(n),pre(n);
	for(int i=0;i<n;i++)cin>>A[i];
	for(int i=0;i<n;i++)pre[i]=(i==0?A[i]:pre[i-1]+A[i]);

	vector<vector<ll>> dp(n,vector<ll>(n,INT_MAX));
		for(int i=0;i<n;i++)dp[i][i]=0;
	for(int i=n-1;i>=0;i--)
	for(int j=i;j<n;j++)
	{
		for(int k=i;k<j;k++)
		{
			dp[i][j]=min(dp[i][j], dp[i][k]+dp[k+1][j]+ pre[j]-pre[i]+A[i]);
		}
		
		
	}
	cout<<dp[0][n-1]<<endl;
}
