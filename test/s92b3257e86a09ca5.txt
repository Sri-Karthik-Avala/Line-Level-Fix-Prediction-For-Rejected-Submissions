#define CLR(mat) memset(mat, 0, sizeof(mat))
typedef long long ll;
const ll sqrtN = 512;
struct SqrtDecomposition {
  ll N, K;
  vector<ll> data;
  vector<ll> bucketSum;
  vector<ll> lazyUpdate;
  SqrtDecomposition(ll n) : N(n) {
    K = (N + sqrtN - 1) / sqrtN;
    data.assign(K * sqrtN, 0);
    bucketSum.assign(K, 0);
    lazyUpdate.assign(K, 1111);
  }
  void eval(ll k) {
    if (lazyUpdate[k] != 1111) {
      for (ll i = k * sqrtN; i < (k + 1) * sqrtN; ++i) {
        data[i] = lazyUpdate[k];
      }
      lazyUpdate[k] = 1111;
    }
  }
  // [s, t)
  void update(ll s, ll t, ll x) {
    for (ll k = 0; k < K; ++k) {
      ll l = k * sqrtN, r = (k + 1) * sqrtN;
      if (r <= s || t <= l)
        continue;
      if (s <= l && r <= t) {
        bucketSum[k] = x * sqrtN;
        lazyUpdate[k] = x;
      } else {
        eval(k);
        for (ll i = max(s, l); i < min(t, r); ++i) {
          bucketSum[k] = x - data[i];
          data[i] = x;
        }
      }
    }
  }
  // [s, t)
  ll find(ll s, ll t) {
    ll ans = 0;
    for (ll k = 0; k < K; ++k) {
      ll l = k * sqrtN, r = (k + 1) * sqrtN;
      if (r <= s || t <= l)
        continue;
      if (s <= l && r <= t) {
        ans += bucketSum[k];
      } else {
        eval(k);
        for (ll i = max(s, l); i < min(t, r); ++i) {
          ans += data[i];
        }
      }
    }
    return ans;
  }
};

int main()
{
  ios::sync_with_stdio(false);
  cin.tie(0);
  ll n, q;
  cin >> n >> q;
  SqrtDecomposition sd(n);
  FOR(i,0,q) {
    ll f;
    cin >> f;
    if(f) {
      ll s,t;
      cin >> s >> t;
      cout << sd.find(s, t+1) << endl;
    } else {
      ll s,t,x;
      cin >> s >> t >> x;
      sd.update(s,t+1,x);
    }
  }
}
