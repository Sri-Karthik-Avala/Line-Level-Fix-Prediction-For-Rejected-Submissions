#include<iostream>
#include<string>
#include<algorithm>
using namespace std;
string S[101]; int x[101], dp[101];
int main() {
	int n; cin >> n;
	for (int i = 0; i < n; i++) {
		S[i] = "";
		while (S[i] == "") {
			getline(cin, S[i]);
		}
		S[i] = S[i].substr(0, S[i].size() - 1);
	}
	for (int i = 0; i < n; i++) {
		int OK = 0;
		if (S[i].size() >= 4) {
			if (S[i].substr(0, 4) == "goto") {
				OK = 1;
				for (int j = 0; j < n; j++) {
					string T = S[i].substr(5, S[i].size() - 5);
					if (T == S[j]) {
						x[i] = j;
					}
				}
			}
		}
		if (OK == 0)x[i] = i;
	}
	for (int i = 0; i < 101; i++)dp[i] = 1 << 30;
	dp[0] = 0;
	for (int i = 0; i < n; i++) {
		for (int j = 0; j < n; j++) {
			dp[j + 1] = min(dp[j + 1], dp[j] + 1);
			dp[x[j] + 1] = min(dp[j + 1], dp[j]);
		}
	}
	cout << dp[n] << endl;
	return 0;
}
