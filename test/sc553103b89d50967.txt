#include <iostream>
#include <cstdio>
#include <cstring>
using namespace std;
int main()
{
	int i,j,k,lt,lb,ans;
	char t[1001],b[1001];
	static long long int dp[1001][1001];
	cin>>t>>b;
	lb=strlen(b);
	lt=strlen(t);
	for (i=0;i<lt;i++)
	{
		for (j=0;j<lb;j++)
		{
			dp[i][j]=1;
		}
	}
	//昇
	j=0;
	for (i=0;i<lt;i++)
	{
		if (t[i]==b[j])
		{
			//printf(",%d(%d)",i,j);
			for (k=0;k<i;k++)
			{
				dp[k][j]=0;//バツ
			}
			j++;
		}
	}
	//降
	j=lb-1;
	for (i=lt-1;i>=0;i--)
	{
		if (t[i]==b[j])
		{
			//printf(",%d(%d)",i,j);
			for (k=i+1;k<lt;k++)
			{
				dp[k][j]=0;
			}
			j--;
		}
	}
	//一列目
	for (i=0;i<lt;i++)
	{
		if (t[i]==b[0]&&dp[i][0]) dp[i][0]=1;
		else dp[i][0]=0;
	}
	//bが1文字でないとき
	for (j=1;j<lb;j++)
	{
		for (i=0;i<lt;i++)
		{
			if (t[i]==b[j]&&dp[i][j])
			{
				dp[i][j]=0;
				for (k=0;k<i;k++)
				{
					dp[i][j]+=dp[k][j-1]%1000000007;
				}
				dp[i][j]=dp[i][j]%1000000007;
			} else {
				dp[i][j]=0;
			}
		}
	}
	ans=0;
	for (i=0;i<lt;i++)
	{
		ans+=dp[i][lb-1]%1000000007;
	}
	ans=ans%1000000007;
	cout<<ans<<endl;
	return 0;
}
