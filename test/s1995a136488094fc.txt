#include <bits/stdc++.h>
using namespace std;
#define mp(a, b) make_pair(a, b)
typedef pair<int, int> pii; 
typedef long long ll;
const int maxn = 2e5 + 10;
int n, m;
int a[maxn];
vector<pii> b, p;
ll ans, tmp, cnt;
int main()
{
	scanf("%d%d", &n, &m);
	for(int i = 1; i <= n; ++i) scanf("%d", &a[i]);
	for(int i = 2; i <= n; ++i) 
	{
		if(a[i - 1] < a[i])
			b.emplace_back(a[i - 1], a[i]), b.emplace_back(a[i - 1] + m, a[i] + m);
		else
			b.emplace_back(a[i - 1], a[i] + m);
	}
	for(int i = 0; i < b.size(); ++i)
	{
		p.emplace_back(b[i].first, i + 1);
		p.emplace_back(b[i].second, -(i + 1));
		if(b[i].first <= m && b[i].second >= m) p.emplace_back(b[i].first + m, i + 1);
	}
	sort(p.begin(), p.end());
	for(auto v : p)
	{
		if(v.second > 0) 
		{
			++cnt;
			tmp += v.first;
		}
		else
		{
			ans = max(ans, (v.first - 1) * cnt - tmp);
			tmp -= b[-(v.second + 1)].first;
			--cnt;
		}
			 
	}
	ans = -ans;
	for(int i = 2; i <= n; ++i) 
		if(a[i - 1] < a[i])
			ans += a[i] - a[i - 1];
		else
			ans += a[i] + m - a[i - 1];
	printf("%lld\n", ans);
	return 0;
}
