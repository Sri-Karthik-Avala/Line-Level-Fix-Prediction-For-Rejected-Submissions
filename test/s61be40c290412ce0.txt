void visit(const vector<vi> & g, int v, int u,
    vector< vector<int> >& tecomp,
    stack<int>& roots, stack<int>& S, vector<bool>& inS,
    vector<int>& num, int& time) {
  num[v] = ++time;
  S.push(v); inS[v] = true;
  roots.push(v);
  for(int w : g[v]) {
    if (num[w] == 0)
      visit(g, w, v, tecomp, roots, S, inS, num, time);
    else if (u != w && inS[w])
      while (num[roots.top()] > num[w]) roots.pop();
  }
  if (v == roots.top()) {
    tecomp.push_back(vector<int>());
    while (1) {
      int w = S.top(); S.pop(); inS[w] = false;
      tecomp.back().push_back(w);
      if (v == w) break;
    }
    roots.pop();
  }
}
vector<vi> calc_te(const vector<vi>& g) {
  const int n = g.size();
  vector<int> num(n);
  vector<bool> inS(n);
	vector<vi> tecomp;
  stack<int> roots, S;
  int time = 0;
  rep(u, n) if (num[u] == 0) {
    visit(g, u, n, tecomp, roots, S, inS, num, time);
  }
	return tecomp;
}
