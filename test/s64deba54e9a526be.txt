template<class T> T div_floor(T a, T b){
    if(b < 0) a *= -1, b *= -1;
    return a>=0 ? a/b : (a+1)/b-1;
}
template<class T> T div_ceil(T a, T b){
    if(b < 0) a *= -1, b *= -1;
    return a>0 ? (a-1)/b+1 : a/b;
}

constexpr lint mod = 1000000007;

template<typename T> struct BinaryIndexedTree{
    vector<T> node;

    BinaryIndexedTree(int n){
        node.resize(n+1, {});
    }

    void update(int i, T x){
        ++i;
        while(i < (int)node.size()){
            node[i] += x;
            i += (i & -i);
        }
    }

    T query(int i){
        ++i;
        lint ret = 0;
        while(i){
            ret += node[i];
            i -= (i & -i);
        }
        return ret;
    }

    T query(int l, int r){
        return query(r-1) - query(l-1);
    }
};

int main(){
    string s;
    cin >> s;
    vector<int> pos[26];
    int n = s.size();
    rep(i, n) pos[s[i] - 'a'].push_back(i);
    int cnt = count_if(pos, pos+26, [](vector<int> v){return v.size() % 2 == 1;});
    if((n%2 ==0 && cnt) || (n%2 == 1 && cnt != 1)){
        puts("-1");
        return 0;
    }

    vector<pii> seg;
    rep(i, 26)if(pos[i].size() >= 2){
        rep(k, pos[i].size()/2) seg.emplace_back(pos[i][k], pos[i][pos[i].size()-1-k]);
    }
    sort(seg.begin(), seg.end());
    vector<pii> v;
    rep(i, seg.size()){
        v.emplace_back(seg[i].fi, i);
        v.emplace_back(seg[i].se, n-1-i);
    }
    rep(i, 26)if(pos[i].size() == 1) v.emplace_back(pos[i][0], n/2);
    sort(v.begin(), v.end());

    BinaryIndexedTree<lint> bit(n);
    lint ans = 0;
    for(auto p: v){
        ans += bit.query(p.se+1, n);
        bit.update(p.se, 1);
    }
    printf("%lld\n", ans);
}
