const int MAX_N = 500001;

int two[10],dp[MAX_N];

int solve(const vi& b)
{
    int n = len(b);
    int sm = accumulate(all(b),0);
    rep(i,sm+1){
        dp[i] = -1;
    }
    dp[0] = 0;
    rep(i,n){
        for(int j = sm/b[i]*b[i]; j >= b[i]; j -= b[i]){
            if(dp[j-b[i]] >= 0){
                cmx(dp[j],dp[j-b[i]]+1);
            }
        }
    }
    int res = 0, nw = 1;
    while(nw <= sm){
        cmx(res,dp[nw]);
        nw *= 2;
    }
    return res;
}

        vi a(n);
        rep(i,n){
            cin >> a[i];

            vi b;
            bool ok = false;
            rep(j,n){

                    rep(k,9){
                        if(i*two[k] == a[j]){
                            b.pb(two[k]);
                            if(!j) ok = true;
                        }
                    }
                }
            }
            if(ok) cmx(ans,solve(b));
        }
    }

}
