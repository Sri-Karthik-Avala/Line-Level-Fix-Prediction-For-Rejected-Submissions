#include <bits/stdc++.h>
typedef long long LL;
#define FOR(i,a,b) for(int i=(a);i<(b);++i)
#define REP(i,n) FOR(i,0,n)

using namespace std;

int main(void)
{
  int n;
  cin >> n;
  vector<int> a(n);
  REP(i,n) cin >> a[i];
  map<int, int> cnt;
  REP(i,n) {
    --cnt[a[i]];
    if((cnt.find(a[n-1-i])!=cnt.end()) && cnt[a[n-1-i]]==1) cnt.erase(a[n-1-i]);
    else --cnt[a[n-1-i]];
    if(cnt.size() == 0) cout << i+1 << endl;
  }
  return 0;
}
