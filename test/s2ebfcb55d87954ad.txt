H,W=map(int, input().split())
ans = float("inf")

#横切れ2
if H%3==0:
    print(0)
    exit()
else:
    ans = min(ans, W)

#タテ2
if W%3==0:
    print(0)
    exit()
else:
    ans = min(ans, H)

for i in range(1, W):
    s1 = H*i
    s2 = (W-i)*(H//2)
    s3 = (W-i)*(H//2 + H%2)
    tmp = max(s1,s2,s3) - min(s1,s2,s3)
    ans = min(ans, tmp)

for i in range(1, H):
    s1 = W*i
    s2 = (W-i)*(W//2)
    s3 = (W-i)*(W//2 + W%2)
    tmp = max(s1,s2,s3) - min(s1,s2,s3)
    ans = min(ans, tmp)

print(ans)
