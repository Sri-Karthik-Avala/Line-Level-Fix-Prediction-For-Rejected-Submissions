import sys
from itertools import permutations

read = sys.stdin.read
readline = sys.stdin.readline
readlines = sys.stdin.readlines
sys.setrecursionlimit(10 ** 9)
INF = 1 << 60
MOD = 1000000007


def main():
    M, K = map(int, readline().split())

    if M == 0:
        if K == 0:
            print(0, 0)
            return
        else:
            print(-1)
            return
    elif M == 1:
        if K == 0:
            print(0, 0, 1, 1)
            return
        else:
            print(-1)
            return
    elif pow(2, M) - 1 < K:
        print(-1)
        return

    vec = [i for i in range(pow(2, M)) if i != K]
    ans = vec + [K] + vec[::-1] + [K]

    print(ans)
    return


if __name__ == '__main__':
    main()
