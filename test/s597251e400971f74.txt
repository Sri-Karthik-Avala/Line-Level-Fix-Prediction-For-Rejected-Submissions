n = int(input())
f = ['']*(n+2)


def question(a):
    print('?', *a)
    response = input()
    if response == "-1":
        exit()
    return response


def chk(x):
    a = [x+j for j in range(n)]
    if f[x] == '':
        f[x] = question(a)
    return f[x]


bottom = 1
top = n+1
f[bottom] = chk(bottom)
f[top] = chk(top)

while top - bottom > 1:
    middle = (top + bottom)//2
    if chk(middle) == f[top]:
        top = middle
    else:
        bottom = middle

a = [bottom+j for j in range(1, n)]
b = []
for x in range(1, 2*n+1):
    if x not in a and x != bottom and x != bottom+n:
        b.append(x)

print(a, b)

ans = []
for x in range(1, 2*n+1):
    if x in a:
        c = b+[x]
    else:
        c = a+[x]
    ans.append(question(c)[0])

print('!', ''.join(ans))
